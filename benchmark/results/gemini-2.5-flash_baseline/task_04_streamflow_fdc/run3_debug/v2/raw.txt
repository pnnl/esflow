import xarray as xr
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import cartopy.crs as ccrs
import cartopy.feature as cfeature
from scipy.stats import wasserstein_distance
import os
import glob
import warnings

# --- Configuration ---
DATA_DIR = "./data/sample/e3sm/"
OBS_DIR = "./data/sample/obs/"
CASE_NAME = "sample.v3.LR.historical"
MOSART_VAR = "RIVER_DISCHARGE_OVER_LAND_LIQ"
START_DATE = "1985-01-01"
END_DATE = "1989-12-31"

# User-specified output directory (absolute path)
OUTPUT_ROOT_DIR = "/Users/zhou014/Local_Drive/Git_repo/esmflow/benchmark/results/gemini-2.5-flash_baseline/task_04_streamflow_fdc/run3_output"
# The script will create a subdirectory for this task within the OUTPUT_ROOT_DIR
OUTPUT_DIR = os.path.join(OUTPUT_ROOT_DIR, "task04_streamflow_fdc")

# Specific gauges for FDC plotting (gauge_id)
PLOT_GAUGES = [
    3629000, # Amazon
    4121801, # Missouri
    4115200, # Columbia
    6742900, # Danube
    2969100, # Mekong
    1159100  # Orange
]

# --- Helper Functions ---
def create_output_dir(path):
    """Creates the output directory if it doesn't exist."""
    try:
        os.makedirs(path, exist_ok=True)
        print(f"Output directory created: {path}")
    except Exception as e:
        print(f"Error creating output directory {path}: {e}")
        raise

def calculate_fdc_points(data):
    """
    Calculates Flow Duration Curve points (exceedance probability vs discharge).
    Data should be a 1D array of discharge values.
    """
    if not isinstance(data, np.ndarray):
        data = np.asarray(data)
    
    if data.size == 0:
        return np.array([]), np.array([])
    
    # Remove NaN values before sorting
    data = data[~np.isnan(data)]
    if data.size == 0:
        return np.array([]), np.array([])

    sorted_data = np.sort(data)[::-1] # Descending order
    n = len(sorted_data)
    # Exceedance probability: (rank / (N+1)) * 100
    # Using (rank - 0.5) / N * 100 or rank / (N+1) * 100 are common.
    # (rank / (N+1)) * 100 ensures 0 and 100 are not included, which is good for log plots.
    exceedance_probabilities = np.arange(1, n + 1) / (n + 1) * 100
    return exceedance_probabilities, sorted_data

# --- Main Script ---
def main():
    create_output_dir(OUTPUT_DIR)

    # 1. Load Gauge Metadata and MOSART Grid Info
    print("Loading gauge metadata and MOSART grid...")
    try:
        gauge_metadata_path = os.path.join(OBS_DIR, "gauge_metadata.csv")
        gauge_metadata = pd.read_csv(gauge_metadata_path)
    except Exception as e:
        print(f"Error loading gauge metadata from {gauge_metadata_path}: {e}")
        return

    # Load a sample MOSART h0 file to get grid info (lat/lon arrays)
    sample_mosart_h0_files = glob.glob(os.path.join(DATA_DIR, "rof", f"{CASE_NAME}.mosart.h0.*.nc"))
    if not sample_mosart_h0_files:
        print(f"Error: No MOSART h0 files found in {os.path.join(DATA_DIR, 'rof')}. Cannot determine grid.")
        return
    
    try:
        with xr.open_dataset(sample_mosart_h0_files[0]) as ds_grid:
            mosart_lats = ds_grid['lat'].values
            mosart_lons = ds_grid['lon'].values
    except Exception as e:
        print(f"Error loading MOSART grid from {sample_mosart_h0_files[0]}: {e}")
        return

    # Match gauges to MOSART grid
    matched_gauges_info = []
    for idx, gauge in gauge_metadata.iterrows():
        gauge_lat = gauge['lat']
        gauge_lon = gauge['lon']

        # Convert gauge_lon to 0-360 if it's -180 to 180, to match E3SM's typical 0-360 convention
        # MOSART lons are typically 0-360.
        gauge_lon_360 = gauge_lon if gauge_lon >= 0 else gauge_lon + 360

        lat_idx = np.argmin(np.abs(mosart_lats - gauge_lat))
        lon_idx = np.argmin(np.abs(mosart_lons - gauge_lon_360)) # Use 0-360 for matching

        matched_gauges_info.append({
            'gauge_id': gauge['gauge_id'],
            'lat': gauge['lat'],
            'lon': gauge['lon'],
            'river_name': gauge['river_name'],
            'mosart_lat_idx': lat_idx,
            'mosart_lon_idx': lon_idx,
            'mosart_lat_val': mosart_lats[lat_idx],
            'mosart_lon_val': mosart_lons[lon_idx]
        })
    matched_gauges_df = pd.DataFrame(matched_gauges_info)
    print(f"Matched {len(matched_gauges_df)} gauges to MOSART grid.")

    # 2. Load E3SM MOSART Daily Data
    print(f"Loading E3SM MOSART daily data for {START_DATE} to {END_DATE}...")
    mosart_h1_files = []
    for year in range(int(START_DATE[:4]), int(END_DATE[:4]) + 1):
        mosart_h1_files.extend(sorted(glob.glob(os.path.join(DATA_DIR, "rof", f"{CASE_NAME}.mosart.h1.{year}*.nc"))))
    
    # Filter files to the exact date range
    start_dt = pd.to_datetime(START_DATE)
    end_dt = pd.to_datetime(END_DATE)
    
    filtered_mosart_h1_files = []
    for f in mosart_h1_files:
        try:
            file_date_str = os.path.basename(f).split('.')[4] # e.g., '1985-01-02-00000'
            # FIX: Extract only the YYYY-MM-DD part for robust parsing
            # The format is YYYY-MM-DD-HHMMSS, so splitting by '-' and taking first 3 parts works.
            date_part = '-'.join(file_date_str.split('-')[:3])
            file_date = pd.to_datetime(date_part)
            if start_dt <= file_date <= end_dt:
                filtered_mosart_h1_files.append(f)
        except ValueError:
            print(f"Warning: Could not parse date from filename {f}. Skipping.")
            continue
    mosart_h1_files = filtered_mosart_h1_files

    if not mosart_h1_files:
        print(f"Error: No MOSART h1 files found for {START_DATE}-{END_DATE} in {os.path.join(DATA_DIR, 'rof')}")
        return

    mosart_discharge_data = {}
    try:
        # Suppress warnings about inconsistent time units or non-monotonic time
        with warnings.catch_warnings():
            warnings.simplefilter("ignore")
            ds_mosart = xr.open_mfdataset(mosart_h1_files, combine='by_coords', decode_times=True, parallel=False)
        
        # Ensure time is sorted and unique, then select the desired period
        ds_mosart = ds_mosart.sel(time=slice(START_DATE, END_DATE))
        ds_mosart = ds_mosart.sortby('time')
        ds_mosart = ds_mosart.sel(time=~ds_mosart.indexes['time'].duplicated())

        for _, gauge_row in matched_gauges_df.iterrows():
            gauge_id = gauge_row['gauge_id']
            lat_idx = gauge_row['mosart_lat_idx']
            lon_idx = gauge_row['mosart_lon_idx']
            
            # Extract discharge for the matched grid cell using .isel for integer indexing
            discharge_series = ds_mosart[MOSART_VAR].isel(lat=lat_idx, lon=lon_idx).to_series()
            mosart_discharge_data[gauge_id] = discharge_series.dropna() # Remove NaNs if any

    except Exception as e:
        print(f"Error loading or processing E3SM MOSART data: {e}")
        return

    # 3. Process Observations and Compute Metrics
    print("Processing observations and computing FDC metrics...")
    all_metrics = []
    all_fdc_data = [] # Stores FDC points for plotting/saving

    for _, gauge_row in matched_gauges_df.iterrows():
        gauge_id = gauge_row['gauge_id']
        river_name = gauge_row['river_name']
        gauge_lat = gauge_row['lat']
        gauge_lon = gauge_row['lon']

        obs_file_path = os.path.join(OBS_DIR, "streamflow", f"{gauge_id}.csv")
        
        try:
            # Load observed data
            df_obs = pd.read_csv(obs_file_path, parse_dates=['date'], index_col='date')
            df_obs = df_obs.loc[START_DATE:END_DATE]
            obs_discharge = df_obs['discharge_m3s'].dropna()

            if obs_discharge.empty:
                print(f"Warning: No observed data for gauge {gauge_id} ({river_name}) in {START_DATE}-{END_DATE}. Skipping.")
                continue

            # Get simulated data for this gauge
            sim_discharge = mosart_discharge_data.get(gauge_id)
            if sim_discharge is None or sim_discharge.empty:
                print(f"Warning: No simulated data for gauge {gauge_id} ({river_name}). Skipping.")
                continue
            
            # Align time series for volume bias calculation
            common_index = obs_discharge.index.intersection(sim_discharge.index)
            if common_index.empty:
                print(f"Warning: No common dates for gauge {gauge_id} ({river_name}). Skipping.")
                continue
            
            obs_aligned = obs_discharge.loc[common_index]
            sim_aligned = sim_discharge.loc[common_index]

            if obs_aligned.empty or sim_aligned.empty:
                print(f"Warning: Aligned data is empty for gauge {gauge_id} ({river_name}). Skipping.")
                continue

            # Calculate FDC points for plotting/saving
            obs_fdc_exceedance, obs_fdc_discharge = calculate_fdc_points(obs_discharge.values)
            sim_fdc_exceedance, sim_fdc_discharge = calculate_fdc_points(sim_discharge.values)

            # Store FDC data
            all_fdc_data.append({
                'gauge_id': gauge_id,
                'river_name': river_name,
                'obs_fdc_exceedance': obs_fdc_exceedance,
                'obs_fdc_discharge': obs_fdc_discharge,
                'sim_fdc_exceedance': sim_fdc_exceedance,
                'sim_fdc_discharge': sim_fdc_discharge
            })

            # Calculate Metrics
            # Volume Bias
            obs_mean = obs_aligned.mean()
            sim_mean = sim_aligned.mean()
            volume_bias = ((sim_mean - obs_mean) / obs_mean) * 100 if obs_mean != 0 else np.nan

            # Wasserstein Distance (Earth Mover's Distance)
            # Use raw, non-aligned data for distribution comparison
            wd = wasserstein_distance(obs_discharge.values, sim_discharge.values)

            # Quantile Ratios (Q10, Q50, Q90)
            # Q10 (high flow) corresponds to 90th percentile
            # Q50 (median flow) corresponds to 50th percentile
            # Q90 (low flow) corresponds to 10th percentile
            q10_obs = np.percentile(obs_discharge.values, 90)
            q50_obs = np.percentile(obs_discharge.values, 50)
            q90_obs = np.percentile(obs_discharge.values, 10)

            q10_sim = np.percentile(sim_discharge.values, 90)
            q50_sim = np.percentile(sim_discharge.values, 50)
            q90_sim = np.percentile(sim_discharge.values, 10)

            q10_ratio = q10_sim / q10_obs if q10_obs != 0 else np.nan
            q50_ratio = q50_sim / q50_obs if q50_obs != 0 else np.nan
            q90_ratio = q90_sim / q90_obs if q90_obs != 0 else np.nan

            all_metrics.append({
                'gauge_id': gauge_id,
                'river_name': river_name,
                'lat': gauge_lat,
                'lon': gauge_lon,
                'volume_bias_percent': volume_bias,
                'wasserstein_distance': wd,
                'Q10_ratio': q10_ratio,
                'Q50_ratio': q50_ratio,
                'Q90_ratio': q90_ratio
            })

        except FileNotFoundError:
            print(f"Warning: Observation file not found for gauge {gauge_id} ({river_name}) at {obs_file_path}. Skipping.")
        except Exception as e:
            print(f"Error processing gauge {gauge_id} ({river_name}): {e}")

    metrics_df = pd.DataFrame(all_metrics)

    # 4. Save Output Data
    print("Saving output data...")
    try:
        metrics_output_path = os.path.join(OUTPUT_DIR, "fdc_metrics_per_gauge.csv")
        metrics_df.to_csv(metrics_output_path, index=False)
        print(f"Saved FDC metrics to {metrics_output_path}")
    except Exception as e:
        print(f"Error saving FDC metrics CSV: {e}")

    # Save FDC percentile data for all gauges
    try:
        combined_fdc_dfs = []
        for fdc_item in all_fdc_data:
            gauge_id = fdc_item['gauge_id']
            river_name = fdc_item['river_name']
            
            obs_df = pd.DataFrame({
                'exceedance_probability': fdc_item['obs_fdc_exceedance'],
                'discharge_obs': fdc_item['obs_fdc_discharge']
            })
            sim_df = pd.DataFrame({
                'exceedance_probability': fdc_item['sim_fdc_exceedance'],
                'discharge_sim': fdc_item['sim_fdc_discharge']
            })
            
            # Merge on exceedance_probability, handling different lengths
            merged_df = pd.merge(obs_df, sim_df, on='exceedance_probability', how='outer')
            merged_df['gauge_id'] = gauge_id
            merged_df['river_name'] = river_name
            combined_fdc_dfs.append(merged_df)
        
        if combined_fdc_dfs:
            all_fdc_output_df = pd.concat(combined_fdc_dfs, ignore_index=True)
            fdc_output_path = os.path.join(OUTPUT_DIR, "fdc_percentile_data_all_gauges.csv")
            all_fdc_output_df.to_csv(fdc_output_path, index=False)
            print(f"Saved FDC percentile data to {fdc_output_path}")
        else:
            print("No FDC data to save.")
    except Exception as e:
        print(f"Error saving FDC percentile data CSV: {e}")

    # 5. Plotting
    print("Generating plots...")
    try:
        # Filter FDC data for plotting gauges
        plot_fdc_data_filtered = [item for item in all_fdc_data if item['gauge_id'] in PLOT_GAUGES]
        
        # Sort plot_fdc_data to match PLOT_GAUGES order for consistent subplot arrangement
        plot_fdc_data_sorted = []
        for g_id in PLOT_GAUGES:
            for item in plot_fdc_data_filtered:
                if item['gauge_id'] == g_id:
                    plot_fdc_data_sorted.append(item)
                    break
        plot_fdc_data_filtered = plot_fdc_data_sorted

        # --- Map of Wasserstein Distance ---
        fig = plt.figure(figsize=(15, 10))
        ax = fig.add_subplot(1, 1, 1, projection=ccrs.PlateCarree())
        
        ax.set_extent([-180, 180, -90, 90], crs=ccrs.PlateCarree())
        ax.add_feature(cfeature.COASTLINE, linewidth=0.8)
        ax.add_feature(cfeature.BORDERS, linestyle=':', linewidth=0.5)
        ax.add_feature(cfeature.LAND, edgecolor='black', facecolor='lightgray')
        ax.add_feature(cfeature.OCEAN, facecolor='lightblue')
        ax.gridlines(draw_labels=True, dms=True, x_inline=False, y_inline=False)

        # Plot all gauges, color by Wasserstein distance
        if not metrics_df.empty:
            # Remove NaNs from Wasserstein distance for plotting
            plot_df = metrics_df.dropna(subset=['wasserstein_distance'])
            if not plot_df.empty:
                scatter = ax.scatter(plot_df['lon'], plot_df['lat'],
                                     c=plot_df['wasserstein_distance'],
                                     cmap='viridis', s=50, transform=ccrs.PlateCarree(),
                                     edgecolor='black', linewidth=0.5)
                cbar = plt.colorbar(scatter, ax=ax, orientation='vertical', shrink=0.7)
                cbar.set_label('Wasserstein Distance (m$^3$/s)')
                ax.set_title(f'Wasserstein Distance of Daily Discharge ({START_DATE}-{END_DATE})')
            else:
                ax.text(0.5, 0.5, "No valid Wasserstein distance data to plot.", transform=ax.transAxes, ha='center', va='center')
        else:
            ax.text(0.5, 0.5, "No metrics data to plot.", transform=ax.transAxes, ha='center', va='center')

        map_output_path = os.path.join(OUTPUT_DIR, "wasserstein_distance_map.png")
        plt.savefig(map_output_path, bbox_inches='tight', dpi=300)
        plt.close(fig)
        print(f"Saved Wasserstein distance map to {map_output_path}")

        # --- FDC Comparison Panels ---
        if plot_fdc_data_filtered:
            n_gauges = len(PLOT_GAUGES)
            n_cols = 2
            n_rows = (n_gauges + n_cols - 1) // n_cols # Ceiling division
            
            fig, axes = plt.subplots(n_rows, n_cols, figsize=(15, 5 * n_rows), sharex=True)
            axes = axes.flatten() # Flatten for easy iteration

            for i, fdc_item in enumerate(plot_fdc_data_filtered):
                ax = axes[i]
                gauge_id = fdc_item['gauge_id']
                river_name = fdc_item['river_name']
                obs_fdc_exceedance = fdc_item['obs_fdc_exceedance']
                obs_fdc_discharge = fdc_item['obs_fdc_discharge']
                sim_fdc_exceedance = fdc_item['sim_fdc_exceedance']
                sim_fdc_discharge = fdc_item['sim_fdc_discharge']

                ax.plot(obs_fdc_exceedance, obs_fdc_discharge, label='Observed', color='blue')
                ax.plot(sim_fdc_exceedance, sim_fdc_discharge, label='E3SM MOSART', color='red', linestyle='--')
                
                ax.set_title(f'{river_name} ({gauge_id})')
                ax.set_xlabel('Exceedance Probability (%)')
                ax.set_ylabel('Discharge (m$^3$/s)')
                ax.grid(True, linestyle=':', alpha=0.7)
                ax.legend()
                ax.set_yscale('log') # FDCs are often plotted on log scale for discharge
                ax.set_xlim(0, 100)

            # Hide any unused subplots
            for j in range(i + 1, len(axes)):
                fig.delaxes(axes[j])

            plt.suptitle(f'Flow Duration Curve Comparison ({START_DATE}-{END_DATE})', fontsize=16)
            plt.tight_layout(rect=[0, 0.03, 1, 0.95]) # Adjust layout to make space for suptitle
            fdc_output_path = os.path.join(OUTPUT_DIR, "fdc_comparison_panels.png")
            plt.savefig(fdc_output_path, bbox_inches='tight', dpi=300)
            plt.close(fig)
            print(f"Saved FDC comparison panels to {fdc_output_path}")
        else:
            print("No FDC data for specified plot gauges.")

    except Exception as e:
        print(f"Error during plotting: {e}")

if __name__ == "__main__":
    main()
