# --- Helper Functions ---

def download_ilamb_data(variable_key, config):
    """Downloads ILAMB data if not already present."""
    local_path = os.path.join(ILAMB_DATA_DIR, config[variable_key]["filename"])
    if not os.path.exists(local_path):
        print(f"Downloading {variable_key} from ILAMB...")
        try:
            response = requests.get(config[variable_key]["url_template"], stream=True)
            response.raise_for_status()
            with open(local_path, 'wb') as f:
                for chunk in response.iter_content(chunk_size=8192):
                    f.write(chunk)
            print(f"Downloaded {variable_key} to {local_path}")
        except requests.exceptions.RequestException as e:
            print(f"Error downloading {variable_key}: {e}")
            return None
    else:
        print(f"{variable_key} already exists at {local_path}")
    return local_path

def load_and_preprocess_ilamb(variable_key, config, time_slice):
    """Loads, preprocesses, and returns climatological mean for ILAMB data."""
    local_path = download_ilamb_data(variable_key, config)
    if not local_path:
        return None

    try:
        ds = xr.open_dataset(local_path)
        var_name_in_file = config[variable_key]["variable"]

        # Handle longitude convention
        if config[variable_key]["lon_convention"] == "0-360":
            if 'lon' in ds.coords and ds['lon'].max() > 180:
                ds['lon'] = xr.where(ds['lon'] > 180, ds['lon'] - 360, ds['lon'])
                ds = ds.sortby('lon')
        
        # Ensure 'time' dimension is present and slice
        if 'time' in ds.dims:
            ds = ds.sel(time=time_slice)
        else:
            print(f"Warning: No 'time' dimension found for {variable_key}. Cannot slice by time.")

        # Apply unit conversion and calculate climatological mean
        data_var = ds[var_name_in_file]
        
        # Calculate climatological mean, handling potential missing time dimension
        if 'time' in data_var.dims:
            climatology = data_var.mean(dim='time')
        else:
            climatology = data_var # Assume it's already a climatology or static field

        climatology = climatology * config[variable_key]["unit_conversion_factor"]

        # Set CRS for rioxarray clipping
        if 'lat' in climatology.dims and 'lon' in climatology.dims:
            climatology = climatology.rio.set_spatial_dims(x_dim="lon", y_dim="lat", inplace=True)
            climatology = climatology.rio.write_crs("epsg:4326", inplace=True)
        
        return climatology
    except Exception as e:
        print(f"Error processing ILAMB data for {variable_key}: {e}")
        return None

def calculate_fdc(discharge_series):
    """Calculates Flow Duration Curve (FDC) values."""
    if discharge_series.empty:
        return pd.Series([]), pd.Series([])
    sorted_discharge = discharge_series.sort_values(ascending=False).dropna()
    exceedance_probability = np.arange(1, len(sorted_discharge) + 1) / (len(sorted_discharge) + 1)
    return sorted_discharge, pd.Series(exceedance_probability, index=sorted_discharge.index)

    with xr.open_mfdataset(elm_files, combine='by_coords', decode_times=True) as ds_elm:
        ds_elm = ds_elm.sel(time=TIME_SLICE)

        # Unit conversion: kg/m2/s or mm/s to mm/day (multiply by 86400)
        # P: RAIN + SNOW
        model_p = (ds_elm['RAIN'] + ds_elm['SNOW']) * 86400
        model_p.attrs['units'] = 'mm/day'
        model_p.attrs['long_name'] = 'Precipitation'

        # ET: QVEGE + QVEGT + QSOIL
        model_et = (ds_elm['QVEGE'] + ds_elm['QVEGT'] + ds_elm['QSOIL']) * 86400
        model_et.attrs['units'] = 'mm/day'
        model_et.attrs['long_name'] = 'Evapotranspiration'

        # Q: QRUNOFF
        model_q = ds_elm['QRUNOFF'] * 86400
        model_q.attrs['units'] = 'mm/day'
        model_q.attrs['long_name'] = 'Total Runoff'

        # Climatological time-mean
        model_p_clim = model_p.mean(dim='time')
        model_et_clim = model_et.mean(dim='time')
        model_q_clim = model_q.mean(dim='time')

        # Set CRS for rioxarray clipping
        if 'lat' in model_p_clim.dims and 'lon' in model_p_clim.dims:
            model_p_clim = model_p_clim.rio.set_spatial_dims(x_dim="lon", y_dim="lat", inplace=True)
            model_p_clim = model_p_clim.rio.write_crs("epsg:4326", inplace=True)
            model_et_clim = model_et_clim.rio.set_spatial_dims(x_dim="lon", y_dim="lat", inplace=True)
            model_et_clim = model_et_clim.rio.write_crs("epsg:4326", inplace=True)
            model_q_clim = model_q_clim.rio.set_spatial_dims(x_dim="lon", y_dim="lat", inplace=True)
            model_q_clim = model_q_clim.rio.write_crs("epsg:4326", inplace=True)

                basin_metrics[basin_name]['model_ET'] = clipped_model_et.mean().compute().item() # Added .compute()
            if model_q_clim is not None:
                clipped_model_q = model_q_clim.rio.clip(basin_polygon.geometry.values, crs=basin_polygon.crs)
                basin_metrics[basin_name]['model_Q'] = clipped_model_q.mean().compute().item() # Added .compute()

                clipped_obs_p = obs_p_clim.rio.clip(basin_polygon.geometry.values, crs=basin_polygon.crs)

            if obs_et_clim is not None:
                clipped_obs_et = obs_et_clim.rio.clip(basin_polygon.geometry.values, crs=basin_polygon.crs)
                basin_metrics[basin_name]['obs_ET'] = clipped_obs_et.mean().compute().item() # Added .compute()
            if obs_q_clim is not None:

        except Exception as e:

            # Fill with NaN if clipping fails

for basin_name, gauge_id in BASINS.items():
    print(f"Processing streamflow for basin: {basin_name} (GRDC ID: {gauge_id})")
    
    # Initialize streamflow metrics
    basin_metrics[basin_name]['streamflow_volume_bias'] = np.nan
    basin_metrics[basin_name]['wasserstein_distance'] = np.nan

    if gauge_id not in gauge_metadata.index:
        print(f"Warning: Gauge {gauge_id} not found in metadata. Skipping streamflow for {basin_name}.")
        continue

    gauge_lat = gauge_metadata.loc[gauge_id, 'lat']
    gauge_lon = gauge_metadata.loc[gauge_id, 'lon']

    # Load observed streamflow
    obs_streamflow_path = os.path.join(DATA_DIR, "obs", "streamflow", f"{gauge_id}.csv")
    try:
        obs_df = pd.read_csv(obs_streamflow_path, parse_dates=['date'], index_col='date')
        obs_df = obs_df.loc[TIME_SLICE]
        obs_discharge = obs_df['discharge_m3s'].dropna()
        print(f"Loaded observed streamflow for {gauge_id} ({len(obs_discharge)} records).")
    except Exception as e:
        print(f"Error loading observed streamflow for {gauge_id}: {e}")
        obs_discharge = pd.Series([])

    if mosart_ds is not None and not obs_discharge.empty:
        try:
            # Find nearest MOSART grid cell
            _, idx = mosart_kdtree.query(np.array([gauge_lat, gauge_lon]))
            mosart_lat_idx = mosart_ds['lat'].isel(lat=idx).item()
            mosart_lon_idx = mosart_ds['lon'].isel(lon=idx).item()
            print(f"Gauge {gauge_id} ({gauge_lat:.2f}, {gauge_lon:.2f}) matched to MOSART grid cell ({mosart_lat_idx:.2f}, {mosart_lon_idx:.2f}).")

            # Extract simulated discharge
            sim_discharge_series = mosart_ds['RIVER_DISCHARGE_OVER_LAND_LIQ'].sel(lat=mosart_lat_idx, lon=mosart_lon_idx, method='nearest').to_series()
            sim_discharge_series = sim_discharge_series.loc[TIME_SLICE].dropna()
            print(f"Extracted simulated streamflow for {gauge_id} ({len(sim_discharge_series)} records).")

            # Align time series
            common_dates = obs_discharge.index.intersection(sim_discharge_series.index)
            if len(common_dates) == 0:
                print(f"Warning: No common dates for observed and simulated streamflow for {gauge_id}. Skipping FDC metrics.")
                continue
            
            obs_discharge_aligned = obs_discharge.loc[common_dates]
            sim_discharge_aligned = sim_discharge_series.loc[common_dates]

            # Compute FDC metrics
            # Volume bias
            obs_total_volume = obs_discharge_aligned.sum() * 86400 # m3/s to m3
            sim_total_volume = sim_discharge_aligned.sum() * 86400
            if obs_total_volume > 0:
                volume_bias = (sim_total_volume - obs_total_volume) / obs_total_volume * 100 # Percentage bias
                basin_metrics[basin_name]['streamflow_volume_bias'] = volume_bias
            else:
                print(f"Warning: Observed total volume is zero for {gauge_id}. Cannot compute volume bias.")
                basin_metrics[basin_name]['streamflow_volume_bias'] = np.nan

            # Wasserstein distance
            # For FDC, we compare the sorted values directly.
            obs_fdc_values, _ = calculate_fdc(obs_discharge_aligned)
            sim_fdc_values, _ = calculate_fdc(sim_discharge_aligned)

            # Ensure both FDCs have the same length for Wasserstein distance, or interpolate
            # For simplicity, we'll take the shorter length and resample
            min_len = min(len(obs_fdc_values), len(sim_fdc_values))
            if min_len > 0:
                wd = wasserstein_distance(obs_fdc_values.iloc[:min_len], sim_fdc_values.iloc[:min_len])
                basin_metrics[basin_name]['wasserstein_distance'] = wd
            else:
                print(f"Warning: FDC values are empty for {gauge_id}. Cannot compute Wasserstein distance.")
                basin_metrics[basin_name]['wasserstein_distance'] = np.nan

        except Exception as e:
            print(f"Error computing streamflow metrics for {basin_name}: {e}")
    elif obs_discharge.empty:
        print(f"Skipping streamflow metrics for {basin_name} due to missing observed data.")
    else:
        print(f"Skipping streamflow metrics for {basin_name} due to missing MOSART data.")

# --- Radar Chart: Multi-variable diagnostic ---
try:
    # Select metrics for radar chart and normalize them
    radar_metrics = summary_df[[
        'P_bias_%', 'ET_bias_%', 'Q_bias_%',
        'water_balance_residual_bias_%', 'streamflow_volume_bias', 'wasserstein_distance'
    ]].copy()

    # Normalize biases to a 0-1 scale, where 0 is perfect, 1 is max deviation
    # For percentage biases, we can take absolute values and scale.
    # For Wasserstein distance, normalize by the max observed discharge or max WD.
    
    # Let's define a 'score' where 0 is best and 1 is worst.
    # For biases, abs(bias) / max_abs_bias
    # For WD, WD / max_WD
    
    # Handle potential inf/-inf values from division by zero
    radar_metrics = radar_metrics.replace([np.inf, -np.inf], np.nan)
    
    # Fill NaNs with 0 for normalization if a metric is missing for a basin,
    # or drop the metric if it's consistently NaN. For now, let's fill with 0
    # to allow the radar chart to be drawn, but this might misrepresent.
    # A better approach might be to exclude the spoke if NaN.
    radar_metrics = radar_metrics.fillna(0) 

    # Normalize each column to [0, 1] range
    normalized_metrics = pd.DataFrame(index=radar_metrics.index, columns=radar_metrics.columns)
    for col in radar_metrics.columns:
        if col == 'wasserstein_distance':
            # For WD, higher is worse. Normalize by max WD.
            max_val = radar_metrics[col].max()
            if max_val > 0:
                normalized_metrics[col] = radar_metrics[col] / max_val
            else:
                normalized_metrics[col] = 0 # All WDs are 0
        else: # For percentage biases, absolute value, then normalize
            abs_vals = radar_metrics[col].abs()
            max_abs_val = abs_vals.max()
            if max_abs_val > 0:
                normalized_metrics[col] = abs_vals / max_abs_val
            else:
                normalized_metrics[col] = 0 # All biases are 0

    # Prepare data for radar chart
    labels = normalized_metrics.columns.tolist()
    num_vars = len(labels)

    angles = np.linspace(0, 2 * np.pi, num_vars, endpoint=False).tolist()
    angles += angles[:1] # Complete the loop

    fig = plt.figure(figsize=(10, 10))
