# The user specified output directory
OUTPUT_DIR = "/Users/zhou014/Local_Drive/Git_repo/esmflow/benchmark/results/gemini-2.5-flash_baseline/task_03_et_benchmark/run2_output"
DATA_ROOT = "./data/sample/e3sm/"

ILAMB_DATA_URL_BASE = "https://www.ilamb.org/ILAMB-Data/DATA/"

E3SM_CASE_NAME = "sample.v3.LR.historical"
E3SM_COMPONENT_DIR = "lnd" # Directory name for ELM files
E3SM_COMPONENT_FILENAME = "elm" # Component name used in the ELM filename
E3SM_STREAM = "h0"
E3SM_ET_VARS = ["QVEGE", "QVEGT", "QSOIL"] # E3SM ELM ET components in kg/m2/s (equivalent to mm/s)

MODIS_VAR_ILAMB = "evspsbl"
MODIS_DATASET_ILAMB = "MODIS"
MODIS_FILENAME_ILAMB = "et_0.5x0.5.nc"
MODIS_VAR_IN_FILE = "et" # Variable name inside the MODIS NetCDF file

START_YEAR = 1985
END_YEAR = 1989
YEARS = range(START_YEAR, END_YEAR + 1)

# --- Helper function for ILAMB data download ---
def download_ilamb_data(variable, dataset, filename, output_path):
    """Downloads a file from the ILAMB data portal if it doesn't exist locally."""
    url = f"{ILAMB_DATA_URL_BASE}/{variable}/{dataset}/{filename}"
    if not os.path.exists(output_path):
        print(f"Downloading {filename} from ILAMB...")
        try:
            response = requests.get(url, stream=True, timeout=60) # Added a timeout for robustness
            response.raise_for_status() # Raise an HTTPError for bad responses (4xx or 5xx)
            with open(output_path, 'wb') as f:
                for chunk in response.iter_content(chunk_size=8192):
                    f.write(chunk)
            print(f"Downloaded {filename} to {output_path}")
        except requests.exceptions.RequestException as e:
            print(f"Error downloading {filename} from {url}: {e}")
            return None
        except Exception as e:
            print(f"An unexpected error occurred during download of {filename}: {e}")
            return None
    else:
        print(f"{filename} already exists at {output_path}, skipping download.")
    return output_path

# --- Helper function to ensure 0-360 longitude and sorting ---
def ensure_0_360_lon(da_or_ds):
    """Ensures DataArray or Dataset has longitude in 0-360 range and sorted."""
    if 'lon' in da_or_ds.coords:
        # Check if longitude is already 0-360
        if da_or_ds['lon'].max() > 180 and da_or_ds['lon'].min() >= 0:
            return da_or_ds # Already 0-360, no change needed
        elif da_or_ds['lon'].min() < 0:
            # Convert -180 to 180 to 0-360
            da_or_ds = da_or_ds.assign_coords(lon=(da_or_ds['lon'] % 360))
            da_or_ds = da_or_ds.sortby('lon')
            print("Longitude converted to 0-360 and sorted.")
    return da_or_ds

# --- 1. Fetch MODIS ET data ---
modis_local_path = os.path.join(OUTPUT_DIR, MODIS_FILENAME_ILAMB)
downloaded_modis_path = download_ilamb_data(MODIS_VAR_ILAMB, MODIS_DATASET_ILAMB, MODIS_FILENAME_ILAMB, modis_local_path)

modis_ds = None
if downloaded_modis_path:
    try:
        modis_ds = xr.open_dataset(downloaded_modis_path)
        print(f"MODIS dataset loaded: {modis_ds}")

        # Standardize coordinate names to 'lat', 'lon', 'time'
        if 'latitude' in modis_ds.coords:
            modis_ds = modis_ds.rename({'latitude': 'lat'})
        if 'longitude' in modis_ds.coords:
            modis_ds = modis_ds.rename({'longitude': 'lon'})
        if 'time' not in modis_ds.coords and 'date' in modis_ds.coords:
             modis_ds = modis_ds.rename({'date': 'time'})

        # Ensure longitude is 0-360
        modis_ds = ensure_0_360_lon(modis_ds)

        # Select the ET variable and rename for consistency
        modis_et = modis_ds[MODIS_VAR_IN_FILE]
        modis_et.name = 'et'

        # Check units: MODIS ET from ILAMB is reported as kg/m2/s in the error log.
        # Convert from kg/m2/s (which is equivalent to mm/s) to mm/day.
        if modis_et.attrs.get('units', '').lower() == 'kg/m2/s' or modis_et.attrs.get('units', '').lower() == 'mm/s':
            modis_et = modis_et * 86400 # seconds per day
            modis_et.attrs['units'] = 'mm/day'
            print(f"MODIS ET converted from kg/m2/s (or mm/s) to mm/day. Units: {modis_et.attrs['units']}")
        else:
            print(f"MODIS ET units: {modis_et.attrs.get('units', 'unknown')}. Assuming it's already mm/day or compatible.")


    except Exception as e:
        print(f"Error loading or processing MODIS data: {e}")
        modis_ds = None

if modis_ds is None:
    print("MODIS data not available. Exiting.")
    exit()

# --- 2. Load E3SM ELM ET data ---
e3sm_file_paths = []
for year in YEARS:
    for month in range(1, 13):
        # Corrected filename: use E3SM_COMPONENT_FILENAME ("elm") for the filename itself
        filename = f"{E3SM_CASE_NAME}.{E3SM_COMPONENT_FILENAME}.{E3SM_STREAM}.{year}-{month:02d}.nc"
        e3sm_file_paths.append(os.path.join(DATA_ROOT, E3SM_COMPONENT_DIR, filename))

e3sm_et = None
try:
    # Open multiple files, select variables, and combine
    # Using `parallel=True` for potentially faster loading with dask
    e3sm_ds = xr.open_mfdataset(e3sm_file_paths, combine='by_coords',
                                    data_vars='minimal', coords='minimal',
                                    compat='override', parallel=True)
    print(f"E3SM dataset loaded: {e3sm_ds}")

    # Calculate total ET from components
    e3sm_et_components = [e3sm_ds[var] for var in E3SM_ET_VARS if var in e3sm_ds]
    if len(e3sm_et_components) != len(E3SM_ET_VARS):
        print(f"Warning: Not all E3SM ET components found. Found: {[v.name for v in e3sm_et_components]}")
        if not e3sm_et_components:
            raise ValueError("No E3SM ET components found in dataset.")

    e3sm_et = sum(e3sm_et_components)
    e3sm_et.name = 'et'
    e3sm_et.attrs['long_name'] = 'Total Evapotranspiration (QVEGE + QVEGT + QSOIL)'
    e3sm_et.attrs['units'] = 'mm/s' # Original units

    # Convert E3SM ET from mm/s to mm/day (1 kg/m2/s = 1 mm/s)
    e3sm_et = e3sm_et * 86400 # seconds per day
    e3sm_et.attrs['units'] = 'mm/day'
    print(f"E3SM ET calculated and converted to mm/day. Units: {e3sm_et.attrs['units']}")

    # Ensure E3SM longitude is 0-360
    e3sm_et = ensure_0_360_lon(e3sm_et)

except Exception as e:
    print(f"Error loading or processing E3SM data: {e}")
    e3sm_et = None

if e3sm_et is None:
    print("E3SM data not available. Exiting.")
    exit()

# --- 3. Extract climatological time-mean ET (1985-1989) ---
print(f"Calculating 5-year climatology ({START_YEAR}-{END_YEAR})...")

# E3SM climatology
e3sm_et_clim = None
try:
    e3sm_et_period = e3sm_et.sel(time=slice(f"{START_YEAR}-01-01", f"{END_YEAR}-12-31"))
    e3sm_et_clim = e3sm_et_period.mean(dim='time', keep_attrs=True)
    print(f"E3SM climatological mean ET calculated. Shape: {e3sm_et_clim.shape}")
except Exception as e:
    print(f"Error calculating E3SM climatology: {e}")

# MODIS climatology
modis_et_clim = None
try:
    modis_et_period = modis_et.sel(time=slice(f"{START_YEAR}-01-01", f"{END_YEAR}-12-31"))
    modis_et_clim = modis_et_period.mean(dim='time', keep_attrs=True)
    print(f"MODIS climatological mean ET calculated. Shape: {modis_et_clim.shape}")
except Exception as e:
    print(f"Error calculating MODIS climatology: {e}")

if e3sm_et_clim is None or modis_et_clim is None:
    print("Climatological means could not be calculated. Exiting.")
    exit()

# --- 4. Regrid E3SM to MODIS grid ---
print("Regridding E3SM ET to MODIS grid...")
try:
    # E3SM ELM output often has 'lat' and 'lon' as coordinates of a 'gridcell' dimension.
    # xarray.interp can often handle this by implicitly mapping the 1D gridcell to the 2D target grid.
    if 'gridcell' in e3sm_et_clim.dims and 'lat' in e3sm_et_clim.coords and 'lon' in e3sm_et_clim.coords:
        print("E3SM data has 'gridcell' dimension. Attempting interpolation directly.")
        e3sm_et_regridded = e3sm_et_clim.interp(lat=modis_et_clim['lat'], lon=modis_et_clim['lon'], method='linear')
    else:
        # Assume E3SM is already on a 2D lat/lon grid
        e3sm_et_regridded = e3sm_et_clim.interp(lat=modis_et_clim['lat'], lon=modis_et_clim['lon'], method='linear')

    print(f"E3SM ET regridded to MODIS grid. Shape: {e3sm_et_regridded.shape}")
except Exception as e:
    print(f"Error regridding E3SM data: {e}")
    e3sm_et_regridded = None

if e3sm_et_regridded is None:
    print("Regridding failed. Exiting.")
    exit()

# Ensure both have the same name for consistency
modis_et_clim.name = 'et'
e3sm_et_regridded.name = 'et'

# --- 5. Compute spatial bias and correlation ---
print("Computing spatial bias and correlation...")
bias_et = None
global_mean_bias = np.nan
spatial_correlation = np.nan

try:
    # Align coordinates before computing bias to ensure they match perfectly
    e3sm_aligned, modis_aligned = xr.align(e3sm_et_regridded, modis_et_clim, join='inner', exclude=['time'])

    bias_et = e3sm_aligned - modis_aligned
    bias_et.name = 'et_bias'
    bias_et.attrs['long_name'] = 'E3SM - MODIS ET Bias'
    bias_et.attrs['units'] = 'mm/day'

    # Calculate global mean bias (weighted by latitude)
    weights = np.cos(np.deg2rad(bias_et['lat']))
    weights.name = "weights"
    global_mean_bias = bias_et.weighted(weights).mean(dim=('lat', 'lon')).item()
    print(f"Global Mean ET Bias (E3SM - MODIS): {global_mean_bias:.2f} mm/day")

    # Calculate spatial correlation
    # Flatten the arrays, removing NaNs for correlation calculation
    e3sm_flat = e3sm_aligned.values.flatten()
    modis_flat = modis_aligned.values.flatten()

    # Mask NaNs to ensure only valid pairs are used
    valid_indices = ~np.isnan(e3sm_flat) & ~np.isnan(modis_flat)
    e3sm_valid = e3sm_flat[valid_indices]
    modis_valid = modis_flat[valid_indices]

    if len(e3sm_valid) > 1 and len(modis_valid) > 1: # Need at least 2 points for correlation
        spatial_correlation, _ = pearsonr(e3sm_valid, modis_valid)
        print(f"Spatial Correlation (E3SM vs MODIS): {spatial_correlation:.2f}")
    else:
        print("Not enough valid data points to compute spatial correlation.")

except Exception as e:
    print(f"Error computing bias or correlation: {e}")

# --- 6. Produce a map of the bias field ---
print("Generating bias map...")
try:
    fig = plt.figure(figsize=(12, 8))
    # Use PlateCarree projection, centered on 180 longitude for a global view with 0-360 data
    ax = fig.add_subplot(1, 1, 1, projection=ccrs.PlateCarree(central_longitude=180))

    # Determine a robust symmetrical colorbar range using percentiles to avoid outliers
    abs_max_bias = np.nanpercentile(np.abs(bias_et.values), 99)
    if not np.isfinite(abs_max_bias) or abs_max_bias == 0:
        abs_max_bias = np.nanmax(np.abs(bias_et.values)) # Fallback if 99th percentile is problematic
        if not np.isfinite(abs_max_bias) or abs_max_bias == 0:
            abs_max_bias = 10 # Default if all else fails (e.g., all NaNs or zero bias)

    bias_et.plot.pcolormesh(ax=ax, transform=ccrs.PlateCarree(),
                            cmap='RdBu_r', # Red-Blue reversed: red for positive bias (model too high), blue for negative
                            cbar_kwargs={'label': 'ET Bias (mm/day)'},
                            vmin=-abs_max_bias,
                            vmax=abs_max_bias)

    ax.coastlines()
    ax.set_title(f'E3SM ELM vs MODIS ET Bias ({START_YEAR}-{END_YEAR} Climatology)\n'
                 f'Global Mean Bias: {global_mean_bias:.2f} mm/day, Spatial R: {spatial_correlation:.2f}')
    ax.gridlines(draw_labels=True, dms=True, x_inline=False, y_inline=False)

    bias_map_path = os.path.join(OUTPUT_DIR, 'et_bias_map.png')
    plt.savefig(bias_map_path, bbox_inches='tight', dpi=300)
    plt.close(fig)
    print(f"Bias map saved to {bias_map_path}")

except Exception as e:
    print(f"Error generating bias map: {e}")

print("Saving numerical results...")
try:
    results_summary_path = os.path.join(OUTPUT_DIR, 'et_benchmark_summary.txt')
    with open(results_summary_path, 'w') as f:

        f.write("-------------------------------------------------------------------\n")

        f.write(f"Spatial Correlation (E3SM vs MODIS): {spatial_correlation:.2f}\n")
        f.write("\n")
        f.write("E3SM ET components: QVEGE, QVEGT, QSOIL\n")
        f.write(f"MODIS ET variable: {MODIS_VAR_IN_FILE}\n")
        f.write(f"Time period: {START_YEAR}-{END_YEAR}\n")
    print(f"Benchmark summary saved to {results_summary_path}")

    # Optionally save the regridded climatologies and bias field as NetCDF
    if e3sm_et_regridded is not None:
        try:
            e3sm_et_regridded.to_netcdf(os.path.join(OUTPUT_DIR, 'e3sm_et_climatology_regridded.nc'))
            print("Regridded E3SM ET climatology saved as NetCDF.")
        except Exception as e:
            print(f"Error saving regridded E3SM ET climatology: {e}")
    if modis_et_clim is not None:
        try:
            modis_et_clim.to_netcdf(os.path.join(OUTPUT_DIR, 'modis_et_climatology.nc'))
            print("MODIS ET climatology saved as NetCDF.")
        except Exception as e:
            print(f"Error saving MODIS ET climatology: {e}")
    if bias_et is not None:
        try:
            bias_et.to_netcdf(os.path.join(OUTPUT_DIR, 'et_bias_field.nc'))
            print("ET bias field saved as NetCDF.")
        except Exception as e:
            print(f"Error saving ET bias field: {e}")

except Exception as e:
    print(f"Error saving numerical results: {e}")

print("Analysis complete.")
