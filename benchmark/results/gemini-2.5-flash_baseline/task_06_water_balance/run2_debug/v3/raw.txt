DATA_ROOT = "./data/sample/e3sm/"

CASE_NAME = "sample.v3.LR.historical"
ELM_DIR = os.path.join(DATA_ROOT, "lnd")

START_YEAR = 1985
END_YEAR = 1989

# Conversion factor from s to day (for kg/m2/s or mm/s to mm/day)
S_TO_DAY = 86400

# Target basins (GRDC_NO and a display name)
TARGET_BASINS = {
    3629000: "Amazon",
    4121801: "Missouri",
    4115200: "Columbia",
    6742900: "Danube",

    1159100: "Orange",

# --- Helper Functions ---
def create_output_dir(path):
    """Creates the output directory if it doesn't exist."""
    try:
        os.makedirs(path, exist_ok=True)
        print(f"Output directory created: {path}")
    except Exception as e:
        print(f"Error creating output directory {path}: {e}")
        raise

def get_elm_files(elm_dir, case_name, start_year, end_year):
    """Generates a list of ELM monthly h0 file paths."""
    file_paths = []
    for year in range(start_year, end_year + 1):
        for month in range(1, 13):
            filename = f"{case_name}.elm.h0.{year:04d}-{month:02d}.nc"
            filepath = os.path.join(elm_dir, filename)
            if os.path.exists(filepath):
                file_paths.append(filepath)
            # else:
            #     print(f"Warning: File not found: {filepath}") # Suppress this warning for cleaner output if many files are missing
    return sorted(file_paths)

def calculate_area_weighted_mean(da, area_weights, spatial_dims):
    """Calculates the area-weighted mean of an xarray DataArray.
    Handles Dask arrays by computing the result before returning a scalar.
    """
    # Handle cases where the DataArray might be empty after masking/clipping
    if da.isnull().all():
        return np.nan

    # Ensure area_weights are aligned with da's spatial dimensions
    if not all(dim in area_weights.dims for dim in spatial_dims):
        raise ValueError(f"Area weights do not have the expected spatial dimensions {spatial_dims}.")

    # Compute weighted sum and total weight
    weighted_sum = (da * area_weights).sum(dim=spatial_dims)
    total_weight = area_weights.sum(dim=spatial_dims)
    
    # Squeeze total_weight to remove any singleton dimensions before computing and converting to scalar
    total_weight_squeezed = total_weight.squeeze()

    # Compute total_weight if it's a dask array before checking its value
    if isinstance(total_weight_squeezed.data, dask.array.Array):
        total_weight_val = total_weight_squeezed.compute().item()
    else:
        total_weight_val = total_weight_squeezed.item()
    
    # Avoid division by zero if total_weight is zero (e.g., no data in basin)
    if total_weight_val == 0:
        return np.nan
    
    # Compute the final result
    result = (weighted_sum / total_weight)
    
    # Squeeze out any dimensions of size 1 before converting to scalar
    result_squeezed = result.squeeze()

    if isinstance(result_squeezed.data, dask.array.Array):
        return result_squeezed.compute().item()
    else:
        return result_squeezed.item()

# --- Main Script ---
def main():
    create_output_dir(OUTPUT_DIR)

    # 1. Load ELM Data
    print(f"Loading ELM data for {START_YEAR}-{END_YEAR}...")
    elm_files = get_elm_files(ELM_DIR, CASE_NAME, START_YEAR, END_YEAR)
    if not elm_files:
        print(f"No ELM files found for {CASE_NAME} in {ELM_DIR} for {START_YEAR}-{END_YEAR}. Exiting.")
        return

    try:
        # Open dataset, handling potential time decoding issues and corrupted files
        datasets = []
        for fpath in elm_files:
            try:
                # Explicitly use 'netcdf4' engine for robustness
                ds = xr.open_dataset(fpath, decode_times=True, engine='netcdf4')
                datasets.append(ds)
            except Exception as e:
                print(f"Warning: Could not open file {fpath}: {e}. Skipping this file.")
                continue
        
        if not datasets:
            print("Error: No ELM datasets could be opened successfully. Exiting.")
            return

        # Combine all successfully opened datasets
        ds_elm = xr.combine_by_coords(datasets, combine_attrs='override')
        
        # Ensure time dimension is properly sorted
        if 'time' in ds_elm.dims:
            ds_elm = ds_elm.sortby('time')
        print("ELM data loaded successfully.")
        # Safely get min/max year, ensuring computation if time is Dask-backed
        min_year = ds_elm['time'].min().dt.year.compute().item()
        max_year = ds_elm['time'].max().dt.year.compute().item()
        print(f"ELM dataset time range: {min_year}-{max_year}")

        # Determine spatial dimensions and area variable
        spatial_dims = None
        if 'gridcell' in ds_elm.dims:
            spatial_dims = ['gridcell']
            # Ensure lat/lon are data variables for unstructured grid
            if 'lat' not in ds_elm.data_vars or 'lon' not in ds_elm.data_vars:
                raise ValueError("ELM dataset with 'gridcell' dimension must have 'lat' and 'lon' as data variables.")
        elif 'lat' in ds_elm.dims and 'lon' in ds_elm.dims:
            spatial_dims = ['lat', 'lon']
        else:
            raise ValueError("ELM dataset does not have 'gridcell' or 'lat'/'lon' dimensions for spatial processing.")
        
        if 'area' not in ds_elm.data_vars:
            raise ValueError("Error: 'area' variable not found in ELM dataset. Cannot perform area-weighted calculations.")
        
        area_weights = ds_elm['area'] # Area in m^2

        # Handle 0-360 longitude convention if necessary for clipping
        # Check if longitudes are 0-360. Use .compute() if 'lon' could be Dask-backed.
        if ds_elm['lon'].max().compute().item() > 180:
            print("Adjusting ELM longitudes from 0-360 to -180-180.")
            # Create a new 'lon' coordinate with adjusted values
            ds_elm = ds_elm.assign_coords(lon=(((ds_elm['lon'] + 180) % 360) - 180)).sortby('lon')
            # If 'lon' was a data variable, it might become a coordinate. Ensure it's handled.
            if 'lon' in ds_elm.data_vars: # This might happen if 'lon' was a data_var and then assigned as coord
                ds_elm = ds_elm.drop_vars('lon') # Remove the old data_var 'lon'
            # Re-assign 'lon' as a coordinate if it was a data variable and now needs to be a coordinate
            if 'lon' not in ds_elm.coords and 'lon' in ds_elm.data_vars:
                 ds_elm = ds_elm.set_coords('lon')


        # 2. Compute Water Balance Components (climatological time-mean)
        print("Computing climatological time-mean water balance components...")
        
        # Precipitation (P): RAIN + SNOW (kg/m2/s -> mm/day)
        P = (ds_elm['RAIN'] + ds_elm['SNOW']).mean(dim='time') * S_TO_DAY
        P.name = 'P'
        P.attrs['units'] = 'mm/day'
        P.attrs['long_name'] = 'Climatological Mean Precipitation'

        # Evapotranspiration (ET): QVEGE + QVEGT + QSOIL (mm/s -> mm/day)
        ET = (ds_elm['QVEGE'] + ds_elm['QVEGT'] + ds_elm['QSOIL']).mean(dim='time') * S_TO_DAY
        ET.name = 'ET'
        ET.attrs['units'] = 'mm/day'
        ET.attrs['long_name'] = 'Climatological Mean Evapotranspiration'

        # Runoff (Q): QRUNOFF (mm/s -> mm/day)
        Q = ds_elm['QRUNOFF'].mean(dim='time') * S_TO_DAY
        Q.name = 'Q'
        Q.attrs['units'] = 'mm/day'
        Q.attrs['long_name'] = 'Climatological Mean Runoff'

        # 3. Area-weighted Global Mean
        print("Computing area-weighted global means...")
        # The calculate_area_weighted_mean function now returns a scalar directly
        global_P = calculate_area_weighted_mean(P, area_weights, spatial_dims)
        global_ET = calculate_area_weighted_mean(ET, area_weights, spatial_dims)
        global_Q = calculate_area_weighted_mean(Q, area_weights, spatial_dims)
        global_residual = global_P - global_ET - global_Q

        print(f"Global Mean P: {global_P:.2f} mm/day")
        print(f"Global Mean ET: {global_ET:.2f} mm/day")
        print(f"Global Mean Q: {global_Q:.2f} mm/day")
        print(f"Global Mean Residual (P - ET - Q): {global_residual:.2f} mm/day")

        # 4. Water Balance Residual Spatial Field
        Residual = P - ET - Q
        Residual.name = 'Residual'
        Residual.attrs['units'] = 'mm/day'
        Residual.attrs['long_name'] = 'Climatological Mean Water Balance Residual (P - ET - Q)'
        print("Water balance residual spatial field computed.")

        # Save global fields to NetCDF
        try:
            # Ensure Dask arrays are computed before saving to NetCDF
            P.compute().to_netcdf(os.path.join(OUTPUT_DIR, "elm_climo_P.nc"))
            ET.compute().to_netcdf(os.path.join(OUTPUT_DIR, "elm_climo_ET.nc"))
            Q.compute().to_netcdf(os.path.join(OUTPUT_DIR, "elm_climo_Q.nc"))
            Residual.compute().to_netcdf(os.path.join(OUTPUT_DIR, "elm_climo_Residual.nc"))
            print("Climatological mean P, ET, Q, Residual saved to NetCDF.")
        except Exception as e:
            print(f"Error saving climatological mean NetCDF files: {e}")

        # 5. Load Basin Polygons
        print("Loading basin polygons...")
        try:
            basins_gdf = gpd.read_file(BASIN_POLYGONS_PATH)
            # Filter for target basins
            target_grdc_nos = list(TARGET_BASINS.keys())
            basins_gdf = basins_gdf[basins_gdf['grdc_no'].isin(target_grdc_nos)].copy()
            # Add basin names
            basins_gdf['basin_name'] = basins_gdf['grdc_no'].map(TARGET_BASINS)
            print(f"Loaded {len(basins_gdf)} target basins.")
        except Exception as e:
            print(f"Error loading basin polygons: {e}")
            return

        # Ensure the ELM data has a CRS for spatial operations
        # Assuming ELM data is WGS84 (EPSG:4326)
        # This is important for geopandas spatial operations and rioxarray if used.
        if 'rioxarray' in globals(): # Check if rioxarray was successfully imported
            for da in [P, ET, Q, Residual, area_weights]:
                if not hasattr(da, 'rio') or da.rio.crs is None:
                    da.rio.write_crs("EPSG:4326", inplace=True)
        else:
            print("Warning: rioxarray not available, CRS handling might be limited.")


        # 6. Clip Fields to Basins and Compute Basin Means
        print("Clipping fields to basins and computing basin means...")
        basin_data = []

        # Manual masking approach for unstructured grid (common for E3SM ELM h0)
        if 'gridcell' in spatial_dims:
            print("ELM data is on a 'gridcell' dimension. Using manual masking for basin clipping.")
            
            # Create a GeoDataFrame of ELM grid cell centroids
            # Ensure 'lon' and 'lat' are computed if they are Dask-backed
            elm_lons = ds_elm['lon'].compute().values
            elm_lats = ds_elm['lat'].compute().values

            elm_points = gpd.GeoDataFrame(
                geometry=gpd.points_from_xy(elm_lons, elm_lats),
                crs="EPSG:4326"
            )
            elm_points['gridcell'] = ds_elm['gridcell'].values # Add gridcell index for later joining
            
            for _, basin in basins_gdf.iterrows():
                basin_name = basin['basin_name']
                basin_geom = basin['geometry']
                
                print(f"  Processing basin: {basin_name}")
                
                try:
                    # Perform spatial join to find ELM grid cells within the basin
                    # Use 'intersects' for containment
                    points_in_basin = elm_points[elm_points.geometry.intersects(basin_geom)]
                    
                    if points_in_basin.empty:
                        print(f"    No ELM grid cells found within {basin_name}. Skipping.")
                        continue
                    
                    # Get the gridcell indices for the current basin
                    basin_gridcells = points_in_basin['gridcell'].values
                    
                    # Select data for these gridcells
                    basin_P = P.sel(gridcell=basin_gridcells)
                    basin_ET = ET.sel(gridcell=basin_gridcells)
                    basin_Q = Q.sel(gridcell=basin_gridcells)
                    basin_Residual = Residual.sel(gridcell=basin_gridcells)
                    
                    # Get area weights for these gridcells
                    basin_area_weights = area_weights.sel(gridcell=basin_gridcells)
                    
                    # Calculate area-weighted means for the basin
                    # The calculate_area_weighted_mean function now returns a scalar directly
                    mean_P = calculate_area_weighted_mean(basin_P, basin_area_weights, spatial_dims)
                    mean_ET = calculate_area_weighted_mean(basin_ET, basin_area_weights, spatial_dims)
                    mean_Q = calculate_area_weighted_mean(basin_Q, basin_area_weights, spatial_dims)
                    mean_Residual = calculate_area_weighted_mean(basin_Residual, basin_area_weights, spatial_dims)
                    
                    basin_data.append({
                        'basin_name': basin_name,
                        'P': mean_P,
                        'ET': mean_ET,
                        'Q': mean_Q,
                        'Residual': mean_Residual
                    })
                except Exception as e:
                    print(f"    Error processing basin {basin_name}: {e}")
                    continue
        else: # Assume 'lat' and 'lon' are dimensions, suitable for rioxarray.clip
            print("ELM data is on 'lat'/'lon' dimensions. Using rioxarray for basin clipping.")
            if 'rioxarray' not in globals():
                print("Error: rioxarray is required for clipping with 'lat'/'lon' dimensions but not found. Exiting.")
                return
            
            for _, basin in basins_gdf.iterrows():
                basin_name = basin['basin_name']
                print(f"  Processing basin: {basin_name}")
                
                try:
                    # Clip the data arrays
                    # rioxarray.clip returns a new DataArray, original is unchanged
                    clipped_P = P.rio.clip([basin['geometry']], P.rio.crs, drop=False)
                    clipped_ET = ET.rio.clip([basin['geometry']], ET.rio.crs, drop=False)
                    clipped_Q = Q.rio.clip([basin['geometry']], Q.rio.crs, drop=False)
                    clipped_Residual = Residual.rio.clip([basin['geometry']], Residual.rio.crs, drop=False)
                    
                    # Mask out NaN values introduced by clipping
                    clipped_P = clipped_P.where(~np.isnan(clipped_P))
                    clipped_ET = clipped_ET.where(~np.isnan(clipped_ET))
                    clipped_Q = clipped_Q.where(~np.isnan(clipped_Q))
                    clipped_Residual = clipped_Residual.where(~np.isnan(clipped_Residual))

                    # Get area weights for the clipped region
                    clipped_area_weights = area_weights.rio.clip([basin['geometry']], area_weights.rio.crs, drop=False)
                    clipped_area_weights = clipped_area_weights.where(~np.isnan(clipped_area_weights))
                    
                    # Calculate area-weighted means for the basin
                    # The calculate_area_weighted_mean function now returns a scalar directly
                    mean_P = calculate_area_weighted_mean(clipped_P, clipped_area_weights, spatial_dims)
                    mean_ET = calculate_area_weighted_mean(clipped_ET, clipped_area_weights, spatial_dims)
                    mean_Q = calculate_area_weighted_mean(clipped_Q, clipped_area_weights, spatial_dims)
                    mean_Residual = calculate_area_weighted_mean(clipped_Residual, clipped_area_weights, spatial_dims)
                    
                    basin_data.append({
                        'basin_name': basin_name,
                        'P': mean_P,
                        'ET': mean_ET,
                        'Q': mean_Q,
                        'Residual': mean_Residual
                    })
                except Exception as e:
                    print(f"    Error clipping or calculating mean for {basin_name}: {e}")
                    continue

        basin_df = pd.DataFrame(basin_data)
        print("Basin-wise water balance components computed.")
        
        # Save basin data to CSV
        try:
            basin_df.to_csv(os.path.join(OUTPUT_DIR, "basin_water_balance.csv"), index=False)
            print("Basin water balance data saved to CSV.")
        except Exception as e:
            print(f"Error saving basin water balance CSV: {e}")

        # 7. Produce Composite Figure
        print("Generating composite figure...")
        try:
            fig = plt.figure(figsize=(15, 12))
            gs = fig.add_gridspec(2, 1, height_ratios=[2, 1]) # Two rows, top for map, bottom for bars

            # --- Top Panel: Global Residual Map ---
            ax_map = fig.add_subplot(gs[0], projection=ccrs.PlateCarree())
            ax_map.set_title(f'ELM Climatological Water Balance Residual (P - ET - Q) {START_YEAR}-{END_YEAR}', fontsize=14)
            ax_map.add_feature(cfeature.COASTLINE, linewidth=0.8)
            ax_map.add_feature(cfeature.BORDERS, linestyle=':', linewidth=0.5)
            ax_map.add_feature(cfeature.LAND, edgecolor='black', facecolor='lightgray')
            ax_map.add_feature(cfeature.OCEAN, facecolor='lightblue')

            # Plot residual
            if 'gridcell' in spatial_dims:
                # Scatter plot for unstructured grid
                # Need to compute Residual before accessing .values for plotting
                Residual_computed = Residual.compute()
                # Ensure 'lon' and 'lat' are computed if they are Dask-backed
                elm_lons = ds_elm['lon'].compute().values
                elm_lats = ds_elm['lat'].compute().values

                valid_indices = ~np.isnan(Residual_computed.values)
                sc = ax_map.scatter(elm_lons[valid_indices], elm_lats[valid_indices], 
                                    c=Residual_computed.values[valid_indices], cmap='RdBu_r', s=1, transform=ccrs.PlateCarree(),
                                    vmin=-5, vmax=5) # Set a reasonable range for residual
            else:
                # pcolormesh for regular lat/lon grid
                # Ensure Residual is computed if it's a Dask array
                sc = Residual.compute().plot.pcolormesh(ax=ax_map, transform=ccrs.PlateCarree(),
                                              cmap='RdBu_r', add_colorbar=False,
                                              vmin=-5, vmax=5) # Set a reasonable range for residual
            
            cbar = plt.colorbar(sc, ax=ax_map, orientation='vertical', pad=0.02, shrink=0.7)
            cbar.set_label('Residual (mm/day)')

            # Overlay basin outlines
            basins_gdf.geometry.boundary.plot(ax=ax_map, color='black', linewidth=1.5, transform=ccrs.PlateCarree())
            
            # Add basin names to map
            for idx, row in basins_gdf.iterrows():
                centroid = row.geometry.centroid
                ax_map.text(centroid.x, centroid.y, row['basin_name'],
                            transform=ccrs.PlateCarree(),
                            fontsize=9, color='darkgreen', ha='center', va='center',
                            bbox=dict(facecolor='white', alpha=0.7, edgecolor='none', boxstyle='round,pad=0.2'))

            ax_map.set_extent([-180, 180, -90, 90], crs=ccrs.PlateCarree()) # Global extent

            # --- Bottom Panel: Per-Basin Bar Charts ---
            ax_bar = fig.add_subplot(gs[1])
            
            if not basin_df.empty:
                bar_width = 0.2
                index = np.arange(len(basin_df))

                bars_P = ax_bar.bar(index - 1.5 * bar_width, basin_df['P'], bar_width, label='P', color='skyblue')
                bars_ET = ax_bar.bar(index - 0.5 * bar_width, basin_df['ET'], bar_width, label='ET', color='lightcoral')
                bars_Q = ax_bar.bar(index + 0.5 * bar_width, basin_df['Q'], bar_width, label='Q', color='lightgreen')
                bars_Residual = ax_bar.bar(index + 1.5 * bar_width, basin_df['Residual'], bar_width, label='Residual', color='gray')

                ax_bar.set_ylabel('Water Flux (mm/day)')
                ax_bar.set_title('Area-Weighted Basin Mean Water Balance Components')
                ax_bar.set_xticks(index)
                ax_bar.set_xticklabels(basin_df['basin_name'], rotation=45, ha='right')
                ax_bar.legend()
                ax_bar.grid(axis='y', linestyle='--', alpha=0.7)
            else:
                ax_bar.text(0.5, 0.5, "No basin data available for bar chart.",
                            horizontalalignment='center', verticalalignment='center',
                            transform=ax_bar.transAxes, fontsize=12)
                ax_bar.set_xticks([])
                ax_bar.set_yticks([])

            plt.tight_layout()
            fig_path = os.path.join(OUTPUT_DIR, "elm_water_balance_composite.png")
            plt.savefig(fig_path, dpi=300)
            print(f"Composite figure saved to {fig_path}")
            plt.close(fig)

        except Exception as e:
            print(f"Error generating composite figure: {e}")

    except Exception as e:
        print(f"An error occurred during ELM data processing: {e}")

if __name__ == "__main__":
    main()
