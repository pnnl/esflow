import xarray as xr
import numpy as np
import matplotlib.pyplot as plt
import cartopy.crs as ccrs
import cartopy.feature as cfeature
import os
import glob

# --- Configuration ---
# Input data paths
BASE_DATA_DIR = "./data/sample/e3sm/"
CASE_NAME = "sample.v3.LR.historical"
COMPONENT_DIR = "lnd"  # Subdirectory name for ELM
COMPONENT_FILE_PREFIX = "elm" # Component prefix in the filename (e.g., .elm.h0.)
H_FILE_TYPE = "h0"  # Monthly output
VARIABLE_NAME = "QRUNOFF"  # Runoff variable in ELM
START_YEAR = 1985
END_YEAR = 1989

# Output directory (as specified in the prompt)
OUTPUT_DIR = "/Users/zhou014/Local_Drive/Git_repo/esmflow/benchmark/results/gemini-2.5-flash_baseline/task_02_seasonal_runoff/run1_output"

# --- Create output directory ---
try:
    os.makedirs(OUTPUT_DIR, exist_ok=True)
    print(f"Output directory created or already exists: {OUTPUT_DIR}")
except Exception as e:
    print(f"Error creating output directory {OUTPUT_DIR}: {e}")
    # If directory creation fails, subsequent file operations will likely fail.
    # For a self-contained script, it's reasonable to exit or raise here.
    exit(1)

# --- File Listing ---
# Construct the file pattern for ELM monthly output
# Corrected: Use COMPONENT_DIR for the subdirectory and COMPONENT_FILE_PREFIX for the filename part
file_pattern = os.path.join(BASE_DATA_DIR, COMPONENT_DIR, f"{CASE_NAME}.{COMPONENT_FILE_PREFIX}.{H_FILE_TYPE}.????-??.nc")
all_files = sorted(glob.glob(file_pattern))

# Filter files for the specified time period (1985-1989)
target_files = []
for f in all_files:
    try:
        # Extract year from filename (e.g., ...1985-01.nc)
        filename = os.path.basename(f)
        year_str = filename.split('.')[-2].split('-')[0]
        year = int(year_str)
        if START_YEAR <= year <= END_YEAR:
            target_files.append(f)
    except Exception as e:
        print(f"Warning: Could not parse year from filename {f}. Skipping. Error: {e}")

if not target_files:
    print(f"Error: No ELM files found for {CASE_NAME} between {START_YEAR}-{END_YEAR} in {os.path.join(BASE_DATA_DIR, COMPONENT_DIR)}")
    exit(1)  # Critical error, cannot proceed without data

print(f"Found {len(target_files)} ELM files for analysis.")

# --- Load Data ---
try:
    # Open multiple files as a single dataset using xarray.open_mfdataset
    # combine='by_coords' ensures files are combined along the time dimension
    # decode_times=True parses time coordinates correctly
    # chunks can improve performance for very large datasets, but for 5 years monthly, it's optional.
    ds = xr.open_mfdataset(target_files, combine='by_coords', decode_times=True)
    print("Dataset loaded successfully.")
    # print(ds) # Uncomment for debugging: inspect dataset structure
except Exception as e:
    print(f"Error loading ELM dataset from {len(target_files)} files: {e}")
    exit(1)  # Critical error

# --- Perform Analysis ---
try:
    # Select the runoff variable and grid cell area
    # QRUNOFF in ELM is typically in mm/s (or kg/m^2/s, which is equivalent for water)
    runoff_raw = ds[VARIABLE_NAME]
    area = ds['area']  # Area in m^2, typically on the 'lndgrid' dimension

    # Convert runoff from mm/s to mm/day for better interpretability
    # 1 mm/s = 86400 mm/day
    runoff_mm_day = runoff_raw * 86400
    runoff_mm_day.attrs['units'] = 'mm/day'
    runoff_mm_day.attrs['long_name'] = f"Runoff ({VARIABLE_NAME})"
    if 'long_name' in runoff_raw.attrs:
        runoff_mm_day.attrs['long_name'] = runoff_raw.attrs['long_name'] + " (converted to mm/day)"
    if 'standard_name' in runoff_raw.attrs:
        runoff_mm_day.attrs['standard_name'] = runoff_raw.attrs['standard_name']

    # Compute climatological mean over the time period (1985-1989)
    # skipna=True ensures NaN values (e.g., over ocean) are ignored
    climatological_mean_runoff = runoff_mm_day.mean(dim='time', skipna=True)
    print("Climatological mean runoff computed.")

    # Compute area-weighted global mean
    # The 'lndgrid' dimension is common to both runoff and area.
    # Multiplying by area and then summing effectively weights each grid cell's contribution.
    # NaNs in runoff (over ocean) will result in NaNs in weighted_runoff, which are then skipped by .sum(skipna=True).
    weighted_runoff = climatological_mean_runoff * area
    global_mean_runoff = weighted_runoff.sum(skipna=True) / area.sum(skipna=True)
    print(f"Area-weighted global mean runoff: {global_mean_runoff.item():.2f} mm/day")

    # Get global minimum and maximum runoff values for statistics and plotting range
    min_runoff = climatological_mean_runoff.min(skipna=True).item()
    max_runoff = climatological_mean_runoff.max(skipna=True).item()
    print(f"Global Minimum Runoff: {min_runoff:.2f} mm/day")
    print(f"Global Maximum Runoff: {max_runoff:.2f} mm/day")

except Exception as e:
    print(f"Error during data analysis: {e}")
    exit(1)  # Critical error

# --- Save Climatological Mean Runoff to NetCDF ---
output_nc_path = os.path.join(OUTPUT_DIR, f"climatological_mean_{VARIABLE_NAME}_{START_YEAR}-{END_YEAR}.nc")
try:
    # Assign a name to the DataArray before saving
    climatological_mean_runoff.name = f"climatological_mean_{VARIABLE_NAME}"
    climatological_mean_runoff.to_netcdf(output_nc_path)
    print(f"Climatological mean runoff saved to: {output_nc_path}")
except Exception as e:
    print(f"Error saving NetCDF file {output_nc_path}: {e}")

# --- Save Global Statistics to Text File ---
output_stats_path = os.path.join(OUTPUT_DIR, f"global_runoff_stats_{START_YEAR}-{END_YEAR}.txt")
try:
    with open(output_stats_path, 'w') as f:
        f.write(f"Climatological Mean Runoff ({START_YEAR}-{END_YEAR})\n")
        f.write(f"Source: E3SM ELM ({CASE_NAME})\n")
        f.write(f"Variable: {VARIABLE_NAME}\n")
        f.write(f"Units: mm/day\n")
        f.write(f"----------------------------------------\n")
        f.write(f"Area-weighted Global Mean: {global_mean_runoff.item():.2f} mm/day\n")
        f.write(f"Global Minimum: {min_runoff:.2f} mm/day\n")
        f.write(f"Global Maximum: {max_runoff:.2f} mm/day\n")
    print(f"Global statistics saved to: {output_stats_path}")
except Exception as e:
    print(f"Error saving statistics file {output_stats_path}: {e}")

# --- Plotting ---
output_plot_path = os.path.join(OUTPUT_DIR, f"map_climatological_mean_{VARIABLE_NAME}_{START_YEAR}-{END_YEAR}.png")
try:
    fig = plt.figure(figsize=(12, 8))
    # Use PlateCarree projection for a global map
    ax = fig.add_subplot(1, 1, 1, projection=ccrs.PlateCarree())

    # Plotting unstructured grid data:
    # xarray's plot method can handle 1D lat/lon coordinates associated with an unstructured dimension ('lndgrid').
    # It effectively plots each grid cell as a colored point or small polygon.
    # For better visualization, one might regrid to a regular lat/lon grid, but this is more complex.
    # The current approach is direct and generally acceptable for global overviews.
    
    # Define colormap and colorbar properties
    cmap = 'viridis' # A good default for continuous data
    cbar_kwargs = {'label': f'{VARIABLE_NAME} ({climatological_mean_runoff.units})'}

    # Plot the data using xarray's built-in plotting function
    # It automatically uses 'lon' and 'lat' coordinates if they are present.
    plot = climatological_mean_runoff.plot(
        ax=ax,
        transform=ccrs.PlateCarree(), # Data coordinates are already PlateCarree (lat/lon)
        cmap=cmap,
        cbar_kwargs=cbar_kwargs,
        extend='max', # Extend colorbar for values above max if data range exceeds colorbar range
        # Optionally set vmin/vmax for fixed color scaling, e.g., vmin=0, vmax=100
    )

    # Add map features
    ax.coastlines(resolution='50m', color='black', linewidth=0.8)
    ax.add_feature(cfeature.BORDERS, linestyle=':', alpha=0.7)
    ax.add_feature(cfeature.LAND, edgecolor='black', facecolor='lightgray', zorder=0)
    ax.add_feature(cfeature.OCEAN, facecolor='lightblue', zorder=0)
    
    # Add gridlines and labels
    gl = ax.gridlines(draw_labels=True, dms=True, x_inline=False, y_inline=False, alpha=0.5, linestyle='--')
    gl.top_labels = False
    gl.right_labels = False

    # Set title
    ax.set_title(f"Climatological Mean Runoff ({VARIABLE_NAME}) {START_YEAR}-{END_YEAR}\n"
                 f"E3SM ELM ({CASE_NAME})", fontsize=14)

    # Add global statistics to the plot as text
    stats_text = (f"Global Mean: {global_mean_runoff.item():.2f} mm/day\n"
                  f"Min: {min_runoff:.2f} mm/day\n"
                  f"Max: {max_runoff:.2f} mm/day")
    ax.text(0.02, 0.02, stats_text, transform=ax.transAxes,
            fontsize=10, verticalalignment='bottom', bbox=dict(boxstyle='round,pad=0.5', fc='white', alpha=0.7))

    plt.tight_layout() # Adjust layout to prevent labels overlapping
    plt.savefig(output_plot_path, dpi=300, bbox_inches='tight')
    plt.close(fig) # Close the figure to free up memory
    print(f"Map saved to: {output_plot_path}")

except Exception as e:
    print(f"Error during plotting: {e}")

print("Script finished.")
