import os
import glob
import xarray as xr
import numpy as np
import matplotlib.pyplot as plt
import cartopy.crs as ccrs
import pandas as pd # Included for general scientific Python environment, though not strictly used in this specific task

# --- Configuration ---
# Input data paths and parameters
E3SM_DATA_DIR = "./data/sample/e3sm/lnd/"
CASE_NAME = "sample.v3.LR.historical"
VARIABLE_NAME = "QRUNOFF"
START_YEAR = 1985
END_YEAR = 1989

# Output directory as specified in the prompt
OUTPUT_DIR = "/Users/zhou014/Local_Drive/Git_repo/esmflow/benchmark/results/gemini-2.5-flash_baseline/task_02_seasonal_runoff/run4_output"

# --- Create output directory ---
try:
    os.makedirs(OUTPUT_DIR, exist_ok=True)
    print(f"Output directory created or already exists: {OUTPUT_DIR}")
except Exception as e:
    print(f"Error creating output directory {OUTPUT_DIR}: {e}")
    # If directory creation fails, we cannot save outputs, so exit.
    exit()

# --- File discovery ---
file_paths = []
for year in range(START_YEAR, END_YEAR + 1):
    # E3SM ELM monthly h0 files pattern: {case_name}.elm.h0.YYYY-MM.nc
    pattern = os.path.join(E3SM_DATA_DIR, f"{CASE_NAME}.elm.h0.{year}-*.nc")
    file_paths.extend(glob.glob(pattern))

if not file_paths:
    print(f"No ELM h0 files found for {CASE_NAME} between {START_YEAR}-{END_YEAR} in {E3SM_DATA_DIR}")
    exit()
else:
    print(f"Found {len(file_paths)} ELM h0 files for analysis.")

# --- Load data ---
ds = None
try:
    # Open multiple files as a single dataset
    # Preprocess to load only necessary variables ('QRUNOFF', 'area', 'lat', 'lon') to save memory
    # 'area' is needed for area-weighted mean, 'lat'/'lon' for plotting
    ds = xr.open_mfdataset(
        file_paths,
        combine='by_coords',
        preprocess=lambda d: d[[VARIABLE_NAME, 'area', 'lat', 'lon']] if all(v in d.data_vars or v in d.coords for v in [VARIABLE_NAME, 'area', 'lat', 'lon']) else d,
        chunks={'time': 12} # Chunking can help with performance for large datasets
    )
    # Ensure we only consider the specified time range, as open_mfdataset might pick up extra files
    ds = ds.sel(time=slice(f'{START_YEAR}', f'{END_YEAR}'))

    print(f"Dataset loaded successfully. Variables: {list(ds.data_vars)}. Coordinates: {list(ds.coords)}")
    print(f"Time range: {ds['time'].min().dt.year.item()}-{ds['time'].max().dt.year.item()}")

except Exception as e:
    print(f"Error loading E3SM ELM data: {e}")
    exit()

# --- Perform analysis ---
climatology_runoff = None
global_mean_runoff_mm_day = None

if ds is not None and VARIABLE_NAME in ds.data_vars and 'area' in ds.data_vars:
    try:
        # 1. Extract climatological mean runoff
        print(f"Calculating climatological mean for {VARIABLE_NAME}...")
        # Calculate mean over the time dimension, skipping NaN values (e.g., over ocean)
        # .compute() is called here to ensure the result is a concrete NumPy array, not a Dask array,
        # which can be more efficient if subsequent operations don't benefit from Dask's lazy evaluation.
        climatology_runoff = ds[VARIABLE_NAME].mean(dim='time', skipna=True).compute()
        print("Climatological mean calculated.")

        # 2. Compute area-weighted global mean
        # 'QRUNOFF' is in mm/s, 'area' is in m^2.
        # xarray handles broadcasting automatically if dimensions match or are compatible.
        # Summing `weighted_runoff` and `ds['area']` with `skipna=True` ensures ocean points (NaNs) are ignored.
        print("Calculating area-weighted global mean...")
        weighted_runoff = climatology_runoff * ds['area']
        
        # The result of sum() on a Dask-backed array is still a Dask array.
        # We need to call .compute() to get the actual scalar value before calling .item().
        global_mean_runoff_mm_s_dask = weighted_runoff.sum(skipna=True) / ds['area'].sum(skipna=True)
        global_mean_runoff_mm_s = global_mean_runoff_mm_s_dask.compute() # Compute the Dask array to a scalar

        # Convert from mm/s to mm/day for better interpretability in output
        MM_S_TO_MM_DAY = 60 * 60 * 24 # seconds/minute * minutes/hour * hours/day
        global_mean_runoff_mm_day = global_mean_runoff_mm_s * MM_S_TO_MM_DAY

        print(f"Area-weighted global mean {VARIABLE_NAME} ({START_YEAR}-{END_YEAR}): {global_mean_runoff_mm_day.item():.3f} mm/day")

    except Exception as e:
        print(f"Error during analysis (climatology or global mean calculation): {e}")
        exit()
else:
    print("Required variables (QRUNOFF, area) not found in dataset or dataset is empty.")
    exit()

# --- Save climatological mean to NetCDF ---
if climatology_runoff is not None:
    try:
        output_nc_path = os.path.join(OUTPUT_DIR, f"elm_{VARIABLE_NAME}_climatology_{START_YEAR}-{END_YEAR}.nc")
        climatology_runoff.to_netcdf(output_nc_path)
        print(f"Climatological mean {VARIABLE_NAME} saved to {output_nc_path}")
    except Exception as e:
        print(f"Error saving climatological mean NetCDF: {e}")

# --- Save global mean statistic to text file ---
if global_mean_runoff_mm_day is not None:
    try:
        output_txt_path = os.path.join(OUTPUT_DIR, f"elm_{VARIABLE_NAME}_global_mean_{START_YEAR}-{END_YEAR}.txt")
        with open(output_txt_path, "w") as f:
            f.write(f"Area-weighted global mean {VARIABLE_NAME} ({START_YEAR}-{END_YEAR}): {global_mean_runoff_mm_day.item():.3f} mm/day\n")
        print(f"Global mean statistic saved to {output_txt_path}")
    except Exception as e:
        print(f"Error saving global mean text file: {e}")

# --- Plotting ---
if climatology_runoff is not None and global_mean_runoff_mm_day is not None:
    try:
        print("Generating map of mean runoff field...")
        fig = plt.figure(figsize=(12, 8))
        # Use PlateCarree projection centered on the Pacific to avoid splitting continents
        ax = fig.add_subplot(1, 1, 1, projection=ccrs.PlateCarree(central_longitude=180))

        # Convert climatology to mm/day for plotting for better interpretability
        climatology_runoff_mm_day = climatology_runoff * MM_S_TO_MM_DAY

        # Plot the climatological mean runoff field
        # E3SM 'lon' coordinates are typically 0-360, which Cartopy's PlateCarree handles correctly.
        # `robust=True` helps with color scaling by ignoring extreme outliers.
        climatology_runoff_mm_day.plot.pcolormesh(
            ax=ax,
            x='lon',
            y='lat',
            transform=ccrs.PlateCarree(),
            cmap='viridis', # A good default, 'YlGnBu' or 'Blues_r' are also good for runoff
            cbar_kwargs={'label': 'Runoff (mm/day)', 'shrink': 0.7},
            add_colorbar=True,
            robust=True
        )

        ax.set_title(f"ELM Climatological Mean {VARIABLE_NAME} ({START_YEAR}-{END_YEAR})", fontsize=14)
        ax.coastlines()
        # Add gridlines with labels for latitude and longitude
        gl = ax.gridlines(draw_labels=True, dms=True, x_inline=False, y_inline=False)
        gl.top_labels = False
        gl.right_labels = False

        # Overlay global statistics on the map
        stats_text = f"Global Mean: {global_mean_runoff_mm_day.item():.3f} mm/day"
        ax.text(0.02, 0.02, stats_text, transform=ax.transAxes,
                fontsize=12, verticalalignment='bottom',
                bbox=dict(boxstyle='round,pad=0.5', fc='white', alpha=0.8))

        output_png_path = os.path.join(OUTPUT_DIR, f"elm_{VARIABLE_NAME}_climatology_map_{START_YEAR}-{END_YEAR}.png")
        plt.savefig(output_png_path, dpi=300, bbox_inches='tight')
        print(f"Map saved to {output_png_path}")
        plt.close(fig) # Close the figure to free memory

    except Exception as e:
        print(f"Error during plotting: {e}")

print("Script finished.")
