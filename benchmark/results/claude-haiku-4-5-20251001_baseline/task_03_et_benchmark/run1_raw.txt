#!/usr/bin/env python
"""
Benchmark E3SM land model evapotranspiration against MODIS satellite observations.
Uses 5-year climatology from 1985-1989.
"""

import os
import sys
import glob
import warnings
import numpy as np
import pandas as pd
import xarray as xr
import matplotlib.pyplot as plt
import cartopy.crs as ccrs
from scipy import stats
from urllib.request import urlretrieve
import tempfile

warnings.filterwarnings('ignore')

# Configuration
OUTPUT_DIR = "/Users/zhou014/Local_Drive/Git_repo/esmflow/benchmark/results/claude-haiku-4-5-20251001_baseline/task_03_et_benchmark/run1_output"
E3SM_DATA_DIR = "./data/sample/e3sm"
CASE_NAME = "sample.v3.LR.historical"
YEARS = [1985, 1986, 1987, 1988, 1989]

# Create output directory
os.makedirs(OUTPUT_DIR, exist_ok=True)

def download_ilamb_data(variable, dataset, filename):
    """Download ILAMB observation data."""
    url = f"https://www.ilamb.org/ILAMB-Data/DATA/{variable}/{dataset}/{filename}"
    output_path = os.path.join(OUTPUT_DIR, filename)
    
    try:
        print(f"Downloading {url}...")
        urlretrieve(url, output_path)
        print(f"Downloaded to {output_path}")
        return output_path
    except Exception as e:
        print(f"Error downloading {url}: {e}")
        return None

def load_elm_data(case_name, years):
    """Load ELM data for specified years and compute ET climatology."""
    try:
        print("Loading ELM data...")
        elm_files = []
        
        for year in years:
            for month in range(1, 13):
                pattern = os.path.join(
                    E3SM_DATA_DIR, "lnd",
                    f"{case_name}.elm.h0.{year:04d}-{month:02d}.nc"
                )
                matching_files = glob.glob(pattern)
                elm_files.extend(matching_files)
        
        if not elm_files:
            print(f"No ELM files found matching pattern")
            return None
        
        print(f"Found {len(elm_files)} ELM files")
        elm_files.sort()
        
        # Load all files
        ds_list = []
        for f in elm_files:
            try:
                ds = xr.open_dataset(f)
                ds_list.append(ds)
            except Exception as e:
                print(f"Error loading {f}: {e}")
                continue
        
        if not ds_list:
            print("No ELM files could be loaded")
            return None
        
        # Concatenate along time dimension
        ds_elm = xr.concat(ds_list, dim='time')
        
        # Compute ET: sum of QVEGE, QVEGT, QSOIL (mm/s)
        # Convert from mm/s to mm/day for easier interpretation
        et_components = []
        for var in ['QVEGE', 'QVEGT', 'QSOIL']:
            if var in ds_elm.data_vars:
                et_components.append(ds_elm[var])
        
        if not et_components:
            print("ET components not found in ELM data")
            return None
        
        et_elm = sum(et_components) * 86400  # Convert mm/s to mm/day
        
        # Compute climatological mean
        et_elm_clim = et_elm.mean(dim='time')
        
        print(f"ELM ET climatology shape: {et_elm_clim.shape}")
        print(f"ELM ET climatology range: {float(et_elm_clim.min()):.2f} to {float(et_elm_clim.max()):.2f} mm/day")
        
        return et_elm_clim
    
    except Exception as e:
        print(f"Error loading ELM data: {e}")
        import traceback
        traceback.print_exc()
        return None

def load_modis_data(modis_file):
    """Load MODIS ET data and compute climatology."""
    try:
        print(f"Loading MODIS data from {modis_file}...")
        ds_modis = xr.open_dataset(modis_file)
        
        # Extract ET variable (should be 'et' according to task description)
        if 'et' in ds_modis.data_vars:
            et_modis = ds_modis['et']
        elif 'evspsbl' in ds_modis.data_vars:
            et_modis = ds_modis['evspsbl']
        else:
            print(f"Available variables: {list(ds_modis.data_vars)}")
            print("ET variable not found")
            return None
        
        # Compute climatological mean
        et_modis_clim = et_modis.mean(dim='time')
        
        print(f"MODIS ET climatology shape: {et_modis_clim.shape}")
        print(f"MODIS ET climatology range: {float(et_modis_clim.min()):.2f} to {float(et_modis_clim.max()):.2f}")
        
        return et_modis_clim
    
    except Exception as e:
        print(f"Error loading MODIS data: {e}")
        import traceback
        traceback.print_exc()
        return None

def regrid_to_common_grid(et_elm, et_modis):
    """Regrid both datasets to a common grid for comparison."""
    try:
        print("Regridding to common grid...")
        
        # Use MODIS grid as reference (typically 0.5x0.5)
        target_lat = et_modis.lat.values
        target_lon = et_modis.lon.values
        
        # Regrid ELM to MODIS grid using nearest neighbor
        et_elm_regrid = et_elm.interp(
            lat=target_lat,
            lon=target_lon,
            method='nearest'
        )
        
        print(f"Regridded ELM shape: {et_elm_regrid.shape}")
        print(f"MODIS shape: {et_modis.shape}")
        
        return et_elm_regrid, et_modis
    
    except Exception as e:
        print(f"Error regridding: {e}")
        import traceback
        traceback.print_exc()
        return None, None

def compute_bias_metrics(et_model, et_obs):
    """Compute bias metrics."""
    try:
        print("Computing bias metrics...")
        
        # Flatten arrays for comparison
        model_flat = et_model.values.flatten()
        obs_flat = et_obs.values.flatten()
        
        # Remove NaN values
        valid_mask = ~(np.isnan(model_flat) | np.isnan(obs_flat))
        model_valid = model_flat[valid_mask]
        obs_valid = obs_flat[valid_mask]
        
        if len(model_valid) == 0:
            print("No valid data for comparison")
            return None
        
        # Compute metrics
        bias = model_valid - obs_valid
        global_mean_bias = np.mean(bias)
        global_rmse = np.sqrt(np.mean(bias**2))
        
        # Spatial correlation
        correlation = np.corrcoef(model_valid, obs_valid)[0, 1]
        
        # Model and obs means
        model_mean = np.mean(model_valid)
        obs_mean = np.mean(obs_valid)
        
        metrics = {
            'global_mean_bias': global_mean_bias,
            'global_rmse': global_rmse,
            'spatial_correlation': correlation,
            'model_mean': model_mean,
            'obs_mean': obs_mean,
            'n_valid_points': len(model_valid)
        }
        
        print(f"Global mean bias: {global_mean_bias:.4f}")
        print(f"Global RMSE: {global_rmse:.4f}")
        print(f"Spatial correlation: {correlation:.4f}")
        print(f"Model mean: {model_mean:.4f}")
        print(f"Obs mean: {obs_mean:.4f}")
        print(f"Valid points: {len(model_valid)}")
        
        return metrics, bias
    
    except Exception as e:
        print(f"Error computing metrics: {e}")
        import traceback
        traceback.print_exc()
        return None, None

def create_bias_map(et_model, et_obs, bias_field, output_file):
    """Create a map of the bias field."""
    try:
        print(f"Creating bias map...")
        
        fig = plt.figure(figsize=(16, 10))
        
        # Bias map
        ax1 = plt.subplot(2, 2, 1, projection=ccrs.PlateCarree())
        bias_2d = bias_field.reshape(et_obs.shape)
        im1 = ax1.contourf(
            et_obs.lon.values, et_obs.lat.values, bias_2d,
            levels=20, cmap='RdBu_r', transform=ccrs.PlateCarree()
        )
        ax1.coastlines()
        ax1.set_title('ET Bias (Model - Obs) [mm/day]', fontsize=12, fontweight='bold')
        plt.colorbar(im1, ax=ax1, label='mm/day')
        
        # Model ET
        ax2 = plt.subplot(2, 2, 2, projection=ccrs.PlateCarree())
        im2 = ax2.contourf(
            et_obs.lon.values, et_obs.lat.values, et_model.values,
            levels=20, cmap='viridis', transform=ccrs.PlateCarree()
        )
        ax2.coastlines()
        ax2.set_title('E3SM ELM ET [mm/day]', fontsize=12, fontweight='bold')
        plt.colorbar(im2, ax=ax2, label='mm/day')
        
        # Obs ET
        ax3 = plt.subplot(2, 2, 3, projection=ccrs.PlateCarree())
        im3 = ax3.contourf(
            et_obs.lon.values, et_obs.lat.values, et_obs.values,
            levels=20, cmap='viridis', transform=ccrs.PlateCarree()
        )
        ax3.coastlines()
        ax3.set_title('MODIS ET [mm/day]', fontsize=12, fontweight='bold')
        plt.colorbar(im3, ax=ax3, label='mm/day')
        
        # Scatter plot
        ax4 = plt.subplot(2, 2, 4)
        model_flat = et_model.values.flatten()
        obs_flat = et_obs.values.flatten()
        valid_mask = ~(np.isnan(model_flat) | np.isnan(obs_flat))
        ax4.scatter(obs_flat[valid_mask], model_flat[valid_mask], alpha=0.3, s=1)
        
        # Add 1:1 line
        min_val = min(obs_flat[valid_mask].min(), model_flat[valid_mask].min())
        max_val = max(obs_flat[valid_mask].max(), model_flat[valid_mask].max())
        ax4.plot([min_val, max_val], [min_val, max_val], 'r--', label='1:1 line')
        
        ax4.set_xlabel('MODIS ET [mm/day]', fontsize=11)
        ax4.set_ylabel('E3SM ELM ET [mm/day]', fontsize=11)
        ax4.set_title('Model vs Observations', fontsize=12, fontweight='bold')
        ax4.legend()
        ax4.grid(True, alpha=0.3)
        
        plt.tight_layout()
        plt.savefig(output_file, dpi=150, bbox_inches='tight')
        print(f"Saved bias map to {output_file}")
        plt.close()
        
    except Exception as e:
        print(f"Error creating bias map: {e}")
        import traceback
        traceback.print_exc()

def save_results(metrics, output_file):
    """Save metrics to CSV file."""
    try:
        print(f"Saving results to {output_file}...")
        
        results_df = pd.DataFrame([metrics])
        results_df.to_csv(output_file, index=False)
        
        print(f"Results saved to {output_file}")
        print("\nMetrics Summary:")
        print(results_df.to_string())
        
    except Exception as e:
        print(f"Error saving results: {e}")
        import traceback
        traceback.print_exc()

def main():
    """Main analysis function."""
    print("=" * 80)
    print("E3SM Land Model ET Benchmark against MODIS")
    print("=" * 80)
    
    # Step 1: Download MODIS data
    print("\n[Step 1] Downloading MODIS ET data...")
    modis_file = download_ilamb_data('evspsbl', 'MODIS', 'et_0.5x0.5.nc')
    
    if modis_file is None or not os.path.exists(modis_file):
        print("Failed to download MODIS data. Attempting to use local copy if available...")
        # Try to find local copy
        local_modis = os.path.join(OUTPUT_DIR, 'et_0.5x0.5.nc')
        if os.path.exists(local_modis):
            modis_file = local_modis
        else:
            print("MODIS data not available. Exiting.")
            return
    
    # Step 2: Load ELM data
    print("\n[Step 2] Loading E3SM ELM data...")
    et_elm = load_elm_data(CASE_NAME, YEARS)
    
    if et_elm is None:
        print("Failed to load ELM data. Exiting.")
        return
    
    # Step 3: Load MODIS data
    print("\n[Step 3] Loading MODIS data...")
    et_modis = load_modis_data(modis_file)
    
    if et_modis is None:
        print("Failed to load MODIS data. Exiting.")
        return
    
    # Step 4: Regrid to common grid
    print("\n[Step 4] Regridding to common grid...")
    et_elm_regrid, et_modis_regrid = regrid_to_common_grid(et_elm, et_modis)
    
    if et_elm_regrid is None:
        print("Failed to regrid data. Exiting.")
        return
    
    # Step 5: Compute bias metrics
    print("\n[Step 5] Computing bias metrics...")
    metrics, bias_field = compute_bias_metrics(et_elm_regrid, et_modis_regrid)
    
    if metrics is None:
        print("Failed to compute metrics. Exiting.")
        return
    
    # Step 6: Create bias map
    print("\n[Step 6] Creating bias map...")
    bias_map_file = os.path.join(OUTPUT_DIR, 'et_bias_map.png')
    create_bias_map(et_elm_regrid, et_modis_regrid, bias_field, bias_map_file)
    
    # Step 7: Save results
    print("\n[Step 7] Saving results...")
    results_file = os.path.join(OUTPUT_DIR, 'et_benchmark_metrics.csv')
    save_results(metrics, results_file)
    
    # Step 8: Save bias field as NetCDF
    print("\n[Step 8] Saving bias field as NetCDF...")
    try:
        bias_2d = bias_field.reshape(et_modis_regrid.shape)
        bias_da = xr.DataArray(
            bias_2d,
            coords={'lat': et_modis_regrid.lat, 'lon': et_modis_regrid.lon},
            dims=['lat', 'lon'],
            name='et_bias'
        )
        bias_da.attrs['units'] = 'mm/day'
        bias_da.attrs['description'] = 'ET bias (E3SM ELM - MODIS)'
        
        bias_nc_file = os.path.join(OUTPUT_DIR, 'et_bias_field.nc')
        bias_da.to_netcdf(bias_nc_file)
        print(f"Saved bias field to {bias_nc_file}")
    except Exception as e:
        print(f"Error saving bias field: {e}")
    
    print("\n" + "=" * 80)
    print("Analysis complete!")
    print("=" * 80)
    print(f"\nOutput files saved to: {OUTPUT_DIR}")
    print(f"  - et_benchmark_metrics.csv")
    print(f"  - et_bias_map.png")
    print(f"  - et_bias_field.nc")

if __name__ == '__main__':
    main()
