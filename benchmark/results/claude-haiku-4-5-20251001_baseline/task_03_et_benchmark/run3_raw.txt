#!/usr/bin/env python
"""
Benchmark E3SM land model evapotranspiration against MODIS satellite observations.
Uses 5-year climatology from 1985-1989.
"""

import os
import sys
import glob
import warnings
import numpy as np
import pandas as pd
import xarray as xr
import matplotlib.pyplot as plt
import cartopy.crs as ccrs
from scipy import stats
from urllib.request import urlretrieve
import tempfile

warnings.filterwarnings('ignore')

# Configuration
OUTPUT_DIR = "/Users/zhou014/Local_Drive/Git_repo/esmflow/benchmark/results/claude-haiku-4-5-20251001_baseline/task_03_et_benchmark/run3_output"
E3SM_DATA_DIR = "./data/sample/e3sm/lnd"
CASE_NAME = "sample.v3.LR.historical"
YEARS = [1985, 1986, 1987, 1988, 1989]

# Create output directory
os.makedirs(OUTPUT_DIR, exist_ok=True)

def download_modis_et():
    """Download MODIS ET dataset from ILAMB."""
    print("Downloading MODIS ET dataset from ILAMB...")
    url = "https://www.ilamb.org/ILAMB-Data/DATA/evspsbl/MODIS/et_0.5x0.5.nc"
    output_file = os.path.join(OUTPUT_DIR, "modis_et.nc")
    
    try:
        urlretrieve(url, output_file)
        print(f"Successfully downloaded MODIS ET to {output_file}")
        return output_file
    except Exception as e:
        print(f"Error downloading MODIS ET: {e}")
        print("Attempting to use local copy if available...")
        return None

def load_e3sm_et_data():
    """Load E3SM ELM ET data for 1985-1989."""
    print("Loading E3SM ELM ET data...")
    
    et_data_list = []
    times = []
    
    for year in YEARS:
        for month in range(1, 13):
            filename = os.path.join(
                E3SM_DATA_DIR,
                f"{CASE_NAME}.elm.h0.{year:04d}-{month:02d}.nc"
            )
            
            if not os.path.exists(filename):
                print(f"Warning: File not found: {filename}")
                continue
            
            try:
                ds = xr.open_dataset(filename)
                
                # Extract ET components (convert from mm/s to mm/day)
                qvege = ds['QVEGE'].values * 86400  # mm/day
                qvegt = ds['QVEGT'].values * 86400  # mm/day
                qsoil = ds['QSOIL'].values * 86400  # mm/day
                
                # Total ET
                et_total = qvege + qvegt + qsoil
                
                et_data_list.append(et_total)
                times.append(f"{year:04d}-{month:02d}")
                
                # Get lat/lon from first file
                if len(et_data_list) == 1:
                    lat = ds['lat'].values
                    lon = ds['lon'].values
                
                ds.close()
                
            except Exception as e:
                print(f"Error loading {filename}: {e}")
                continue
    
    if not et_data_list:
        print("Error: No E3SM data loaded")
        return None, None, None
    
    # Stack into array (time, lat, lon)
    et_array = np.array(et_data_list)
    
    # Create xarray dataset
    et_ds = xr.Dataset(
        {
            'et': (['time', 'lat', 'lon'], et_array)
        },
        coords={
            'time': times,
            'lat': lat,
            'lon': lon
        }
    )
    
    print(f"Loaded E3SM ET data: shape {et_array.shape}")
    return et_ds, lat, lon

def load_modis_et(modis_file):
    """Load MODIS ET data."""
    print("Loading MODIS ET data...")
    
    try:
        ds = xr.open_dataset(modis_file)
        
        # Extract ET variable (named 'et' in MODIS file)
        if 'et' in ds.data_vars:
            et_var = 'et'
        elif 'evspsbl' in ds.data_vars:
            et_var = 'evspsbl'
        else:
            print(f"Available variables: {list(ds.data_vars)}")
            raise ValueError("ET variable not found in MODIS file")
        
        et_data = ds[et_var].values
        
        # Get coordinates
        if 'lat' in ds.coords:
            lat = ds['lat'].values
        else:
            lat = ds.coords[list(ds.coords)[0]].values
            
        if 'lon' in ds.coords:
            lon = ds['lon'].values
        else:
            lon = ds.coords[list(ds.coords)[1]].values
        
        print(f"Loaded MODIS ET data: shape {et_data.shape}")
        print(f"MODIS ET range: {np.nanmin(et_data):.2f} to {np.nanmax(et_data):.2f}")
        
        return et_data, lat, lon
        
    except Exception as e:
        print(f"Error loading MODIS ET: {e}")
        return None, None, None

def compute_climatology(data_array):
    """Compute time-mean climatology."""
    return np.nanmean(data_array, axis=0)

def regrid_to_common_grid(e3sm_et, e3sm_lat, e3sm_lon, modis_et, modis_lat, modis_lon):
    """Regrid E3SM data to MODIS grid using nearest neighbor."""
    print("Regridding E3SM to MODIS grid...")
    
    # Create meshgrids
    modis_lon_grid, modis_lat_grid = np.meshgrid(modis_lon, modis_lat)
    e3sm_lon_grid, e3sm_lat_grid = np.meshgrid(e3sm_lon, e3sm_lat)
    
    # Flatten for easier indexing
    modis_lon_flat = modis_lon_grid.flatten()
    modis_lat_flat = modis_lat_grid.flatten()
    e3sm_lon_flat = e3sm_lon_grid.flatten()
    e3sm_lat_flat = e3sm_lat_grid.flatten()
    
    # Regrid E3SM to MODIS grid
    e3sm_regridded = np.zeros_like(modis_et)
    
    for i in range(len(modis_lon_flat)):
        # Find nearest E3SM grid cell
        dist = np.sqrt(
            (e3sm_lon_flat - modis_lon_flat[i])**2 +
            (e3sm_lat_flat - modis_lat_flat[i])**2
        )
        nearest_idx = np.argmin(dist)
        nearest_i = nearest_idx // len(e3sm_lon)
        nearest_j = nearest_idx % len(e3sm_lon)
        
        e3sm_regridded.flat[i] = e3sm_et[nearest_i, nearest_j]
    
    e3sm_regridded = e3sm_regridded.reshape(modis_et.shape)
    
    print(f"Regridded E3SM to MODIS grid: {e3sm_regridded.shape}")
    return e3sm_regridded

def compute_bias_metrics(e3sm_et, modis_et):
    """Compute bias metrics."""
    print("Computing bias metrics...")
    
    # Mask out NaN values
    valid_mask = ~(np.isnan(e3sm_et) | np.isnan(modis_et))
    
    e3sm_valid = e3sm_et[valid_mask]
    modis_valid = modis_et[valid_mask]
    
    # Bias
    bias = e3sm_et - modis_et
    
    # Global mean bias
    global_mean_bias = np.nanmean(bias)
    
    # Spatial correlation
    correlation = np.corrcoef(e3sm_valid, modis_valid)[0, 1]
    
    # RMSE
    rmse = np.sqrt(np.nanmean(bias**2))
    
    # Mean absolute error
    mae = np.nanmean(np.abs(bias))
    
    print(f"Global mean bias: {global_mean_bias:.4f} mm/day")
    print(f"Spatial correlation: {correlation:.4f}")
    print(f"RMSE: {rmse:.4f} mm/day")
    print(f"MAE: {mae:.4f} mm/day")
    
    return {
        'global_mean_bias': global_mean_bias,
        'correlation': correlation,
        'rmse': rmse,
        'mae': mae,
        'bias_field': bias
    }

def plot_bias_map(bias_field, lat, lon, output_file):
    """Create map of bias field."""
    print(f"Creating bias map: {output_file}")
    
    try:
        fig = plt.figure(figsize=(14, 8))
        ax = plt.axes(projection=ccrs.PlateCarree())
        
        # Plot bias
        im = ax.contourf(
            lon, lat, bias_field,
            levels=np.linspace(-2, 2, 21),
            cmap='RdBu_r',
            transform=ccrs.PlateCarree()
        )
        
        ax.coastlines()
        ax.gridlines(draw_labels=True)
        
        cbar = plt.colorbar(im, ax=ax, orientation='horizontal', pad=0.05)
        cbar.set_label('ET Bias (mm/day)', fontsize=12)
        
        plt.title('E3SM ELM vs MODIS ET Bias (1985-1989 Climatology)', fontsize=14)
        
        plt.tight_layout()
        plt.savefig(output_file, dpi=150, bbox_inches='tight')
        plt.close()
        
        print(f"Saved bias map to {output_file}")
        
    except Exception as e:
        print(f"Error creating bias map: {e}")

def plot_comparison_maps(e3sm_et, modis_et, lat, lon, output_dir):
    """Create comparison maps of E3SM, MODIS, and bias."""
    print("Creating comparison maps...")
    
    try:
        fig, axes = plt.subplots(1, 3, figsize=(18, 5), 
                                 subplot_kw={'projection': ccrs.PlateCarree()})
        
        # E3SM ET
        im1 = axes[0].contourf(
            lon, lat, e3sm_et,
            levels=np.linspace(0, 5, 21),
            cmap='viridis',
            transform=ccrs.PlateCarree()
        )
        axes[0].coastlines()
        axes[0].gridlines(draw_labels=True)
        axes[0].set_title('E3SM ELM ET', fontsize=12)
        cbar1 = plt.colorbar(im1, ax=axes[0], orientation='horizontal', pad=0.05)
        cbar1.set_label('mm/day', fontsize=10)
        
        # MODIS ET
        im2 = axes[1].contourf(
            lon, lat, modis_et,
            levels=np.linspace(0, 5, 21),
            cmap='viridis',
            transform=ccrs.PlateCarree()
        )
        axes[1].coastlines()
        axes[1].gridlines(draw_labels=True)
        axes[1].set_title('MODIS ET', fontsize=12)
        cbar2 = plt.colorbar(im2, ax=axes[1], orientation='horizontal', pad=0.05)
        cbar2.set_label('mm/day', fontsize=10)
        
        # Bias
        bias = e3sm_et - modis_et
        im3 = axes[2].contourf(
            lon, lat, bias,
            levels=np.linspace(-2, 2, 21),
            cmap='RdBu_r',
            transform=ccrs.PlateCarree()
        )
        axes[2].coastlines()
        axes[2].gridlines(draw_labels=True)
        axes[2].set_title('E3SM - MODIS Bias', fontsize=12)
        cbar3 = plt.colorbar(im3, ax=axes[2], orientation='horizontal', pad=0.05)
        cbar3.set_label('mm/day', fontsize=10)
        
        plt.suptitle('E3SM ELM vs MODIS ET Comparison (1985-1989 Climatology)', 
                     fontsize=14, y=1.02)
        plt.tight_layout()
        
        output_file = os.path.join(output_dir, 'et_comparison_maps.png')
        plt.savefig(output_file, dpi=150, bbox_inches='tight')
        plt.close()
        
        print(f"Saved comparison maps to {output_file}")
        
    except Exception as e:
        print(f"Error creating comparison maps: {e}")

def save_results(metrics, output_dir):
    """Save results to CSV."""
    print("Saving results...")
    
    try:
        results_file = os.path.join(output_dir, 'et_benchmark_results.csv')
        
        results_df = pd.DataFrame({
            'Metric': ['Global Mean Bias (mm/day)', 'Spatial Correlation', 
                      'RMSE (mm/day)', 'MAE (mm/day)'],
            'Value': [metrics['global_mean_bias'], metrics['correlation'],
                     metrics['rmse'], metrics['mae']]
        })
        
        results_df.to_csv(results_file, index=False)
        print(f"Saved results to {results_file}")
        
        # Print summary
        print("\n" + "="*50)
        print("ET BENCHMARK SUMMARY (1985-1989 Climatology)")
        print("="*50)
        print(results_df.to_string(index=False))
        print("="*50 + "\n")
        
    except Exception as e:
        print(f"Error saving results: {e}")

def save_netcdf_output(e3sm_et, modis_et, bias, lat, lon, output_dir):
    """Save output data to NetCDF."""
    print("Saving NetCDF output...")
    
    try:
        output_file = os.path.join(output_dir, 'et_benchmark_data.nc')
        
        ds = xr.Dataset(
            {
                'e3sm_et': (['lat', 'lon'], e3sm_et),
                'modis_et': (['lat', 'lon'], modis_et),
                'bias': (['lat', 'lon'], bias)
            },
            coords={
                'lat': lat,
                'lon': lon
            }
        )
        
        ds['e3sm_et'].attrs['units'] = 'mm/day'
        ds['e3sm_et'].attrs['description'] = 'E3SM ELM ET climatology (1985-1989)'
        
        ds['modis_et'].attrs['units'] = 'mm/day'
        ds['modis_et'].attrs['description'] = 'MODIS ET climatology (1985-1989)'
        
        ds['bias'].attrs['units'] = 'mm/day'
        ds['bias'].attrs['description'] = 'E3SM minus MODIS ET bias'
        
        ds.to_netcdf(output_file)
        print(f"Saved NetCDF output to {output_file}")
        
    except Exception as e:
        print(f"Error saving NetCDF: {e}")

def main():
    """Main execution function."""
    print("="*60)
    print("E3SM ELM ET Benchmark against MODIS (1985-1989)")
    print("="*60 + "\n")
    
    # Download MODIS ET
    modis_file = download_modis_et()
    
    if modis_file is None or not os.path.exists(modis_file):
        print("Error: Could not obtain MODIS ET file")
        return
    
    # Load E3SM ET data
    e3sm_ds, e3sm_lat, e3sm_lon = load_e3sm_et_data()
    
    if e3sm_ds is None:
        print("Error: Could not load E3SM ET data")
        return
    
    # Load MODIS ET data
    modis_et_raw, modis_lat, modis_lon = load_modis_et(modis_file)
    
    if modis_et_raw is None:
        print("Error: Could not load MODIS ET data")
        return
    
    # Compute climatologies
    print("\nComputing climatologies...")
    e3sm_et_clim = compute_climatology(e3sm_ds['et'].values)
    modis_et_clim = compute_climatology(modis_et_raw)
    
    print(f"E3SM ET climatology range: {np.nanmin(e3sm_et_clim):.2f} to {np.nanmax(e3sm_et_clim):.2f} mm/day")
    print(f"MODIS ET climatology range: {np.nanmin(modis_et_clim):.2f} to {np.nanmax(modis_et_clim):.2f} mm/day")
    
    # Regrid E3SM to MODIS grid
    e3sm_et_regridded = regrid_to_common_grid(
        e3sm_et_clim, e3sm_lat, e3sm_lon,
        modis_et_clim, modis_lat, modis_lon
    )
    
    # Compute bias metrics
    metrics = compute_bias_metrics(e3sm_et_regridded, modis_et_clim)
    
    # Create visualizations
    plot_bias_map(metrics['bias_field'], modis_lat, modis_lon,
                  os.path.join(OUTPUT_DIR, 'et_bias_map.png'))
    
    plot_comparison_maps(e3sm_et_regridded, modis_et_clim, modis_lat, modis_lon,
                        OUTPUT_DIR)
    
    # Save results
    save_results(metrics, OUTPUT_DIR)
    save_netcdf_output(e3sm_et_regridded, modis_et_clim, metrics['bias_field'],
                      modis_lat, modis_lon, OUTPUT_DIR)
    
    print("\nBenchmarking complete!")
    print(f"All output saved to: {OUTPUT_DIR}")

if __name__ == '__main__':
    main()
