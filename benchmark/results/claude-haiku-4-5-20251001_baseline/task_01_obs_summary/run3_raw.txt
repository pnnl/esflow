```python
import os
import pandas as pd
import numpy as np
from pathlib import Path

# Create output directory
output_dir = "/Users/zhou014/Local_Drive/Git_repo/esmflow/benchmark/results/claude-haiku-4-5-20251001_baseline/task_01_obs_summary/run3_output"
os.makedirs(output_dir, exist_ok=True)

# Load gauge metadata
metadata_file = "./data/sample/obs/gauge_metadata.csv"
try:
    gauge_metadata = pd.read_csv(metadata_file)
    print(f"Loaded gauge metadata from {metadata_file}")
    print(f"Available gauges: {gauge_metadata['gauge_id'].tolist()}")
except Exception as e:
    print(f"Error loading gauge metadata: {e}")
    gauge_metadata = pd.DataFrame()

# Define target gauges
target_gauges = {
    'Amazon': 3629000,
    'Missouri': 4121801,
    'Columbia': 4115200,
    'Danube': 6742900,
    'Mekong': 2969100,
    'Orange': 1159100
}

# Time period
start_date = '1985-01-01'
end_date = '1989-12-31'

# Store results
results = []

# Process each gauge
for gauge_name, gauge_id in target_gauges.items():
    streamflow_file = f"./data/sample/obs/streamflow/{gauge_id}.csv"
    
    try:
        # Load streamflow data
        df = pd.read_csv(streamflow_file)
        df['date'] = pd.to_datetime(df['date'])
        
        # Filter to time period
        df_filtered = df[(df['date'] >= start_date) & (df['date'] <= end_date)].copy()
        
        if len(df_filtered) == 0:
            print(f"Warning: No data for {gauge_name} ({gauge_id}) in period {start_date} to {end_date}")
            continue
        
        # Extract monthly data (use daily values and aggregate to monthly)
        df_filtered['year_month'] = df_filtered['date'].dt.to_period('M')
        monthly_data = df_filtered.groupby('year_month')['discharge_m3s'].mean()
        
        # Compute summary statistics
        mean_val = monthly_data.mean()
        std_val = monthly_data.std()
        min_val = monthly_data.min()
        max_val = monthly_data.max()
        count = len(monthly_data)
        
        # Get metadata for this gauge
        gauge_info = gauge_metadata[gauge_metadata['gauge_id'] == gauge_id]
        if len(gauge_info) > 0:
            lat = gauge_info['lat'].values[0]
            lon = gauge_info['lon'].values[0]
            area_km2 = gauge_info['area_km2'].values[0]
            river_name = gauge_info['river_name'].values[0]
        else:
            lat = np.nan
            lon = np.nan
            area_km2 = np.nan
            river_name = gauge_name
        
        results.append({
            'gauge_id': gauge_id,
            'gauge_name': gauge_name,
            'river_name': river_name,
            'lat': lat,
            'lon': lon,
            'area_km2': area_km2,
            'mean_discharge_m3s': mean_val,
            'std_discharge_m3s': std_val,
            'min_discharge_m3s': min_val,
            'max_discharge_m3s': max_val,
            'n_months': count
        })
        
        print(f"Processed {gauge_name} ({gauge_id}): {count} months, mean={mean_val:.2f} m3/s")
        
    except FileNotFoundError:
        print(f"Error: Streamflow file not found for {gauge_name} ({gauge_id}): {streamflow_file}")
    except Exception as e:
        print(f"Error processing {gauge_name} ({gauge_id}): {e}")

# Create results dataframe
results_df = pd.DataFrame(results)

# Save results to CSV
output_file = os.path.join(output_dir, "obs_summary_statistics.csv")
try:
    results_df.to_csv(output_file, index=False)
    print(f"\nSaved summary statistics to {output_file}")
except Exception as e:
    print(f"Error saving results: {e}")

# Print summary
print("\n" + "="*80)
print("SUMMARY STATISTICS FOR GRDC GAUGES (1985-1989)")
print("="*80)
print(results_df.to_string(index=False))
print("="*80)
```
