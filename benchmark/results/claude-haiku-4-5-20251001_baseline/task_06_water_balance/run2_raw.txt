#!/usr/bin/env python
"""
E3SM Land Model Water Balance Closure Analysis
Analyzes monthly ELM output for 1985-1989 to check water balance closure.
"""

import os
import sys
import warnings
import numpy as np
import pandas as pd
import xarray as xr
import matplotlib.pyplot as plt
import matplotlib.patches as mpatches
from matplotlib.patches import Polygon as MplPolygon
import cartopy.crs as ccrs
import cartopy.feature as cfeature
from scipy.spatial import cKDTree
import json
from pathlib import Path

warnings.filterwarnings('ignore')

# Configuration
DATA_DIR = "./data/sample/e3sm"
OBS_DIR = "./data/sample/obs"
OUTPUT_DIR = "/Users/zhou014/Local_Drive/Git_repo/esmflow/benchmark/results/claude-haiku-4-5-20251001_baseline/task_06_water_balance/run2_output"
CASE_NAME = "sample.v3.LR.historical"
COMPONENT = "elm"
START_YEAR = 1985
END_YEAR = 1989

# Basin IDs and names
BASINS = {
    3629000: "Amazon",
    4121801: "Missouri",
    4115200: "Columbia",
    6742900: "Danube",
    2969100: "Mekong",
    1159100: "Orange"

def create_output_dir():
    """Create output directory if it doesn't exist."""
    os.makedirs(OUTPUT_DIR, exist_ok=True)
    print(f"Output directory: {OUTPUT_DIR}")

def load_elm_data():
    """Load and concatenate ELM monthly data for 1985-1989."""
    print("Loading ELM data...")
    data_files = []
    
    for year in range(START_YEAR, END_YEAR + 1):
        for month in range(1, 13):
            filename = f"{CASE_NAME}.{COMPONENT}.h0.{year:04d}-{month:02d}.nc"
            filepath = os.path.join(DATA_DIR, "lnd", filename)
            if os.path.exists(filepath):
                data_files.append(filepath)
                print(f"  Found: {filename}")
    
    if not data_files:
        print(f"ERROR: No ELM data files found in {os.path.join(DATA_DIR, 'lnd')}")
        return None
    
    # Load and concatenate
    datasets = [xr.open_dataset(f) for f in sorted(data_files)]
    ds = xr.concat(datasets, dim='time')
    
    print(f"Loaded {len(data_files)} files, time range: {ds.time.values[0]} to {ds.time.values[-1]}")
    return ds

def compute_water_balance_components(ds):
    """Compute water balance components from ELM data."""
    print("Computing water balance components...")
    
    # Precipitation: RAIN + SNOW (kg/m2/s -> mm/day)
    # 1 kg/m2/s = 86.4 mm/day
    rain = ds['RAIN'].fillna(0)  # kg/m2/s
    snow = ds['SNOW'].fillna(0)  # kg/m2/s
    precip = (rain + snow) * 86.4  # mm/day
    
    # Evapotranspiration: QVEGE + QVEGT + QSOIL (mm/s -> mm/day)
    # 1 mm/s = 86400 mm/day
    qvege = ds['QVEGE'].fillna(0)  # mm/s
    qvegt = ds['QVEGT'].fillna(0)  # mm/s
    qsoil = ds['QSOIL'].fillna(0)  # mm/s
    evap = (qvege + qvegt + qsoil) * 86400  # mm/day
    
    # Total runoff: QRUNOFF (mm/s -> mm/day)
    runoff = ds['QRUNOFF'].fillna(0) * 86400  # mm/day
    
    # Water balance residual: P - ET - Q
    residual = precip - evap - runoff
    
    print(f"  Precipitation range: {float(precip.min()):.2f} to {float(precip.max()):.2f} mm/day")
    print(f"  Evapotranspiration range: {float(evap.min()):.2f} to {float(evap.max()):.2f} mm/day")
    print(f"  Runoff range: {float(runoff.min()):.2f} to {float(runoff.max()):.2f} mm/day")
    print(f"  Residual range: {float(residual.min()):.2f} to {float(residual.max()):.2f} mm/day")
    
    return {
        'precip': precip,
        'evap': evap,
        'runoff': runoff,
        'residual': residual
    }

def compute_climatological_mean(components):
    """Compute time-mean fields."""
    print("Computing climatological means...")
    
    clim = {}
    for key, data in components.items():
        clim[key] = data.mean(dim='time')
        print(f"  {key}: {float(clim[key].mean()):.4f} mm/day (global mean)")
    
    return clim

def compute_area_weighted_mean(data, lat_name='lat', lon_name='lon'):
    """Compute area-weighted global mean."""
    if lat_name not in data.dims and lat_name not in data.coords:
        return float(data.mean())
    
    # Get latitude
    if lat_name in data.coords:
        lat = data.coords[lat_name]
    else:
        lat = data[lat_name]
    
    # Compute weights (cosine of latitude)
    weights = np.cos(np.deg2rad(lat))
    
    # Compute weighted mean
    weighted_mean = (data * weights).sum() / weights.sum()
    return float(weighted_mean)

def load_basin_polygons():
    """Load basin polygons from GeoJSON."""
    print("Loading basin polygons...")
    
    geojson_path = os.path.join(OBS_DIR, "basin_polygons.geojson")
    if not os.path.exists(geojson_path):
        print(f"WARNING: Basin polygons not found at {geojson_path}")
        return None
    
    with open(geojson_path, 'r') as f:
        geojson_data = json.load(f)
    
    basins_dict = {}
    for feature in geojson_data['features']:
        grdc_no = feature['properties'].get('grdc_no')
        if grdc_no in BASINS:
            basins_dict[grdc_no] = feature['geometry']
            print(f"  Loaded basin {grdc_no} ({BASINS[grdc_no]})")
    
    return basins_dict

def polygon_to_mask(geometry, lat, lon):
    """Convert GeoJSON polygon to boolean mask."""
    from matplotlib.path import Path
    
    if geometry['type'] == 'Polygon':
        coords = geometry['coordinates'][0]
    elif geometry['type'] == 'MultiPolygon':
        coords = geometry['coordinates'][0][0]
    else:
        return None
    
    # Create mesh grid
    lon_grid, lat_grid = np.meshgrid(lon.values, lat.values)
    points = np.column_stack([lon_grid.ravel(), lat_grid.ravel()])
    
    # Create path and check if points are inside
    path = Path(coords)
    mask = path.contains_points(points).reshape(lon_grid.shape)
    
    return mask

def clip_to_basins(clim_data, basin_polygons, lat, lon):
    """Clip climatological data to basin boundaries."""
    print("Clipping data to basins...")
    
    basin_stats = {}
    
    for basin_id, basin_name in BASINS.items():
        if basin_id not in basin_polygons:
            print(f"  WARNING: Basin {basin_id} ({basin_name}) not found in polygons")
            continue
        
        try:
            mask = polygon_to_mask(basin_polygons[basin_id], lat, lon)
            if mask is None:
                print(f"  WARNING: Could not create mask for basin {basin_id}")
                continue
            
            # Compute basin statistics
            stats = {}
            for key, data in clim_data.items():
                masked_data = data.where(mask, drop=False)
                stats[key] = float(masked_data.mean())
            
            basin_stats[basin_id] = {
                'name': basin_name,
                'mask': mask,
                'stats': stats
            }
            print(f"  {basin_name}: P={stats['precip']:.2f}, ET={stats['evap']:.2f}, Q={stats['runoff']:.2f}, Residual={stats['residual']:.2f} mm/day")
        
        except Exception as e:
            print(f"  ERROR processing basin {basin_id}: {e}")
    
    return basin_stats

def create_composite_figure(clim_data, basin_stats, basin_polygons):
    """Create composite figure with global map and basin bar charts."""
    print("Creating composite figure...")
    
    try:
        fig = plt.figure(figsize=(16, 12))
        
        # Global residual map (top)
        ax_map = plt.subplot(2, 1, 1, projection=ccrs.PlateCarree())
        
        # Plot residual field
        residual = clim_data['residual']
        lat = residual.lat if 'lat' in residual.coords else residual.coords['lat']
        lon = residual.lon if 'lon' in residual.coords else residual.coords['lon']
        
        # Plot with colormap
        vmin, vmax = np.percentile(residual.values, [2, 98])
        im = ax_map.contourf(lon, lat, residual, levels=20, cmap='RdBu_r', 
                             vmin=vmin, vmax=vmax, transform=ccrs.PlateCarree())
        
        # Add basin outlines
        colors = plt.cm.Set3(np.linspace(0, 1, len(BASINS)))
        for idx, (basin_id, basin_name) in enumerate(BASINS.items()):
            if basin_id in basin_polygons:
                geom = basin_polygons[basin_id]
                if geom['type'] == 'Polygon':
                    coords = np.array(geom['coordinates'][0])
                    ax_map.plot(coords[:, 0], coords[:, 1], color=colors[idx], 
                               linewidth=2, label=basin_name, transform=ccrs.PlateCarree())
        
        ax_map.coastlines()
        ax_map.add_feature(cfeature.BORDERS, linestyle=':')
        ax_map.set_global()
        ax_map.set_title('Global Water Balance Residual (P - ET - Q) [mm/day]', fontsize=14, fontweight='bold')
        ax_map.legend(loc='lower left', fontsize=9, ncol=2)
        cbar = plt.colorbar(im, ax=ax_map, orientation='horizontal', pad=0.05, shrink=0.8)
        cbar.set_label('Residual [mm/day]')
        
        # Basin bar charts (bottom)
        ax_bars = plt.subplot(2, 1, 2)
        
        basin_names = []
        precip_vals = []
        evap_vals = []
        runoff_vals = []
        residual_vals = []
        
        for basin_id in sorted(BASINS.keys()):
            if basin_id in basin_stats:
                basin_names.append(BASINS[basin_id])
                stats = basin_stats[basin_id]['stats']
                precip_vals.append(stats['precip'])
                evap_vals.append(stats['evap'])
                runoff_vals.append(stats['runoff'])
                residual_vals.append(stats['residual'])
        
        x = np.arange(len(basin_names))
        width = 0.2
        
        ax_bars.bar(x - 1.5*width, precip_vals, width, label='Precipitation', color='blue', alpha=0.7)
        ax_bars.bar(x - 0.5*width, evap_vals, width, label='Evapotranspiration', color='red', alpha=0.7)
        ax_bars.bar(x + 0.5*width, runoff_vals, width, label='Runoff', color='green', alpha=0.7)
        ax_bars.bar(x + 1.5*width, residual_vals, width, label='Residual', color='orange', alpha=0.7)
        
        ax_bars.set_xlabel('Basin', fontsize=12, fontweight='bold')
        ax_bars.set_ylabel('Water Flux [mm/day]', fontsize=12, fontweight='bold')
        ax_bars.set_title('Water Balance Components by Basin (1985-1989 Climatology)', fontsize=14, fontweight='bold')
        ax_bars.set_xticks(x)
        ax_bars.set_xticklabels(basin_names, rotation=45, ha='right')
        ax_bars.legend(fontsize=10)
        ax_bars.grid(axis='y', alpha=0.3)
        ax_bars.axhline(y=0, color='black', linestyle='-', linewidth=0.5)
        
        plt.tight_layout()
        
        output_file = os.path.join(OUTPUT_DIR, "water_balance_composite.png")
        plt.savefig(output_file, dpi=150, bbox_inches='tight')
        print(f"  Saved: {output_file}")
        plt.close()
    
    except Exception as e:
        print(f"ERROR creating composite figure: {e}")

def save_global_statistics(clim_data):
    """Save global statistics to CSV."""
    print("Saving global statistics...")
    
    try:
        stats = {}
        for key, data in clim_data.items():
            lat = data.lat if 'lat' in data.coords else data.coords['lat']
            lon = data.lon if 'lon' in data.coords else data.coords['lon']
            
            # Area-weighted mean
            weights = np.cos(np.deg2rad(lat))
            weighted_mean = float((data * weights).sum() / weights.sum())
            
            # Simple mean
            simple_mean = float(data.mean())
            
            # Std dev
            std = float(data.std())
            
            stats[key] = {
                'area_weighted_mean': weighted_mean,
                'simple_mean': simple_mean,
                'std_dev': std,
                'min': float(data.min()),
                'max': float(data.max())
            }
        
        df = pd.DataFrame(stats).T
        output_file = os.path.join(OUTPUT_DIR, "global_statistics.csv")
        df.to_csv(output_file)
        print(f"  Saved: {output_file}")
        print(df)
    
    except Exception as e:
        print(f"ERROR saving global statistics: {e}")

def save_basin_statistics(basin_stats):
    """Save basin statistics to CSV."""
    print("Saving basin statistics...")
    
    try:
        data_list = []
        for basin_id in sorted(BASINS.keys()):
            if basin_id in basin_stats:
                row = {'basin_id': basin_id, 'basin_name': BASINS[basin_id]}
                row.update(basin_stats[basin_id]['stats'])
                data_list.append(row)
        
        df = pd.DataFrame(data_list)
        output_file = os.path.join(OUTPUT_DIR, "basin_statistics.csv")
        df.to_csv(output_file, index=False)
        print(f"  Saved: {output_file}")
        print(df)
    
    except Exception as e:
        print(f"ERROR saving basin statistics: {e}")

def save_netcdf_output(clim_data):
    """Save climatological fields to NetCDF."""
    print("Saving NetCDF output...")
    
    try:
        # Create dataset
        ds_out = xr.Dataset({
            'precipitation': clim_data['precip'],
            'evapotranspiration': clim_data['evap'],
            'runoff': clim_data['runoff'],
            'residual': clim_data['residual']
        })
        
        # Add attributes
        ds_out['precipitation'].attrs = {'units': 'mm/day', 'long_name': 'Total Precipitation (RAIN + SNOW)'}
        ds_out['evapotranspiration'].attrs = {'units': 'mm/day', 'long_name': 'Total Evapotranspiration (QVEGE + QVEGT + QSOIL)'}
        ds_out['runoff'].attrs = {'units': 'mm/day', 'long_name': 'Total Runoff (QRUNOFF)'}
        ds_out['residual'].attrs = {'units': 'mm/day', 'long_name': 'Water Balance Residual (P - ET - Q)'}
        
        output_file = os.path.join(OUTPUT_DIR, "water_balance_climatology.nc")
        ds_out.to_netcdf(output_file)
        print(f"  Saved: {output_file}")
    
    except Exception as e:
        print(f"ERROR saving NetCDF output: {e}")

def main():
    """Main analysis function."""
    print("=" * 80)
    print("E3SM Land Model Water Balance Closure Analysis")
    print("=" * 80)
    
    create_output_dir()
    
    # Load data
    ds = load_elm_data()
    if ds is None:
        print("ERROR: Could not load ELM data")
        return
    
    # Compute water balance components
    components = compute_water_balance_components(ds)
    
    # Compute climatological means
    clim_data = compute_climatological_mean(components)
    
    # Load basin polygons
    basin_polygons = load_basin_polygons()
    if basin_polygons is None:
        print("WARNING: Could not load basin polygons, skipping basin analysis")
        basin_stats = {}
    else:
        # Clip to basins
        lat = clim_data['precip'].lat if 'lat' in clim_data['precip'].coords else clim_data['precip'].coords['lat']
        lon = clim_data['precip'].lon if 'lon' in clim_data['precip'].coords else clim_data['precip'].coords['lon']
        basin_stats = clip_to_basins(clim_data, basin_polygons, lat, lon)
    
    # Save outputs
    save_global_statistics(clim_data)
    save_basin_statistics(basin_stats)
    save_netcdf_output(clim_data)
    
    # Create figures
    if basin_polygons:
        create_composite_figure(clim_data, basin_stats, basin_polygons)
    
    print("=" * 80)
    print("Analysis complete!")
    print("=" * 80)

if __name__ == "__main__":
    main()
