name: "ELM Runoff Climatology 1985-1989"
description: |
  Extract the ELM QRUNOFF field for 1985-1989, compute the area-weighted global mean, and plot the mean runoff map with global statistics overlaid.

settings:
  case_name: "sample.v3.LR.historical"
  data_dir: "./data/sample/e3sm/"
  output_dir: "./output/task02_seasonal_runoff"
  style: cream_ink

steps:
  - id: extract_qrunoff_mean_field
    tool: extract_gridded_field
    params:
      data_dir: "${settings.data_dir}"
      case_name: "${settings.case_name}"
      component: elm
      variable: "QRUNOFF"
      years: [1985, 1986, 1987, 1988, 1989]
    outputs:
      field_file: qrunoff_mean_1985_1989.nc

  - id: compute_global_mean_runoff
    tool: compute_zonal_stats
    params:
      field_file: "${extract_qrunoff_mean_field.outputs.field_file}"
    outputs:
      stats_file: qrunoff_global_stats_1985_1989.csv

  - id: plot_runoff_map
    tool: plot_gridded_map
    params:
      field_file: "${extract_qrunoff_mean_field.outputs.field_file}"
      title: "ELM QRUNOFF Mean (1985-1989)"
      stats_file: "${compute_global_mean_runoff.outputs.stats_file}"
    outputs:
      plot_file: qrunoff_mean_map_1985_1989.png
