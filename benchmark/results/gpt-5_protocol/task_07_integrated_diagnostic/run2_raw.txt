name: "Integrated Water Cycle Diagnostics 1985-1989 for Six Major Basins"
description: |
  Evaluate E3SM per-basin water cycle components and streamflow distributional metrics over 1985-1989
  for six major basins (Amazon 3629000, Missouri 4121801, Columbia 4115200, Danube 6742900, Mekong 2969100, Orange 1159100).
  Extract model fields (P, ET, Q), fetch and extract observation fields (GPCC pr, MODIS ET, LORA runoff),
  compute basin means, derive flow duration curve metrics from MOSART-simulated and observed discharge,
  combine into a per-basin budget summary, and visualize with grouped bar charts and radar diagnostics.

settings:
  case_name: "sample.v3.LR.historical"
  data_dir: "./data/sample/e3sm/"
  output_dir: "./output/task07_integrated_diagnostic"
  style: cream_ink
  basins_file: "./data/sample/obs/basin_polygons.geojson"
  gauge_metadata_file: "./data/sample/obs/gauge_metadata.csv"
  obs_streamflow_dir: "./data/sample/obs/streamflow"
  years: [1985, 1986, 1987, 1988, 1989]
  gauge_ids: "3629000,4121801,4115200,6742900,2969100,1159100"

steps:
  - id: load_gauge_metadata
    tool: load_obs_metadata
    params:
      metadata_file: "${settings.gauge_metadata_file}"
    outputs:
      metadata_file: gauge_metadata.valid.csv
      n_gauges: n_gauges.txt

  - id: match_gauges_to_mosart
    tool: match_to_grid
    params:
      gauge_metadata: "${load_gauge_metadata.outputs.metadata_file}"
      data_dir: "${settings.data_dir}"
      case_name: "${settings.case_name}"
      component: mosart
    outputs:
      matched_file: matched_gauges_mosart.csv
      n_matched: n_matched_mosart.txt

  - id: extract_model_precip_field
    tool: extract_gridded_field
    params:
      data_dir: "${settings.data_dir}"
      case_name: "${settings.case_name}"
      component: elm
      variable: "RAIN+SNOW"
      years: ${settings.years}
    outputs:
      field_file: model_precip_1985-1989.nc

  - id: extract_model_et_field
    tool: extract_gridded_field
    params:
      data_dir: "${settings.data_dir}"
      case_name: "${settings.case_name}"
      component: elm
      variable: "QVEGE+QVEGT+QSOIL"
      years: ${settings.years}
    outputs:
      field_file: model_et_1985-1989.nc

  - id: extract_model_runoff_field
    tool: extract_gridded_field
    params:
      data_dir: "${settings.data_dir}"
      case_name: "${settings.case_name}"
      component: elm
      variable: "QRUNOFF"
      years: ${settings.years}
    outputs:
      field_file: model_runoff_1985-1989.nc

  - id: fetch_obs_precip
    tool: fetch_ilamb_data
    params:
      variable: pr
      dataset: GPCCv2018
    outputs:
      data_file: ilamb_pr_GPCCv2018.nc

  - id: extract_obs_precip_field
    tool: extract_gridded_field
    params:
      obs_file: "${fetch_obs_precip.outputs.data_file}"
      variable: pr
      years: ${settings.years}
    outputs:
      field_file: obs_precip_1985-1989.nc

  - id: fetch_obs_et
    tool: fetch_ilamb_data
    params:
      variable: evspsbl
      dataset: MODIS
    outputs:
      data_file: ilamb_et_MODIS.nc

  - id: extract_obs_et_field
    tool: extract_gridded_field
    params:
      obs_file: "${fetch_obs_et.outputs.data_file}"
      variable: et
      years: ${settings.years}
    outputs:
      field_file: obs_et_1985-1989.nc

  - id: fetch_obs_runoff
    tool: fetch_ilamb_data
    params:
      variable: mrro
      dataset: LORA
    outputs:
      data_file: ilamb_mrro_LORA.nc

  - id: extract_obs_runoff_field
    tool: extract_gridded_field
    params:
      obs_file: "${fetch_obs_runoff.outputs.data_file}"
      variable: mrro
      years: ${settings.years}
    outputs:
      field_file: obs_runoff_1985-1989.nc

  - id: basin_means_model_precip
    tool: extract_basin_mean
    params:
      field_file: "${extract_model_precip_field.outputs.field_file}"
      basins_file: "${settings.basins_file}"
      gauge_ids: "${settings.gauge_ids}"
    outputs:
      basin_means_file: basin_means_model_precip.csv

  - id: basin_means_model_et
    tool: extract_basin_mean
    params:
      field_file: "${extract_model_et_field.outputs.field_file}"
      basins_file: "${settings.basins_file}"
      gauge_ids: "${settings.gauge_ids}"
    outputs:
      basin_means_file: basin_means_model_et.csv

  - id: basin_means_model_runoff
    tool: extract_basin_mean
    params:
      field_file: "${extract_model_runoff_field.outputs.field_file}"
      basins_file: "${settings.basins_file}"
      gauge_ids: "${settings.gauge_ids}"
    outputs:
      basin_means_file: basin_means_model_runoff.csv

  - id: basin_means_obs_precip
    tool: extract_basin_mean
    params:
      field_file: "${extract_obs_precip_field.outputs.field_file}"
      basins_file: "${settings.basins_file}"
      gauge_ids: "${settings.gauge_ids}"
    outputs:
      basin_means_file: basin_means_obs_precip.csv

  - id: basin_means_obs_et
    tool: extract_basin_mean
    params:
      field_file: "${extract_obs_et_field.outputs.field_file}"
      basins_file: "${settings.basins_file}"
      gauge_ids: "${settings.gauge_ids}"
    outputs:
      basin_means_file: basin_means_obs_et.csv

  - id: basin_means_obs_runoff
    tool: extract_basin_mean
    params:
      field_file: "${extract_obs_runoff_field.outputs.field_file}"
      basins_file: "${settings.basins_file}"
      gauge_ids: "${settings.gauge_ids}"
    outputs:
      basin_means_file: basin_means_obs_runoff.csv

  - id: extract_sim_discharge
    tool: extract_e3sm_timeseries
    params:
      data_dir: "${settings.data_dir}"
      case_name: "${settings.case_name}"
      component: mosart
      variable: RIVER_DISCHARGE_OVER_LAND_LIQ
      locations_file: "${match_gauges_to_mosart.outputs.matched_file}"
      years: ${settings.years}
    outputs:
      timeseries_file: sim_discharge_1985-1989.csv
      n_locations: n_locations_sim.txt

  - id: extract_obs_discharge
    tool: extract_obs_timeseries
    params:
      obs_dir: "${settings.obs_streamflow_dir}"
      gauge_metadata: "${load_gauge_metadata.outputs.metadata_file}"
      years: ${settings.years}
      gauge_ids: "${settings.gauge_ids}"
    outputs:
      timeseries_file: obs_discharge_1985-1989.csv
      n_gauges: n_gauges_obs_timeseries.txt

  - id: compute_fdc
    tool: compute_fdc_metrics
    params:
      sim_file: "${extract_sim_discharge.outputs.timeseries_file}"
      obs_file: "${extract_obs_discharge.outputs.timeseries_file}"
    outputs:
      metrics_file: fdc_metrics_1985-1989.csv
      fdc_file: fdc_percentiles_1985-1989.csv

  - id: compute_basin_budget_summary
    tool: compute_basin_budget
    params:
      model_precip_file: "${basin_means_model_precip.outputs.basin_means_file}"
      model_et_file: "${basin_means_model_et.outputs.basin_means_file}"
      model_runoff_file: "${basin_means_model_runoff.outputs.basin_means_file}"
      obs_precip_file: "${basin_means_obs_precip.outputs.basin_means_file}"
      obs_et_file: "${basin_means_obs_et.outputs.basin_means_file}"
      obs_runoff_file: "${basin_means_obs_runoff.outputs.basin_means_file}"
      fdc_metrics_file: "${compute_fdc.outputs.metrics_file}"
      locations_file: "${load_gauge_metadata.outputs.metadata_file}"
    outputs:
      budget_file: basin_budget_summary_1985-1989.csv

  - id: plot_budget_bars
    tool: plot_basin_budget_comparison
    params:
      budget_file: "${compute_basin_budget_summary.outputs.budget_file}"
    outputs:
      plot_file: basin_budget_comparison_1985-1989.png

  - id: plot_budget_radar
    tool: plot_basin_radar
    params:
      budget_file: "${compute_basin_budget_summary.outputs.budget_file}"
    outputs:
      plot_file: basin_budget_radar_1985-1989.png
