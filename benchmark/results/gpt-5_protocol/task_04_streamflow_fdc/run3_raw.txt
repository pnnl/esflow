name: "Streamflow FDC 1985-1989"
description: |
  Match gauges to E3SM MOSART grid, extract daily simulated and observed discharge for 1985-1989,
  compute flow duration curve (FDC) distributional metrics, save metrics and FDC percentile CSVs,
  and plot a map of Wasserstein distance with FDC panels for selected gauges.

settings:
  case_name: "sample.v3.LR.historical"
  data_dir: "./data/sample/e3sm/"
  output_dir: "./output/task04_streamflow_fdc"
  style: cream_ink
  obs_dir: "./data/sample/obs/streamflow/"
  obs_metadata: "./data/sample/obs/gauge_metadata.csv"

steps:
  - id: load_metadata
    tool: load_obs_metadata
    params:
      metadata_file: "${settings.obs_metadata}"
    outputs:
      metadata_file: "metadata_validated.csv"
      n_gauges: "metadata_ngauges.txt"

  - id: match_gauges
    tool: match_to_grid
    params:
      gauge_metadata: "${load_metadata.outputs.metadata_file}"
      data_dir: "${settings.data_dir}"
      case_name: "${settings.case_name}"
      component: "mosart"
    outputs:
      matched_file: "matched_gauges.csv"
      n_matched: "n_matched.txt"

  - id: extract_sim_timeseries
    tool: extract_e3sm_timeseries
    params:
      data_dir: "${settings.data_dir}"
      case_name: "${settings.case_name}"
      component: "mosart"
      variable: "RIVER_DISCHARGE_OVER_LAND_LIQ"
      locations_file: "${match_gauges.outputs.matched_file}"
      years:
        - 1985
        - 1986
        - 1987
        - 1988
        - 1989
      frequency: "daily"
    outputs:
      timeseries_file: "sim_discharge_daily_1985_1989.csv"
      n_locations: "sim_nlocations.txt"

  - id: extract_obs_timeseries
    tool: extract_obs_timeseries
    params:
      obs_dir: "${settings.obs_dir}"
      gauge_metadata: "${load_metadata.outputs.metadata_file}"
      years:
        - 1985
        - 1986
        - 1987
        - 1988
        - 1989
      gauge_ids: ""
    outputs:
      timeseries_file: "obs_discharge_daily_1985_1989.csv"
      n_gauges: "obs_ngauges.txt"

  - id: compute_fdc
    tool: compute_fdc_metrics
    params:
      sim_file: "${extract_sim_timeseries.outputs.timeseries_file}"
      obs_file: "${extract_obs_timeseries.outputs.timeseries_file}"
    outputs:
      metrics_file: "fdc_metrics_1985_1989.csv"
      fdc_file: "fdc_percentiles_1985_1989.csv"

  - id: plot_fdc_panels
    tool: plot_fdc
    params:
      metrics_file: "${compute_fdc.outputs.metrics_file}"
      fdc_file: "${compute_fdc.outputs.fdc_file}"
      locations_file: "${load_metadata.outputs.metadata_file}"
      metric: "wasserstein"
      gauge_ids: "3629000,4121801,4115200,6742900,2969100,1159100"
    outputs:
      plot_file: "fdc_wasserstein_map_and_panels_1985_1989.png"
