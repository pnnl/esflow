name: "Irrigation Demand 1985-1989"
description: |
  Extract irrigation water demand from ELM output (1985-1989), compute the global mean,
  and plot a global map of the mean irrigation water demand. Saves a CSV of the global mean
  and a PNG map.
settings:
  case_name: "*"
  data_dir: "./data/sample/model/"
  output_dir: "./output/test_irrigation_demand"
  style: cream_ink
steps:
  - id: extract_irrigation_field
    tool: extract_gridded_field
    params:
      data_dir: "${settings.data_dir}"
      case_name: "${settings.case_name}"
      component: "elm"
      variable: "QIRRIG_DEMAND"
      years: [1985, 1986, 1987, 1988, 1989]
    outputs:
      field_file: "irrigation_demand_1985-1989_mean.nc"
  - id: compute_global_mean
    tool: compute_zonal_stats
    params:
      field_file: "${extract_irrigation_field.outputs.field_file}"
      variable: ""
      lat_bands: ""
    outputs:
      stats_file: "irrigation_demand_1985-1989_global_mean.csv"
  - id: plot_irrigation_map
    tool: plot_gridded_map
    params:
      field_file: "${extract_irrigation_field.outputs.field_file}"
      variable: ""
      title: "Mean irrigation water demand (1985-1989)"
      units: ""
      stats_file: "${compute_global_mean.outputs.stats_file}"
    outputs:
      plot_file: "irrigation_demand_1985-1989_map.png"
