name: "GRDC Obs Monthly Streamflow Summary 1985-1989"
description: |
  Load GRDC gauge metadata, extract monthly streamflow observations for six selected gauges from 1985-1989, and compute per-gauge summary statistics.

settings:
  case_name: "obs_only"
  data_dir: "./data/sample"
  output_dir: "./output/task01_obs_summary"
  style: cream_ink

steps:
  - id: load_gauge_metadata
    tool: load_obs_metadata
    params:
      metadata_file: "./data/sample/obs/gauge_metadata.csv"
    outputs:
      metadata_file: "${settings.output_dir}/gauge_metadata_validated.csv"

  - id: extract_obs_streamflow
    tool: extract_obs_timeseries
    params:
      obs_dir: "./data/sample/obs/streamflow"
      gauge_metadata: "${load_gauge_metadata.outputs.metadata_file}"
      years: [1985, 1986, 1987, 1988, 1989]
      gauge_ids: "3629000,4121801,4115200,6742900,2969100,1159100"
    outputs:
      timeseries_file: "${settings.output_dir}/obs_streamflow_1985_1989.csv"

  - id: compute_obs_summary_stats
    tool: compute_summary_stats
    params:
      timeseries_file: "${extract_obs_streamflow.outputs.timeseries_file}"
      gauge_metadata: "${load_gauge_metadata.outputs.metadata_file}"
      top_n: 0
      rank_by: "mean"
    outputs:
      stats_file: "${settings.output_dir}/obs_streamflow_summary_1985_1989.csv"
