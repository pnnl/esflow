name: "Basin Streamflow Validation 1985-1989"
description: |
  Match gauges to MOSART grid, extract simulated and observed discharge for 1985-1989,
  compute validation metrics (RMSE, NSE, KGE, PBIAS), and generate per-basin
  figures showing basin boundaries and sim-vs-obs time series for six major basins:
  Amazon (3629000), Missouri (4121801), Columbia (4115200), Danube (6742900),
  Mekong (2969100), and Orange (1159100).

settings:
  case_name: "sample.v3.LR.historical"
  data_dir: "./data/sample/e3sm/"
  output_dir: "./output/task05_basin_streamflow"
  style: cream_ink

steps:
  - id: load_gauge_metadata
    tool: load_obs_metadata
    params:
      metadata_file: "./data/sample/obs/gauge_metadata.csv"
    outputs:
      metadata_file: "gauges.validated.csv"
      n_gauges: "gauges.count.txt"

  - id: match_gauges_to_mosart_grid
    tool: match_to_grid
    params:
      gauge_metadata: "${load_gauge_metadata.outputs.metadata_file}"
      data_dir: "${settings.data_dir}"
      case_name: "${settings.case_name}"
      component: "mosart"
    outputs:
      matched_file: "gauges.matched.csv"
      n_matched: "gauges.matched.count.txt"

  - id: extract_sim_discharge
    tool: extract_e3sm_timeseries
    params:
      data_dir: "${settings.data_dir}"
      case_name: "${settings.case_name}"
      component: "mosart"
      variable: "RIVER_DISCHARGE_OVER_LAND_LIQ"
      locations_file: "${match_gauges_to_mosart_grid.outputs.matched_file}"
      years: [1985, 1986, 1987, 1988, 1989]
      frequency: "monthly"
    outputs:
      timeseries_file: "sim_discharge_1985-1989.csv"
      n_locations: "sim_locations.count.txt"

  - id: extract_obs_discharge
    tool: extract_obs_timeseries
    params:
      obs_dir: "./data/sample/obs/streamflow/"
      gauge_metadata: "${load_gauge_metadata.outputs.metadata_file}"
      years: [1985, 1986, 1987, 1988, 1989]
      gauge_ids: "3629000,4121801,4115200,6742900,2969100,1159100"
    outputs:
      timeseries_file: "obs_discharge_1985-1989.csv"
      n_gauges: "obs_gauges.count.txt"

  - id: compute_validation_metrics
    tool: compute_metrics
    params:
      sim_file: "${extract_sim_discharge.outputs.timeseries_file}"
      obs_file: "${extract_obs_discharge.outputs.timeseries_file}"
    outputs:
      metrics_file: "metrics_discharge_1985-1989.csv"
      n_gauges: "metrics.count.txt"

  - id: plot_basin_timeseries_panels
    tool: plot_basin_timeseries
    params:
      basins_file: "./data/sample/obs/basin_polygons.geojson"
      sim_file: "${extract_sim_discharge.outputs.timeseries_file}"
      obs_file: "${extract_obs_discharge.outputs.timeseries_file}"
      locations_file: "${load_gauge_metadata.outputs.metadata_file}"
      gauge_ids: "3629000,4121801,4115200,6742900,2969100,1159100"
      metrics_file: "${compute_validation_metrics.outputs.metrics_file}"
    outputs:
      plot_files: "basin_timeseries_plots.json"
