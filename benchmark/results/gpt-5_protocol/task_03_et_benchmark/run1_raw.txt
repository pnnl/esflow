name: "ELM ET vs MODIS Benchmark (1985-1989)"
description: |
  Benchmark E3SM ELM evapotranspiration against MODIS observations using the 5-year climatology (1985-1989). Fetch MODIS ET, extract model and obs time-mean ET fields, compute spatial bias (model − obs) with global stats, and plot the bias map.

settings:
  case_name: "sample.v3.LR.historical"
  data_dir: "./data/sample/e3sm/"
  output_dir: "./output/task03_et_benchmark"
  style: cream_ink

steps:
  - id: fetch_modis_et
    tool: fetch_ilamb_data
    params:
      variable: evspsbl
      dataset: MODIS
    outputs:
      data_file: modis_et.nc

  - id: extract_model_et
    tool: extract_gridded_field
    params:
      data_dir: "${settings.data_dir}"
      case_name: "${settings.case_name}"
      component: elm
      variable: "QVEGE+QVEGT+QSOIL"
      years:
        - 1985
        - 1986
        - 1987
        - 1988
        - 1989
    outputs:
      field_file: elm_et_mean_1985-1989.nc

  - id: extract_obs_et
    tool: extract_gridded_field
    params:
      obs_file: "${fetch_modis_et.outputs.data_file}"
      variable: et
      years:
        - 1985
        - 1986
        - 1987
        - 1988
        - 1989
    outputs:
      field_file: modis_et_mean_1985-1989.nc

  - id: compute_bias
    tool: compute_spatial_bias
    params:
      field_a: "${extract_model_et.outputs.field_file}"
      field_b: "${extract_obs_et.outputs.field_file}"
    outputs:
      bias_file: et_bias_model_minus_obs.nc
      stats_file: et_bias_stats.csv

  - id: plot_bias_map
    tool: plot_gridded_map
    params:
      field_file: "${compute_bias.outputs.bias_file}"
      title: "ELM − MODIS ET Bias (1985-1989)"
    outputs:
      plot_file: et_bias_map.png
