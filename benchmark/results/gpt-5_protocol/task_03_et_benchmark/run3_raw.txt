name: "ET Benchmark vs MODIS (1985-1989)"
description: |
  Benchmark E3SM ELM evapotranspiration against MODIS observations using a 5-year (1985-1989) climatology. Fetch MODIS ET, extract time-mean ET from model (QVEGE+QVEGT+QSOIL) and observations, compute spatial bias (model minus obs) with global statistics, and plot the bias map.

settings:
  case_name: "sample.v3.LR.historical"
  data_dir: "./data/sample/e3sm/"
  output_dir: "./output/task03_et_benchmark"
  style: cream_ink

steps:
  - id: fetch_modis_et
    tool: fetch_ilamb_data
    params:
      variable: evspsbl
      dataset: MODIS
    outputs:
      data_file: modis_et.nc

  - id: extract_model_et
    tool: extract_gridded_field
    params:
      data_dir: "${settings.data_dir}"
      case_name: "${settings.case_name}"
      component: elm
      variable: "QVEGE+QVEGT+QSOIL"
      years:
        - 1985
        - 1986
        - 1987
        - 1988
        - 1989
    outputs:
      field_file: model_et_1985-1989.nc

  - id: extract_obs_et
    tool: extract_gridded_field
    params:
      obs_file: "${fetch_modis_et.outputs.data_file}"
      variable: et
      years:
        - 1985
        - 1986
        - 1987
        - 1988
        - 1989
    outputs:
      field_file: obs_et_1985-1989.nc

  - id: compute_et_bias
    tool: compute_spatial_bias
    params:
      field_a: "${extract_model_et.outputs.field_file}"
      field_b: "${extract_obs_et.outputs.field_file}"
    outputs:
      bias_file: et_bias_model_minus_modis.nc
      stats_file: et_bias_stats.csv

  - id: plot_bias_map
    tool: plot_gridded_map
    params:
      field_file: "${compute_et_bias.outputs.bias_file}"
      title: "ET Bias (Model − MODIS), 1985–1989"
    outputs:
      plot_file: et_bias_map.png
