def find_geometry_by_grdc(geojson_data, gauge_id_str):
    if geojson_data is None:
        return None
    features = geojson_data.get("features", [])
    for feat in features:
        props = feat.get("properties", {})
        grdc_no = str(props.get("grdc_no", "")).strip()
        if grdc_no == str(gauge_id_str).strip():
            return feat.get("geometry", None)
    return None

def polygon_bounds_from_geojson_geom(geom):
    # Return minx, miny, maxx, maxy in lon/lat
    if geom is None:
        return None
    minx = miny = np.inf
    maxx = maxy = -np.inf
    gtype = geom.get("type", "")
    coords = geom.get("coordinates", [])
    if gtype == "Polygon":
        polys = [coords]
    elif gtype == "MultiPolygon":
        polys = coords
    else:
        return None
    for poly in polys:
        # poly is list of rings; take all rings for bounds
        for ring in poly:
            xs = [c[0] for c in ring]
            ys = [c[1] for c in ring]
            if len(xs) == 0:
                continue
            minx = min(minx, np.min(xs))
            maxx = max(maxx, np.max(xs))
            miny = min(miny, np.min(ys))
            maxy = max(maxy, np.max(ys))
    if not np.isfinite(minx):
        return None
    return (minx, miny, maxx, maxy)

def plot_polygon_on_axes(ax, geom, edgecolor="k", linewidth=1.5):
    if geom is None:
        return
    gtype = geom.get("type", "")
    coords = geom.get("coordinates", [])
    if shapely_shape is not None:
        try:
            shp = shapely_shape(geom)
            ax.add_geometries([shp], crs=ccrs.PlateCarree(), facecolor="none", edgecolor=edgecolor, linewidth=linewidth)
            return
        except Exception:
            pass
    # Fallback manual plotting without shapely
    if gtype == "Polygon":
        polys = [coords]
    elif gtype == "MultiPolygon":
        polys = coords
    else:
        polys = []
    for poly in polys:
        # Outer ring is poly[0]
        if len(poly) == 0:
            continue
        outer = poly[0]
        xs = [c[0] for c in outer]
        ys = [c[1] for c in outer]
        ax.plot(xs, ys, transform=ccrs.PlateCarree(), color=edgecolor, linewidth=linewidth)

def haversine_dist(lat1, lon1, lat2, lon2):
    # lat/lon in degrees; returns distance in km
    R = 6371.0
    lat1r = np.radians(lat1)
    lat2r = np.radians(lat2)
    dlat = lat2r - lat1r
    dlon = np.radians(lon2 - lon1)
    a = np.sin(dlat / 2.0)**2 + np.cos(lat1r) * np.cos(lat2r) * np.sin(dlon / 2.0)**2
    c = 2 * np.arcsin(np.sqrt(a))
    return R * c

def find_lat_lon_vars(ds):
    lat_name = None
    lon_name = None
    for cand in LAT_NAME_CANDIDATES:
        if cand in ds.variables:
            lat_name = cand
            break
    for cand in LON_NAME_CANDIDATES:
        if cand in ds.variables:
            lon_name = cand
            break
    # Fallback search in coords
    if lat_name is None:
        for cand in LAT_NAME_CANDIDATES:
            if cand in ds.coords:
                lat_name = cand
                break
    if lon_name is None:
        for cand in LON_NAME_CANDIDATES:
            if cand in ds.coords:
                lon_name = cand
                break
    if lat_name is None or lon_name is None:
        # Try to detect by attributes
        for v in ds.variables:
            var = ds[v]
            units = str(var.attrs.get("units", "")).lower()
            standard_name = str(var.attrs.get("standard_name", "")).lower()
            if ("degree" in units or "degrees_north" in units) and ("lat" in v.lower() or "latitude" in standard_name or "lat" in standard_name):
                lat_name = v
            if ("degree" in units or "degrees_east" in units) and ("lon" in v.lower() or "longitude" in standard_name or "lon" in standard_name):
                lon_name = v
    return lat_name, lon_name

def adjust_gauge_lon_to_grid(gauge_lon, grid_lon_values):
    lon_vals = np.asarray(grid_lon_values).ravel()
    if not np.isfinite(lon_vals).any():
        return gauge_lon
    lon_min = np.nanmin(lon_vals)
    lon_max = np.nanmax(lon_vals)
    if lon_max > 180 and lon_min >= 0:
        # Grid in 0-360
        return gauge_lon % 360.0
    else:
        # Grid in -180..180
        lon = gauge_lon
        if lon > 180:
            lon = lon - 360.0
        return lon

def nearest_grid_index(ds, lat_name, lon_name, pt_lat, pt_lon, river_var):
    latv = ds[lat_name]
    lonv = ds[lon_name]
    # Adjust lon to grid convention
    adj_pt_lon = adjust_gauge_lon_to_grid(pt_lon, lonv.values)
    # Build coordinate arrays
    dims = None
    is_1d_same_dim = False
    if latv.ndim == 1 and lonv.ndim == 1:
        if latv.dims == lonv.dims and latv.size == lonv.size:
            # Possibly unstructured grid (1D)
            is_1d_same_dim = True
        else:
            # Structured grid; meshgrid to 2D
            lat2d = np.repeat(latv.values[:, None], lonv.size, axis=1)
            lon2d = np.repeat(lonv.values[None, :], latv.size, axis=0)
            dims = (latv.dims[0], lonv.dims[0])
    if latv.ndim == 2 and lonv.ndim == 2:
        lat2d = latv.values
        lon2d = lonv.values
        dims = latv.dims

    if is_1d_same_dim:
        # 1D grid along single dimension
        lat1d = latv.values
        lon1d = lonv.values
        dists = haversine_dist(pt_lat, adj_pt_lon, lat1d, lon1d)
        idx = int(np.nanargmin(dists))
        dims_map = {latv.dims[0]: idx} if latv.dims[0] in ds[river_var].dims else {}
        if not dims_map and lonv.dims[0] in ds[river_var].dims:
            dims_map = {lonv.dims[0]: idx}
        # Return nearest gridpoint coords
        try:
            near_lat = float(lat1d[idx])
            near_lon = float(lon1d[idx])
        except Exception:
            near_lat = np.nan
            near_lon = np.nan
        return dims_map, near_lat, near_lon

    if dims is not None:
        # Create flat arrays to compute distance
        flat_lat = lat2d.ravel()
        flat_lon = lon2d.ravel()
        dists = haversine_dist(pt_lat, adj_pt_lon, flat_lat, flat_lon)
        k = int(np.nanargmin(dists))
        ny, nx = lat2d.shape
        iy, ix = np.unravel_index(k, (ny, nx))
        # Map to variable dims
        var_dims = ds[river_var].dims
        dims_map = {}
        # dims is e.g., ('y', 'x') or ('lat', 'lon')
        if dims[0] in var_dims and dims[1] in var_dims:
            dims_map[dims[0]] = int(iy)
            dims_map[dims[1]] = int(ix)
        else:
            # Try to find matching var dims by position
            if len(var_dims) >= 2:
                sdims = [d for d in var_dims if d != "time"]
                if len(sdims) >= 2:
                    dims_map[sdims[0]] = int(iy)
                    dims_map[sdims[1]] = int(ix)
        try:
            near_lat = float(lat2d[iy, ix])
            near_lon = float(lon2d[iy, ix])
        except Exception:
            near_lat = np.nan
            near_lon = np.nan
        return dims_map, near_lat, near_lon

    # Fallback: try to isel with heuristic on var dims
    sdims = [d for d in ds[river_var].dims if d != "time"]
    dims_map = {}
    if len(sdims) == 1:
        # Loop through a sample of points to find nearest
        npts = ds[river_var].sizes[sdims[0]]
        try:
            lat1d = latv.values
            lon1d = lonv.values
            dists = haversine_dist(pt_lat, adj_pt_lon, lat1d, lon1d)
            idx = int(np.nanargmin(dists))
            dims_map[sdims[0]] = idx
            near_lat = float(lat1d[idx])
            near_lon = float(lon1d[idx])
            return dims_map, near_lat, near_lon
        except Exception:
            pass
    return {}, np.nan, np.nan

def compute_metrics(sim, obs):
    # sim, obs: numpy arrays aligned, with no NaNs
    sim = np.asarray(sim, dtype=float)
    obs = np.asarray(obs, dtype=float)
    mask = np.isfinite(sim) & np.isfinite(obs)
    sim = sim[mask]
    obs = obs[mask]
    n = sim.size
    if n == 0:
        return dict(RMSE=np.nan, NSE=np.nan, KGE=np.nan, PBIAS=np.nan, N=0)
    rmse = np.sqrt(np.mean((sim - obs) ** 2))
    obs_mean = np.mean(obs)
    denom = np.sum((obs - obs_mean) ** 2)
    if denom == 0:
        nse = np.nan
    else:
        nse = 1.0 - np.sum((sim - obs) ** 2) / denom
    # KGE
    if np.std(sim) == 0 or np.std(obs) == 0:
        r = np.corrcoef(sim, obs)[0, 1] if (np.std(sim) != 0 and np.std(obs) != 0) else np.nan
    else:
        r = np.corrcoef(sim, obs)[0, 1]
    alpha = np.std(sim) / np.std(obs) if np.std(obs) != 0 else np.nan
    beta = np.mean(sim) / np.mean(obs) if np.mean(obs) != 0 else np.nan
    if np.any(np.isnan([r, alpha, beta])):
        kge = np.nan
    else:
        kge = 1.0 - np.sqrt((r - 1.0)**2 + (alpha - 1.0)**2 + (beta - 1.0)**2)
    # PBIAS
    denom_pbias = np.sum(obs)
    if denom_pbias == 0:
        pbias = np.nan
    else:
        pbias = 100.0 * np.sum(sim - obs) / denom_pbias
    return dict(RMSE=rmse, NSE=nse, KGE=kge, PBIAS=pbias, N=n)

def extract_model_timeseries(paths, river_var, dims_map):
    if len(paths) == 0:
        return None
    try:
        with xr.open_mfdataset(paths, combine="by_coords") as ds:
            if river_var not in ds.variables:
                print(f"ERROR: Variable {river_var} not found in dataset.", file=sys.stderr)
                return None
            da = ds[river_var]
            # isel along spatial dims
            if dims_map:
                da_pt = da.isel(**dims_map)
            else:
                print("ERROR: No spatial index mapping available to extract point time series.", file=sys.stderr)
                return None
            # Ensure time coordinate present
            time_coord = None
            for coord in da_pt.coords:
                if np.issubdtype(da_pt[coord].dtype, np.datetime64):
                    time_coord = coord
                    break
            if time_coord is None:
                # fallback: check "time"
                if "time" in da_pt.coords:
                    time_coord = "time"
            if time_coord is None:
                print("ERROR: No time coordinate found for river discharge.", file=sys.stderr)
                return None
            if time_coord != "time":
                da_pt = da_pt.rename({time_coord: "time"})
            ts = da_pt.to_series()
            ts = ts.sort_index()
            return ts
    except Exception as e:
        print(f"ERROR: Failed to open and extract model data: {e}", file=sys.stderr)
        return None

        # Ensure gauge_id as str without decimals
        df["gauge_id"] = df["gauge_id"].astype(str).str.strip()
        return df

def read_obs_timeseries(gauge_id):
    # Returns daily series of discharge (m3/s) indexed by pandas DatetimeIndex
    obs_path = os.path.join(OBS_STREAMFLOW_DIR, f"{gauge_id}.csv")
    try:
        df = pd.read_csv(obs_path, parse_dates=["date"])
        if "discharge_m3s" not in df.columns:
            # Try alternative name
            for c in df.columns:
                if "discharge" in c.lower() and "m3" in c.lower():
                    df = df.rename(columns={c: "discharge_m3s"})
                    break
        df = df.dropna(subset=["date", "discharge_m3s"])
        # Filter time period
        df = df[(df["date"] >= pd.Timestamp(f"{START_YEAR}-01-01")) & (df["date"] <= pd.Timestamp(f"{END_YEAR}-12-31"))]
        if df.empty:
            return None
        df = df.set_index("date").sort_index()
        obs_series = df["discharge_m3s"].astype(float)
        return obs_series
    except Exception as e:
        print(f"ERROR: Unable to read observation for gauge {gauge_id}: {e}", file=sys.stderr)
        return None

def monthly_mean_series(ts, start_year, end_year):
    # Compute monthly mean robustly for DatetimeIndex or CFTimeIndex
    if ts is None or len(ts) == 0:
        return pd.Series(dtype=float)
    # Convert to pandas Series if needed
    try:
        series = ts.copy()
    except Exception:
        series = pd.Series(ts)

    # Drop NaNs
    series = series[~pd.isnull(series.values)]

    idx = series.index

    years = []
    months = []
    for t in idx:
        # cftime objects have year/month attributes
        try:
            y = int(getattr(t, "year"))
            m = int(getattr(t, "month"))
        except Exception:
            # Fallback to pandas Timestamp
            tt = pd.Timestamp(t)
            y = tt.year
            m = tt.month
        years.append(y)
        months.append(m)

    df = pd.DataFrame({"year": years, "month": months, "value": np.asarray(series.values, dtype=float)})
    grouped = df.groupby(["year", "month"], as_index=True)["value"].mean()
    # Create pandas DatetimeIndex at month start
    times = [pd.Timestamp(year=int(y), month=int(m), day=1) for (y, m) in grouped.index]
    monthly = pd.Series(grouped.values, index=pd.DatetimeIndex(times)).sort_index()
    # Filter to range
    start_ts = pd.Timestamp(f"{start_year}-01-01")
    end_ts = pd.Timestamp(f"{end_year}-12-01")
    monthly = monthly[(monthly.index >= start_ts) & (monthly.index <= end_ts)]
    monthly.name = "monthly_mean"
    return monthly

def align_monthly_series(sim_ts, obs_ts):
    # Compute monthly means and align overlapping period
    if sim_ts is None or obs_ts is None:
        return None, None
    sim_monthly = monthly_mean_series(sim_ts, START_YEAR, END_YEAR)
    obs_monthly = monthly_mean_series(obs_ts, START_YEAR, END_YEAR)
    if sim_monthly.empty or obs_monthly.empty:
        return None, None
    aligned = pd.concat(
        [
            sim_monthly.rename("sim_discharge_m3s"),
            obs_monthly.rename("obs_discharge_m3s")
        ],
        axis=1
    ).dropna()
    return aligned["sim_discharge_m3s"], aligned["obs_discharge_m3s"]

        ax_map.set_title(f"{basin_name} Basin (Gauge {gauge_id})", fontsize=12)

        # Time series
        ax_ts = fig.add_subplot(gs[0, 1])
        ax_ts.plot(sim_ts.index, sim_ts.values, label="E3SM MOSART (monthly)", color="orange", lw=1.5)
        ax_ts.plot(obs_ts.index, obs_ts.values, label="Observed (monthly mean)", color="blue", lw=1.2, alpha=0.8)
        ax_ts.set_xlabel("Time")
        ax_ts.set_ylabel("Discharge (m3/s)")
        ax_ts.grid(True, ls="--", alpha=0.3)
        ax_ts.legend()
        ax_ts.set_title(f"Simulated vs Observed Discharge (1985-1989)", fontsize=11)

        fig.tight_layout()
        fig.savefig(out_png, dpi=150)
        plt.close(fig)
    except Exception as e:
        print(f"ERROR: Failed to create/save figure {out_png}: {e}", file=sys.stderr)

def main():
    ensure_output_dir(OUTPUT_DIR)

    # Load gauge metadata
    meta = read_gauge_metadata(GAUGE_METADATA_PATH)
    if meta is None or meta.empty:
        print("ERROR: Gauge metadata not found or empty.", file=sys.stderr)
        return

    # Load basin polygons
    basin_geojson = load_geojson(BASIN_GEOJSON_PATH)

    # Open sample file to get lat/lon grid and var presence
    ds_sample, sample_path = open_sample_file()
    if ds_sample is None:
        print("ERROR: Could not open any sample MOSART file. Exiting.", file=sys.stderr)
        return

    # Identify lat/lon variable names
    lat_name, lon_name = find_lat_lon_vars(ds_sample)
    if lat_name is None or lon_name is None:
        print("ERROR: Could not identify latitude/longitude variables.", file=sys.stderr)
        try:
            ds_sample.close()
        except Exception:
            pass
        return

    if RIVER_VAR not in ds_sample.variables:
        print(f"ERROR: Variable {RIVER_VAR} not found in sample file {sample_path}.", file=sys.stderr)
        try:
            ds_sample.close()
        except Exception:
            pass
        return

    # File list for full period
    file_list = build_file_list()
    if len(file_list) == 0:
        print("ERROR: No MOSART files found for the specified period.", file=sys.stderr)
        try:
            ds_sample.close()
        except Exception:
            pass
        return

    # Prepare metrics collection
    metrics_rows = []

    # Loop through target basins/gauges
    for basin_name, gauge_id in TARGET_BASINS.items():
        print(f"Processing basin {basin_name} (Gauge {gauge_id}) ...")
        # Find gauge in metadata
        row = meta[meta["gauge_id"].astype(str).str.strip() == str(gauge_id).strip()]
        if row.empty:
            print(f"WARNING: Gauge ID {gauge_id} not found in metadata. Skipping.", file=sys.stderr)
            continue
        gauge_lat = float(row.iloc[0]["lat"])
        gauge_lon = float(row.iloc[0]["lon"])

        # Find nearest grid index on sample dataset
        try:
            dims_map, near_lat, near_lon = nearest_grid_index(ds_sample, lat_name, lon_name, gauge_lat, gauge_lon, RIVER_VAR)
        except Exception as e:
            print(f"ERROR: Failed finding nearest grid for gauge {gauge_id}: {e}", file=sys.stderr)
            continue

        if not dims_map:
            print(f"ERROR: Could not map spatial indices for gauge {gauge_id}. Skipping.", file=sys.stderr)
            continue

        # Extract model time series across all files
        sim_ts = extract_model_timeseries(file_list, RIVER_VAR, dims_map)
        if sim_ts is None or sim_ts.empty:
            print(f"WARNING: No simulated time series extracted for gauge {gauge_id}. Skipping.", file=sys.stderr)
            continue

        # Read observation time series (daily)
        obs_ts = read_obs_timeseries(gauge_id)
        if obs_ts is None or obs_ts.empty:
            print(f"WARNING: No observation data for gauge {gauge_id}. Skipping.", file=sys.stderr)
            continue

        # Align to monthly and overlapping period
        sim_monthly, obs_monthly = align_monthly_series(sim_ts, obs_ts)
        if sim_monthly is None or obs_monthly is None or sim_monthly.empty or obs_monthly.empty:
            print(f"WARNING: Aligned monthly data empty for gauge {gauge_id}. Skipping.", file=sys.stderr)
            continue

        # Compute metrics
        metrics = compute_metrics(sim_monthly.values, obs_monthly.values)

        # Save time series CSV
        ts_df = pd.DataFrame({
            "time": sim_monthly.index,
            "sim_discharge_m3s": sim_monthly.values,
            "obs_discharge_m3s": obs_monthly.values
        })
        ts_csv_path = os.path.join(OUTPUT_DIR, f"timeseries_{gauge_id}_{basin_name}.csv")
        try:
            ts_df.to_csv(ts_csv_path, index=False)
        except Exception as e:
            print(f"ERROR: Failed to save timeseries CSV for gauge {gauge_id}: {e}", file=sys.stderr)

        # Create and save figure
        basin_geom = find_geometry_by_grdc(basin_geojson, gauge_id)
        fig_path = os.path.join(OUTPUT_DIR, f"basin_{gauge_id}_{basin_name}.png")
        make_figure_map_timeseries(
            basin_name=basin_name,
            gauge_id=gauge_id,
            basin_geom=basin_geom,
            gauge_lat=gauge_lat,
            gauge_lon=gauge_lon,
            grid_lat=near_lat,
            grid_lon=near_lon,
            sim_ts=sim_monthly,
            obs_ts=obs_monthly,
            out_png=fig_path
        )

        # Append metrics
        metrics_rows.append({
            "basin_name": basin_name,
            "gauge_id": gauge_id,
            "gauge_lat": gauge_lat,
            "gauge_lon": gauge_lon,
            "grid_lat": near_lat,
            "grid_lon": near_lon,
            "RMSE_m3s": metrics["RMSE"],
            "NSE": metrics["NSE"],
            "KGE": metrics["KGE"],
            "PBIAS_percent": metrics["PBIAS"],
            "n_months": metrics["N"]
        })

    # Close sample dataset
    try:
        ds_sample.close()
    except Exception:
        pass

    # Save metrics summary
    if len(metrics_rows) > 0:
        metrics_df = pd.DataFrame(metrics_rows)
        metrics_csv = os.path.join(OUTPUT_DIR, "metrics_summary.csv")
        try:
            metrics_df.to_csv(metrics_csv, index=False)
            print(f"Saved metrics summary to {metrics_csv}")
        except Exception as e:
            print(f"ERROR: Failed to save metrics summary CSV: {e}", file=sys.stderr)
    else:
        print("No metrics to save.", file=sys.stderr)

if __name__ == "__main__":
    warnings.filterwarnings("ignore")
    main()
