

name: "Streamflow FDC Distributional Comparison"
description: |
  Compare the statistical distribution of E3SM simulated river discharge against
  observations for 1985-1989 using flow duration curve (FDC) metrics. Matches all
  gauges to the MOSART grid, extracts daily simulated and observed discharge,
  computes distributional metrics (volume bias, Wasserstein distance, Q10/Q50/Q90
  ratios), and produces a combined map + FDC panel figure for six selected basins.

settings:
  case_name: "sample.v3.LR.historical"
  data_dir: "./data/sample/e3sm/"
  obs_dir: "./data/sample/obs/streamflow/"
  metadata_file: "./data/sample/obs/gauge_metadata.csv"
  output_dir: "./output/task04_streamflow_fdc"
  years: [1985, 1986, 1987, 1988, 1989]
  selected_gauges: "3629000,4121801,4115200,6742900,2969100,1159100"

steps:
  - id: load_gauge_metadata
    tool: load_obs_metadata
    params:
      metadata_file: "${settings.metadata_file}"
    outputs:
      metadata_file: gauge_metadata_validated.csv

  - id: match_gauges
    tool: match_to_grid
    params:
      gauge_metadata: "${load_gauge_metadata.outputs.metadata_file}"
      data_dir: "${settings.data_dir}"
      case_name: "${settings.case_name}"
      component: "mosart"
    outputs:
      matched_file: matched_gauges.csv

  - id: extract_sim_discharge
    tool: extract_e3sm_timeseries
    params:
      data_dir: "${settings.data_dir}"
      case_name: "${settings.case_name}"
      component: "mosart"
      variable: "RIVER_DISCHARGE_OVER_LAND_LIQ"
      locations_file: "${match_gauges.outputs.matched_file}"
      years: "${settings.years}"
      frequency: "daily"
    outputs:
      timeseries_file: sim_discharge_daily_1985_1989.csv

  - id: extract_obs_discharge
    tool: extract_obs_timeseries
    params:
      obs_dir: "${settings.obs_dir}"
      gauge_metadata: "${load_gauge_metadata.outputs.metadata_file}"
      years: "${settings.years}"
    outputs:
      timeseries_file: obs_discharge_daily_1985_1989.csv

  - id: compute_fdc
    tool: compute_fdc_metrics
    params:
      sim_file: "${extract_sim_discharge.outputs.timeseries_file}"
      obs_file: "${extract_obs_discharge.outputs.timeseries_file}"
    outputs:
      metrics_file: fdc_metrics.csv
      fdc_file: fdc_percentiles.csv

  - id: plot_fdc_comparison
    tool: plot_fdc
    params:
      metrics_file: "${compute_fdc.outputs.metrics_file}"
      fdc_file: "${compute_fdc.outputs.fdc_file}"
      locations_file: "${match_gauges.outputs.matched_file}"
      metric: "wasserstein"
      gauge_ids: "${settings.selected_gauges}"
    outputs:
      plot_file: fdc_wasserstein_map_and_panels.png
