name: "Integrated Basin Water Cycle Diagnostic"
description: |
  Per-basin integrated water cycle evaluation for E3SM over 1985-1989 across six
  major river basins (Amazon, Missouri, Columbia, Danube, Mekong, Orange).
  Diagnoses model biases in precipitation, ET, runoff, water balance, and
  streamflow using FDC metrics, basin budget tables, bar charts, and radar plots.

settings:
  case_name: "sample.v3.LR.historical"
  data_dir: "./data/sample/e3sm/"
  output_dir: "./output/task07_integrated_diagnostic"
  basins_file: "./data/sample/obs/basin_polygons.geojson"
  gauge_metadata: "./data/sample/obs/gauge_metadata.csv"
  obs_streamflow_dir: "./data/sample/obs/streamflow/"
  gauge_ids: "3629000,4121801,4115200,6742900,2969100,1159100"
  years:
    - 1985
    - 1986
    - 1987
    - 1988
    - 1989

steps:
  # ── Part 1: Extract model fields ──────────────────────────────────────────
  - id: extract_model_precip
    tool: extract_gridded_field
    params:
      data_dir: "${settings.data_dir}"
      case_name: "${settings.case_name}"
      component: "elm"
      variable: "RAIN+SNOW"
      years: "${settings.years}"
    outputs:
      field_file: "${settings.output_dir}/model_precip_field.nc"

  - id: extract_model_et
    tool: extract_gridded_field
    params:
      data_dir: "${settings.data_dir}"
      case_name: "${settings.case_name}"
      component: "elm"
      variable: "QVEGE+QVEGT+QSOIL"
      years: "${settings.years}"
    outputs:
      field_file: "${settings.output_dir}/model_et_field.nc"

  - id: extract_model_runoff
    tool: extract_gridded_field
    params:
      data_dir: "${settings.data_dir}"
      case_name: "${settings.case_name}"
      component: "elm"
      variable: "QRUNOFF"
      years: "${settings.years}"
    outputs:
      field_file: "${settings.output_dir}/model_runoff_field.nc"

  # ── Part 2: Fetch and extract observation fields ──────────────────────────
  - id: fetch_obs_precip
    tool: fetch_ilamb_data
    params:
      variable: "pr"
      dataset: "GPCCv2018"
    outputs:
      data_file: "${settings.output_dir}/obs_raw_precip.nc"

  - id: fetch_obs_et
    tool: fetch_ilamb_data
    params:
      variable: "evspsbl"
      dataset: "MODIS"
    outputs:
      data_file: "${settings.output_dir}/obs_raw_et.nc"

  - id: fetch_obs_runoff
    tool: fetch_ilamb_data
    params:
      variable: "mrro"
      dataset: "LORA"
    outputs:
      data_file: "${settings.output_dir}/obs_raw_runoff.nc"

  - id: extract_obs_precip
    tool: extract_gridded_field
    params:
      obs_file: "${fetch_obs_precip.outputs.data_file}"
      variable: "pr"
      years: "${settings.years}"
    outputs:
      field_file: "${settings.output_dir}/obs_precip_field.nc"

  - id: extract_obs_et
    tool: extract_gridded_field
    params:
      obs_file: "${fetch_obs_et.outputs.data_file}"
      variable: "et"
      years: "${settings.years}"
    outputs:
      field_file: "${settings.output_dir}/obs_et_field.nc"

  - id: extract_obs_runoff
    tool: extract_gridded_field
    params:
      obs_file: "${fetch_obs_runoff.outputs.data_file}"
      variable: "mrro"
      years: "${settings.years}"
    outputs:
      field_file: "${settings.output_dir}/obs_runoff_field.nc"

  # ── Part 3: Clip to basin means ──────────────────────────────────────────
  - id: basin_model_precip
    tool: extract_basin_mean
    params:
      field_file: "${extract_model_precip.outputs.field_file}"
      basins_file: "${settings.basins_file}"
      gauge_ids: "${settings.gauge_ids}"
    outputs:
      basin_means_file: "${settings.output_dir}/basin_model_precip.csv"

  - id: basin_model_et
    tool: extract_basin_mean
    params:
      field_file: "${extract_model_et.outputs.field_file}"
      basins_file: "${settings.basins_file}"
      gauge_ids: "${settings.gauge_ids}"
    outputs:
      basin_means_file: "${settings.output_dir}/basin_model_et.csv"

  - id: basin_model_runoff
    tool: extract_basin_mean
    params:
      field_file: "${extract_model_runoff.outputs.field_file}"
      basins_file: "${settings.basins_file}"
      gauge_ids: "${settings.gauge_ids}"
    outputs:
      basin_means_file: "${settings.output_dir}/basin_model_runoff.csv"

  - id: basin_obs_precip
    tool: extract_basin_mean
    params:
      field_file: "${extract_obs_precip.outputs.field_file}"
      basins_file: "${settings.basins_file}"
      gauge_ids: "${settings.gauge_ids}"
    outputs:
      basin_means_file: "${settings.output_dir}/basin_obs_precip.csv"

  - id: basin_obs_et
    tool: extract_basin_mean
    params:
      field_file: "${extract_obs_et.outputs.field_file}"
      basins_file: "${settings.basins_file}"
      gauge_ids: "${settings.gauge_ids}"
    outputs:
      basin_means_file: "${settings.output_dir}/basin_obs_et.csv"

  - id: basin_obs_runoff
    tool: extract_basin_mean
    params:
      field_file: "${extract_obs_runoff.outputs.field_file}"
      basins_file: "${settings.basins_file}"
      gauge_ids: "${settings.gauge_ids}"
    outputs:
      basin_means_file: "${settings.output_dir}/basin_obs_runoff.csv"

  # ── Part 4: Streamflow FDC metrics ───────────────────────────────────────
  - id: load_metadata
    tool: load_obs_metadata
    params:
      metadata_file: "${settings.gauge_metadata}"
    outputs:
      metadata_file: "${settings.output_dir}/validated_gauge_metadata.csv"
      n_gauges: n_gauges

  - id: match_gauges
    tool: match_to_grid
    params:
      gauge_metadata: "${load_metadata.outputs.metadata_file}"
      data_dir: "${settings.data_dir}"
      case_name: "${settings.case_name}"
      component: "mosart"
    outputs:
      matched_file: "${settings.output_dir}/matched_gauges.csv"
      n_matched: n_matched

  - id: extract_sim_discharge
    tool: extract_e3sm_timeseries
    params:
      data_dir: "${settings.data_dir}"
      case_name: "${settings.case_name}"
      component: "mosart"
      variable: "RIVER_DISCHARGE_OVER_LAND_LIQ"
      locations_file: "${match_gauges.outputs.matched_file}"
      years: "${settings.years}"
      frequency: "monthly"
    outputs:
      timeseries_file: "${settings.output_dir}/sim_discharge.csv"
      n_locations: n_locations

  - id: extract_obs_discharge
    tool: extract_obs_timeseries
    params:
      obs_dir: "${settings.obs_streamflow_dir}"
      gauge_metadata: "${load_metadata.outputs.metadata_file}"
      years: "${settings.years}"
      gauge_ids: "${settings.gauge_ids}"
    outputs:
      timeseries_file: "${settings.output_dir}/obs_discharge.csv"
      n_gauges: n_obs_gauges

  - id: compute_fdc
    tool: compute_fdc_metrics
    params:
      sim_file: "${extract_sim_discharge.outputs.timeseries_file}"
      obs_file: "${extract_obs_discharge.outputs.timeseries_file}"
    outputs:
      metrics_file: "${settings.output_dir}/fdc_metrics.csv"
      fdc_file: "${settings.output_dir}/fdc_percentiles.csv"

  # ── Part 5: Combine and visualise ────────────────────────────────────────
  - id: compute_budget
    tool: compute_basin_budget
    params:
      model_precip_file: "${basin_model_precip.outputs.basin_means_file}"
      model_et_file: "${basin_model_et.outputs.basin_means_file}"
      model_runoff_file: "${basin_model_runoff.outputs.basin_means_file}"
      obs_precip_file: "${basin_obs_precip.outputs.basin_means_file}"
      obs_et_file: "${basin_obs_et.outputs.basin_means_file}"
      obs_runoff_file: "${basin_obs_runoff.outputs.basin_means_file}"
      fdc_metrics_file: "${compute_fdc.outputs.metrics_file}"
      locations_file: "${load_metadata.outputs.metadata_file}"
    outputs:
      budget_file: "${settings.output_dir}/basin_budget_summary.csv"

  - id: plot_budget_comparison
    tool: plot_basin_budget_comparison
    params:
      budget_file: "${compute_budget.outputs.budget_file}"
    outputs:
      plot_file: "${settings.output_dir}/basin_budget_comparison.png"

  - id: plot_radar
    tool: plot_basin_radar
    params:
      budget_file: "${compute_budget.outputs.budget_file}"
    outputs:
      plot_file: "${settings.output_dir}/basin_radar_diagnostic.png"
