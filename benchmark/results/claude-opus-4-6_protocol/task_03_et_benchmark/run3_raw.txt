
name: "ET Benchmark: E3SM ELM vs MODIS (1985-1989)"
description: |
  Benchmark E3SM land model evapotranspiration against MODIS satellite observations.
  Computes a 5-year climatological mean (1985-1989) for both model (QVEGE+QVEGT+QSOIL)
  and MODIS ET, then evaluates the spatial bias, global mean bias, RMSE, and spatial
  correlation. Produces a 3-panel comparison map (obs, model, bias).

settings:
  case_name: "sample.v3.LR.historical"
  data_dir: "./data/sample/e3sm/"
  output_dir: "./output/task03_et_benchmark"
  years:
    - 1985
    - 1986
    - 1987
    - 1988
    - 1989

steps:
  - id: fetch_modis_et
    tool: fetch_ilamb_data
    params:
      variable: "evspsbl"
      dataset: "MODIS"
    outputs:
      data_file: modis_et_raw.nc

  - id: extract_model_et
    tool: extract_gridded_field
    params:
      data_dir: "${settings.data_dir}"
      case_name: "${settings.case_name}"
      component: "elm"
      variable: "QVEGE+QVEGT+QSOIL"
      years: ${settings.years}
    outputs:
      field_file: model_et_field.nc

  - id: extract_obs_et
    tool: extract_gridded_field
    params:
      obs_file: "${fetch_modis_et.outputs.data_file}"
      variable: "et"
      years: ${settings.years}
    outputs:
      field_file: obs_et_field.nc

  - id: compute_et_bias
    tool: compute_spatial_bias
    params:
      field_a: "${extract_model_et.outputs.field_file}"
      field_b: "${extract_obs_et.outputs.field_file}"
    outputs:
      bias_file: et_bias_field.nc
      stats_file: et_bias_stats.csv

  - id: compute_global_stats
    tool: compute_zonal_stats
    params:
      field_file: "${compute_et_bias.outputs.bias_file}"
    outputs:
      stats_file: et_global_zonal_stats.csv

  - id: plot_bias_map
    tool: plot_gridded_map
    params:
      field_file: "${compute_et_bias.outputs.bias_file}"
      title: "ET Bias (E3SM ELM − MODIS) 1985-1989"
      units: "mm/day"
      stats_file: "${compute_et_bias.outputs.stats_file}"
    outputs:
      plot_file: et_bias_map.png

  - id: plot_three_panel
    tool: plot_bias_comparison
    params:
      obs_file: "${extract_obs_et.outputs.field_file}"
      sim_file: "${extract_model_et.outputs.field_file}"
      bias_file: "${compute_et_bias.outputs.bias_file}"
      obs_label: "MODIS ET"
      sim_label: "E3SM ELM ET (QVEGE+QVEGT+QSOIL)"
      bias_label: "Bias (E3SM − MODIS)"
      time_range: "1985-1989"
      units: "mm/day"
      stats_file: "${compute_et_bias.outputs.stats_file}"
    outputs:
      plot_file: et_bias_comparison_3panel.png
