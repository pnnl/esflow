

name: "Irrigation Water Demand Analysis"
description: |
  Extract ELM irrigation water demand (QIRRIG) for 1985-1989,
  compute global mean zonal statistics, and produce a global map
  of the time-averaged field.

settings:
  data_dir: "./data/sample/model"
  case_name: "*"
  component: "elm"
  variable: "QIRRIG"
  years: [1985, 1986, 1987, 1988, 1989]
  output_dir: "./output/test_irrigation_demand"

steps:
  - id: extract_irrig_field
    tool: extract_gridded_field
    params:
      data_dir: "${settings.data_dir}"
      case_name: "${settings.case_name}"
      component: "${settings.component}"
      variable: "${settings.variable}"
      years: ${settings.years}
    outputs:
      field_file: "${settings.output_dir}/qirrig_mean_1985_1989.nc"

  - id: compute_global_mean
    tool: compute_zonal_stats
    params:
      field_file: "${extract_irrig_field.outputs.field_file}"
      variable: "${settings.variable}"
      lat_bands: "global:-90:90,tropics:-30:30,northern_midlat:30:60,southern_midlat:-60:-30,arctic:60:90,antarctic:-90:-60"
    outputs:
      stats_file: "${settings.output_dir}/qirrig_global_mean_climatology.csv"

  - id: plot_irrig_map
    tool: plot_gridded_map
    params:
      field_file: "${extract_irrig_field.outputs.field_file}"
      variable: "${settings.variable}"
      title: "Mean Irrigation Water Demand (QIRRIG) 1985–1989"
      units: "mm/s"
      stats_file: "${compute_global_mean.outputs.stats_file}"
    outputs:
      plot_file: "${settings.output_dir}/qirrig_mean_1985_1989_map.png"
