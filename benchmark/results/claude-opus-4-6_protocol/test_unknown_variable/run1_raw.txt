
name: "Irrigation Water Withdrawal Analysis"
description: |
  Extract ELM irrigation water withdrawal (QIRRIG) for 1985-1989,
  compute global mean statistics, and produce a global map of the
  mean irrigation withdrawal rate.

settings:
  data_dir: "./data/sample/model"
  case_name: "sample"
  output_dir: "./output/test_irrigation"

steps:
  - id: extract_qirrig_field
    tool: extract_gridded_field
    params:
      data_dir: "${settings.data_dir}"
      case_name: "${settings.case_name}"
      component: "elm"
      variable: "QIRRIG"
      years: [1985, 1986, 1987, 1988, 1989]
    outputs:
      field_file: "${settings.output_dir}/qirrig_mean_1985_1989.nc"

  - id: compute_global_mean
    tool: compute_zonal_stats
    params:
      field_file: "${extract_qirrig_field.outputs.field_file}"
      variable: "QIRRIG"
      lat_bands: "global:-90:90,tropics:-30:30,midlat_north:30:60,midlat_south:-60:-30,highlat_north:60:90"
    outputs:
      stats_file: "${settings.output_dir}/qirrig_global_mean_stats.csv"

  - id: plot_irrigation_map
    tool: plot_gridded_map
    params:
      field_file: "${extract_qirrig_field.outputs.field_file}"
      variable: "QIRRIG"
      title: "Mean Irrigation Water Withdrawal (QIRRIG) 1985-1989"
      units: "mm/s"
      stats_file: "${compute_global_mean.outputs.stats_file}"
    outputs:
      plot_file: "${settings.output_dir}/qirrig_map_1985_1989.png"
