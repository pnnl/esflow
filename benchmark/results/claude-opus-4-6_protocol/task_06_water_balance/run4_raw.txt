
name: "ELM Water Balance Closure Check"
description: |
  Check E3SM ELM water balance closure (P - ET - Q ≈ 0) over 1985-1989.
  Extract time-mean fields for precipitation (RAIN+SNOW), evapotranspiration
  (QVEGE+QVEGT+QSOIL), and total runoff (QRUNOFF). Compute global area-weighted
  means and spatial residual. Clip to six major basins and produce a composite
  water balance figure.

settings:
  data_dir: "./data/sample/e3sm/"
  case_name: "sample.v3.LR.historical"
  component: "elm"
  output_dir: "./output/task06_water_balance"
  basins_file: "./data/sample/obs/basin_polygons.geojson"
  gauge_metadata: "./data/sample/obs/gauge_metadata.csv"
  gauge_ids: "3629000,4121801,4115200,6742900,2969100,1159100"
  years:
    - 1985
    - 1986
    - 1987
    - 1988
    - 1989

steps:
  - id: extract_precip_field
    tool: extract_gridded_field
    params:
      data_dir: "${settings.data_dir}"
      case_name: "${settings.case_name}"
      component: "${settings.component}"
      variable: "RAIN+SNOW"
      years: "${settings.years}"
    outputs:
      field_file: "${settings.output_dir}/field_precip.nc"

  - id: extract_et_field
    tool: extract_gridded_field
    params:
      data_dir: "${settings.data_dir}"
      case_name: "${settings.case_name}"
      component: "${settings.component}"
      variable: "QVEGE+QVEGT+QSOIL"
      years: "${settings.years}"
    outputs:
      field_file: "${settings.output_dir}/field_et.nc"

  - id: extract_runoff_field
    tool: extract_gridded_field
    params:
      data_dir: "${settings.data_dir}"
      case_name: "${settings.case_name}"
      component: "${settings.component}"
      variable: "QRUNOFF"
      years: "${settings.years}"
    outputs:
      field_file: "${settings.output_dir}/field_runoff.nc"

  - id: zonal_stats_precip
    tool: compute_zonal_stats
    params:
      field_file: "${extract_precip_field.outputs.field_file}"
    outputs:
      stats_file: "${settings.output_dir}/zonal_stats_precip.csv"

  - id: zonal_stats_et
    tool: compute_zonal_stats
    params:
      field_file: "${extract_et_field.outputs.field_file}"
    outputs:
      stats_file: "${settings.output_dir}/zonal_stats_et.csv"

  - id: zonal_stats_runoff
    tool: compute_zonal_stats
    params:
      field_file: "${extract_runoff_field.outputs.field_file}"
    outputs:
      stats_file: "${settings.output_dir}/zonal_stats_runoff.csv"

  - id: compute_p_minus_et
    tool: compute_spatial_bias
    params:
      field_a: "${extract_precip_field.outputs.field_file}"
      field_b: "${extract_et_field.outputs.field_file}"
    outputs:
      bias_file: "${settings.output_dir}/field_p_minus_et.nc"
      stats_file: "${settings.output_dir}/stats_p_minus_et.csv"

  - id: compute_residual
    tool: compute_spatial_bias
    params:
      field_a: "${compute_p_minus_et.outputs.bias_file}"
      field_b: "${extract_runoff_field.outputs.field_file}"
    outputs:
      bias_file: "${settings.output_dir}/field_residual.nc"
      stats_file: "${settings.output_dir}/stats_residual.csv"

  - id: zonal_stats_residual
    tool: compute_zonal_stats
    params:
      field_file: "${compute_residual.outputs.bias_file}"
    outputs:
      stats_file: "${settings.output_dir}/zonal_stats_residual.csv"

  - id: basin_mean_precip
    tool: extract_basin_mean
    params:
      field_file: "${extract_precip_field.outputs.field_file}"
      basins_file: "${settings.basins_file}"
      gauge_ids: "${settings.gauge_ids}"
    outputs:
      basin_means_file: "${settings.output_dir}/basin_mean_precip.csv"

  - id: basin_mean_et
    tool: extract_basin_mean
    params:
      field_file: "${extract_et_field.outputs.field_file}"
      basins_file: "${settings.basins_file}"
      gauge_ids: "${settings.gauge_ids}"
    outputs:
      basin_means_file: "${settings.output_dir}/basin_mean_et.csv"

  - id: basin_mean_runoff
    tool: extract_basin_mean
    params:
      field_file: "${extract_runoff_field.outputs.field_file}"
      basins_file: "${settings.basins_file}"
      gauge_ids: "${settings.gauge_ids}"
    outputs:
      basin_means_file: "${settings.output_dir}/basin_mean_runoff.csv"

  - id: basin_mean_residual
    tool: extract_basin_mean
    params:
      field_file: "${compute_residual.outputs.bias_file}"
      basins_file: "${settings.basins_file}"
      gauge_ids: "${settings.gauge_ids}"
    outputs:
      basin_means_file: "${settings.output_dir}/basin_mean_residual.csv"

  - id: plot_residual_map
    tool: plot_gridded_map
    params:
      field_file: "${compute_residual.outputs.bias_file}"
      title: "ELM Water Balance Residual (P - ET - Q) 1985-1989"
      units: "mm/s"
      stats_file: "${zonal_stats_residual.outputs.stats_file}"
    outputs:
      plot_file: "${settings.output_dir}/map_residual.png"

  - id: plot_water_balance_composite
    tool: plot_water_balance_basins
    params:
      residual_file: "${compute_residual.outputs.bias_file}"
      basin_precip_file: "${basin_mean_precip.outputs.basin_means_file}"
      basin_et_file: "${basin_mean_et.outputs.basin_means_file}"
      basin_runoff_file: "${basin_mean_runoff.outputs.basin_means_file}"
      basin_residual_file: "${basin_mean_residual.outputs.basin_means_file}"
      basins_file: "${settings.basins_file}"
      gauge_ids: "${settings.gauge_ids}"
      locations_file: "${settings.gauge_metadata}"
    outputs:
      plot_file: "${settings.output_dir}/water_balance_basins.png"
