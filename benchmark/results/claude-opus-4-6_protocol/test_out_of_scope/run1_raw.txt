

name: "ENSO 3.4 Index Analysis"
description: |
  Approximate ENSO 3.4 region analysis from monthly SST data for 1985-1989.
  The available ESMFlow tool catalog supports gridded field extraction and
  area-weighted zonal statistics but does not include tools for monthly
  regional time-series extraction, anomaly (climatology removal) computation,
  or running-mean smoothing on gridded ocean data.  This workflow therefore
  provides the closest feasible approximation: it extracts the time-mean SST
  field over 1985-1989, computes area-weighted statistics for the Niño 3.4
  latitude band (5°S–5°N), and produces a global SST map with the regional
  statistics overlaid.

settings:
  sst_file: "./data/sample/model/sst_monthly.nc"
  output_dir: "./output/test_enso"
  variable: "tos"
  years:
    - 1985
    - 1986
    - 1987
    - 1988
    - 1989

steps:
  - id: extract_sst_mean
    tool: extract_gridded_field
    params:
      obs_file: "${settings.sst_file}"
      variable: "${settings.variable}"
      years: "${settings.years}"
    outputs:
      field_file: "${settings.output_dir}/sst_mean_1985_1989.nc"

  - id: nino34_zonal_stats
    tool: compute_zonal_stats
    params:
      field_file: "${extract_sst_mean.outputs.field_file}"
      variable: "${settings.variable}"
      lat_bands: "nino34:-5:5"
    outputs:
      stats_file: "${settings.output_dir}/nino34_zonal_stats.csv"

  - id: plot_sst_map
    tool: plot_gridded_map
    params:
      field_file: "${extract_sst_mean.outputs.field_file}"
      variable: "${settings.variable}"
      title: "Mean SST 1985-1989 — Niño 3.4 Band (5°S–5°N)"
      units: "K"
      stats_file: "${nino34_zonal_stats.outputs.stats_file}"
    outputs:
      plot_file: "${settings.output_dir}/enso34_sst_map.png"
