import os
import glob
import numpy as np
import xarray as xr
import pandas as pd
import matplotlib.pyplot as plt
import cartopy.crs as ccrs
import geopandas as gpd
from matplotlib.path import Path

# Input/output paths and settings
data_dir = "./data/sample/e3sm"
case_name = "sample.v3.LR.historical"
basin_geojson = "./data/sample/obs/basin_polygons.geojson"
output_dir = "/Users/zhou014/Local_Drive/Git_repo/esmflow/benchmark/results/o4-mini_baseline/task_06_water_balance/run3_output"

# Basin codes and names
basins = {
    3629000: "Amazon",
    4121801: "Missouri",
    4115200: "Columbia",
    6742900: "Danube",
    2969100: "Mekong",
    1159100: "Orange"
}

# Create output directory
os.makedirs(output_dir, exist_ok=True)

# 1. Load ELM monthly output and compute climatologies
try:
    elm_files = sorted(glob.glob(os.path.join(data_dir, "lnd", f"{case_name}.elm.h0.*.nc")))
    ds = xr.open_mfdataset(elm_files, combine='by_coords')
    ds = ds.sel(time=slice("1985-01-01", "1989-12-31"))
    ds = ds.load()
    # Compute climatological time mean (convert kg/m2/s -> mm/s by density=1, then to mm/day)
    P = (ds["RAIN"] + ds["SNOW"]).mean("time") * 86400.0
    ET = (ds["QVEGE"] + ds["QVEGT"] + ds["QSOIL"]).mean("time") * 86400.0
    Q = ds["QRUNOFF"].mean("time") * 86400.0
    Residual = P - ET - Q
except Exception as e:
    print("Error loading or processing ELM data:", e)
    raise

# 2. Compute area weights and global means
try:
    lat = ds["lat"]
    lon = ds["lon"]
    weights_lat = np.cos(np.deg2rad(lat))
    weights_1d = xr.DataArray(weights_lat, coords={"lat": lat}, dims=("lat",))
    weights2d = weights_1d.broadcast_like(P)
    def global_weighted_mean(field):
        num = (field * weights2d).sum(dim=("lat", "lon"))
        den = weights2d.sum(dim=("lat", "lon"))
        return (num / den).values.item()
    global_P = global_weighted_mean(P)
    global_ET = global_weighted_mean(ET)
    global_Q = global_weighted_mean(Q)
    global_Res = global_weighted_mean(Residual)
    gm_df = pd.DataFrame({
        "component": ["P", "ET", "Q", "Residual"],
        "mean_mm_per_day": [global_P, global_ET, global_Q, global_Res]
    })
    gm_df.to_csv(os.path.join(output_dir, "global_means.csv"), index=False)
except Exception as e:
    print("Error computing global means:", e)
    raise

# 3. Save climatologies and residual to NetCDF
try:
    ds_out = xr.Dataset({
        "P": P,
        "ET": ET,
        "Q": Q,
        "Residual": Residual
    })
    ds_out.to_netcdf(os.path.join(output_dir, "climatology_PETQ_residual.nc"))
except Exception as e:
    print("Error saving NetCDF:", e)

# 4. Load basin polygons and compute per-basin means
try:
    basin_gdf = gpd.read_file(basin_geojson)
    lon2d, lat2d = np.meshgrid(lon.values, lat.values)
    pts = np.vstack((lon2d.flatten(), lat2d.flatten())).T
    basin_stats = []
    for code, name in basins.items():
        sub = basin_gdf[basin_gdf["grdc_no"] == code]
        if sub.empty:
            continue
        poly = sub.geometry.unary_union
        path = Path(np.array(poly.exterior.coords))
        mask_flat = path.contains_points(pts)
        mask = mask_flat.reshape(lat2d.shape)
        mask_da = xr.DataArray(mask, dims=("lat", "lon"), coords={"lat": lat, "lon": lon})
        def basin_mean(field):
            num = (field * weights2d * mask_da).sum(dim=("lat", "lon"))
            den = (weights2d * mask_da).sum(dim=("lat", "lon"))
            return (num / den).values.item()
        bp = basin_mean(P)
        bet = basin_mean(ET)
        bq = basin_mean(Q)
        br = basin_mean(Residual)
        basin_stats.append({
            "basin": name,
            "P_mm_per_day": bp,
            "ET_mm_per_day": bet,
            "Q_mm_per_day": bq,
            "Residual_mm_per_day": br
        })
    basin_df = pd.DataFrame(basin_stats)
    basin_df.to_csv(os.path.join(output_dir, "basin_means.csv"), index=False)
except Exception as e:
    print("Error computing basin means:", e)
    raise

# 5. Plot composite figure
try:
    fig = plt.figure(figsize=(18, 10))
    import matplotlib.gridspec as gridspec
    gs = gridspec.GridSpec(2, 3, height_ratios=[2, 1], hspace=0.3, wspace=0.2)
    ax_map = fig.add_subplot(gs[0, :], projection=ccrs.PlateCarree())
    vmin = -np.max(np.abs(Residual.values))
    vmax = np.max(np.abs(Residual.values))
    pcm = ax_map.pcolormesh(lon, lat, Residual, cmap="RdBu_r", vmin=vmin, vmax=vmax, transform=ccrs.PlateCarree())
    ax_map.coastlines()
    for code, name in basins.items():
        sub = basin_gdf[basin_gdf["grdc_no"] == code]
        for geom in sub.geometry:
            ax_map.add_geometries([geom], ccrs.PlateCarree(), edgecolor="black", facecolor="none", linewidth=1)
    cb = fig.colorbar(pcm, ax=ax_map, orientation="horizontal", pad=0.05, shrink=0.8)
    cb.set_label("Residual (mm/day)")
    ax_map.set_title("Global Water Balance Residual (P - ET - Q)")
    for i, row in basin_df.iterrows():
        ax = fig.add_subplot(gs[1, i])
        vals = [row["P_mm_per_day"], row["ET_mm_per_day"], row["Q_mm_per_day"], row["Residual_mm_per_day"]]
        ax.bar(["P", "ET", "Q", "Res"], vals, color=["blue", "green", "orange", "red"])
        ax.set_title(row["basin"])
        ax.set_ylabel("mm/day")
        ax.grid(True, linestyle="--", alpha=0.5)
    plt.tight_layout()
    fig.savefig(os.path.join(output_dir, "water_balance_composite.png"), dpi=150)
    plt.close(fig)
except Exception as e:
    print("Error creating or saving plot:", e)
    raise
