import os
import glob
import json
import numpy as np
import pandas as pd
import xarray as xr
import matplotlib.pyplot as plt
import cartopy.crs as ccrs
import cartopy.feature as cfeature
from shapely.geometry import shape
from scipy.stats import pearsonr

def compute_rmse(sim, obs):
    return np.sqrt(np.nanmean((sim - obs) ** 2))

def compute_nse(sim, obs):
    denom = np.nansum((obs - np.nanmean(obs)) ** 2)
    num = np.nansum((obs - sim) ** 2)
    return 1.0 - num / denom if denom != 0 else np.nan

def compute_kge(sim, obs):
    valid = ~np.isnan(sim) & ~np.isnan(obs)
    if np.sum(valid) < 2:
        return np.nan
    r, _ = pearsonr(obs[valid], sim[valid])
    alpha = np.nanstd(sim) / np.nanstd(obs) if np.nanstd(obs) != 0 else np.nan
    beta = np.nansum(sim) / np.nansum(obs) if np.nansum(obs) != 0 else np.nan
    return 1.0 - np.sqrt((r - 1) ** 2 + (alpha - 1) ** 2 + (beta - 1) ** 2)

def compute_pbias(sim, obs):
    denom = np.nansum(obs)
    num = np.nansum(sim - obs)
    return 100.0 * num / denom if denom != 0 else np.nan

if __name__ == "__main__":
    case_name = "sample.v3.LR.historical"
    sim_dir = "./data/sample/e3sm/rof/"
    obs_dir = "./data/sample/obs/streamflow/"
    gauge_meta_file = "./data/sample/obs/gauge_metadata.csv"
    basin_geojson = "./data/sample/obs/basin_polygons.geojson"
    output_dir = "/Users/zhou014/Local_Drive/Git_repo/esmflow/benchmark/results/o4-mini_baseline/task_05_basin_streamflow/run4_output"
    os.makedirs(output_dir, exist_ok=True)

    basins = {
        3629000: "Amazon",
        4121801: "Missouri",
        4115200: "Columbia",
        6742900: "Danube",
        2969100: "Mekong",
        1159100: "Orange"
    }

    try:
        gdf_meta = pd.read_csv(gauge_meta_file, dtype={'gauge_id': str})
    except Exception as e:
        print(f"Error reading gauge metadata: {e}")
        gdf_meta = pd.DataFrame(columns=['gauge_id', 'lat', 'lon', 'area_km2', 'river_name'])
    gdf_meta['gauge_id'] = gdf_meta['gauge_id'].astype(int)
    sub_meta = gdf_meta[gdf_meta['gauge_id'].isin(basins.keys())].copy()
    sub_meta['gauge_id'] = sub_meta['gauge_id'].astype(str)

    basin_shapes = {}
    try:
        with open(basin_geojson) as f:
            gj = json.load(f)
        for feat in gj['features']:
            bid = int(feat['properties'].get('grdc_no', -1))
            if bid in basins:
                basin_shapes[bid] = shape(feat['geometry'])
    except Exception as e:
        print(f"Error reading basin polygons: {e}")

    sim_files = sorted(glob.glob(os.path.join(sim_dir, f"{case_name}.mosart.h0.*.nc")))
    ds = None
    try:
        ds = xr.open_mfdataset(sim_files, combine='by_coords', decode_times=True)
        ds = ds.sel(time=slice("1985-01-01", "1989-12-31"))
        sim_var = ds['RIVER_DISCHARGE_OVER_LAND_LIQ']
        lat_arr = ds['lat'].values
        lon_arr = ds['lon'].values
        sim_time = pd.to_datetime([str(t) for t in ds['time'].values])
    except Exception as e:
        print(f"Error loading simulation data: {e}")

    metrics_list = []

    if ds is not None:
        river_dim = sim_var.dims[1]
        for bid, bname in basins.items():
            gid = str(bid)
            row = sub_meta[sub_meta['gauge_id'] == gid]
            if row.empty:
                print(f"No metadata for basin {bname} ({bid})")
                continue
            glat = float(row.lat.values[0])
            glon = float(row.lon.values[0])
            try:
                dist2 = (lat_arr - glat)**2 + (lon_arr - glon)**2
                idx = int(np.nanargmin(dist2))
            except Exception as e:
                print(f"Error finding nearest cell for {bname}: {e}")
                continue

            try:
                sim_series = sim_var.isel({river_dim: idx}).values
                sim_pd = pd.Series(sim_series, index=sim_time)
            except Exception as e:
                print(f"Error extracting simulation series for {bname}: {e}")
                continue

            obs_file = os.path.join(obs_dir, f"{gid}.csv")
            try:
                df_obs = pd.read_csv(obs_file, parse_dates=['date'])
                df_obs = df_obs.set_index('date').sort_index()
                df_obs = df_obs['1985-01-01':'1989-12-31']
                obs_monthly = df_obs['discharge_m3s'].resample('MS').mean()
            except Exception as e:
                print(f"Error loading obs data for {bname}: {e}")
                continue

            df_cmp = pd.DataFrame({
                'sim': sim_pd,
                'obs': obs_monthly
            }).dropna()
            if df_cmp.empty:
                print(f"No overlapping data for {bname}")
                continue

            sim_arr = df_cmp['sim'].values
            obs_arr = df_cmp['obs'].values

            rmse = compute_rmse(sim_arr, obs_arr)
            nse = compute_nse(sim_arr, obs_arr)
            kge = compute_kge(sim_arr, obs_arr)
            pbias = compute_pbias(sim_arr, obs_arr)
            metrics_list.append({
                'basin_id': bid,
                'basin_name': bname,
                'RMSE_m3s': rmse,
                'NSE': nse,
                'KGE': kge,
                'PBIAS_%': pbias
            })

            try:
                fig = plt.figure(figsize=(14,6))
                ax_map = fig.add_subplot(1,2,1, projection=ccrs.PlateCarree())
                ax_map.add_feature(cfeature.LAND)
                ax_map.add_feature(cfeature.OCEAN)
                ax_map.add_feature(cfeature.COASTLINE)
                ax_map.add_feature(cfeature.BORDERS, linestyle=':')
                if bid in basin_shapes:
                    poly = basin_shapes[bid]
                    ax_map.add_geometries([poly], crs=ccrs.PlateCarree(),
                                          facecolor='none', edgecolor='red', linewidth=2)
                    minx, miny, maxx, maxy = poly.bounds
                    ax_map.set_extent([minx-2, maxx+2, miny-2, maxy+2])
                ax_map.plot(glon, glat, marker='o', color='blue', markersize=6, transform=ccrs.PlateCarree())
                ax_map.set_title(f"{bname} Basin")

                ax_ts = fig.add_subplot(1,2,2)
                df_cmp['obs'].plot(ax=ax_ts, label='Observed', color='black')
                df_cmp['sim'].plot(ax=ax_ts, label='Simulated', color='blue')
                ax_ts.set_xlabel("Time")
                ax_ts.set_ylabel("Discharge (m3/s)")
                ax_ts.set_title(f"{bname} Discharge 1985-1989\nRMSE={rmse:.1f}, NSE={nse:.2f}, KGE={kge:.2f}, PBIAS={pbias:.1f}%")
                ax_ts.legend()
                fig.tight_layout()
                figfile = os.path.join(output_dir, f"basin_{bname}_{bid}.png")
                fig.savefig(figfile)
                plt.close(fig)
            except Exception as e:
                print(f"Error plotting for {bname}: {e}")

    try:
        df_metrics = pd.DataFrame(metrics_list)
        metrics_file = os.path.join(output_dir, "basin_streamflow_metrics.csv")
        df_metrics.to_csv(metrics_file, index=False)
    except Exception as e:
        print(f"Error saving metrics CSV: {e}")
