import os
import glob
import requests
import numpy as np
import pandas as pd
import xarray as xr
import geopandas as gpd
from shapely.geometry import Point
import matplotlib.pyplot as plt
from scipy.stats import wasserstein_distance

def ensure_dir(path):
    try:
        os.makedirs(path, exist_ok=True)
    except Exception as e:
        print(f"Could not create directory {path}: {e}")

def download_if_needed(url, local_path):
    if not os.path.exists(local_path):
        try:
            print(f"Downloading {url} to {local_path}")
            r = requests.get(url, stream=True)
            r.raise_for_status()
            with open(local_path, 'wb') as f:
                for chunk in r.iter_content(chunk_size=8192):
                    f.write(chunk)
        except Exception as e:
            print(f"Error downloading {url}: {e}")

def load_model_elm(case, start_year=1985, end_year=1989):
    path_pattern = f"./data/sample/e3sm/lnd/{case}.elm.h0.*.nc"
    files = sorted(glob.glob(path_pattern))
    ds = xr.open_mfdataset(files, combine='by_coords', parallel=False)
    ds = ds.sel(time=slice(f"{start_year}-01-01", f"{end_year}-12-31"))
    P = (ds.RAIN + ds.SNOW) * 86400.0
    ET = (ds.QVEGE + ds.QVEGT + ds.QSOIL) * 86400.0
    Q = ds.QRUNOFF * 86400.0
    return {
        'P': P.mean('time'),
        'ET': ET.mean('time'),
        'Q': Q.mean('time')
    }

def load_obs_field(variable, dataset, filename, varname, out_dir):
    url = f"https://www.ilamb.org/ILAMB-Data/DATA/{variable}/{dataset}/{filename}"
    local = os.path.join(out_dir, filename)
    download_if_needed(url, local)
    ds = xr.open_dataset(local)
    da = ds[varname]
    if 'lon' in da.coords:
        lon = da.lon.values
        if lon.max() > 180:
            lon2 = ((lon + 180) % 360) - 180
            da = da.assign_coords(lon=lon2).sortby('lon')
    da = da.sel(time=slice("1985-01-01", "1989-12-31"))
    return da.mean('time')

def basin_average(da, basins_gdf, basin_id):
    poly = basins_gdf.loc[basins_gdf.grdc_no == basin_id].geometry.values[0]
    if not poly.is_valid:
        poly = poly.buffer(0)
    lats = da.lat.values if 'lat' in da.coords else da.y.values
    lons = da.lon.values if 'lon' in da.coords else da.x.values
    mask = np.zeros((len(lats), len(lons)), dtype=bool)
    for i, lat in enumerate(lats):
        for j, lon in enumerate(lons):
            try:
                if poly.contains(Point(lon, lat)):
                    mask[i, j] = True
            except Exception:
                pass
    data = da.values
    masked = np.where(mask, data, np.nan)
    return np.nanmean(masked)

def match_gauges_to_basins(gauge_meta, basins_gdf, basin_ids):
    gauge_meta['basin_id'] = None
    for idx, row in gauge_meta.iterrows():
        pt = Point(row.lon, row.lat)
        for bid in basin_ids:
            poly = basins_gdf.loc[basins_gdf.grdc_no == bid].geometry.values[0]
            if not poly.is_valid:
                poly = poly.buffer(0)
            if poly.contains(pt):
                gauge_meta.at[idx, 'basin_id'] = bid
                break
    return gauge_meta.dropna(subset=['basin_id'])

def compute_streamflow_metrics(case, gauges_df, mosart_h0_file, daily_pattern):
    ds0 = xr.open_dataset(mosart_h0_file)
    ds_daily = xr.open_mfdataset(glob.glob(daily_pattern), combine='by_coords', parallel=False, decode_times=True)
    lat0 = ds0.lat.values
    lon0 = ds0.lon.values
    results = []
    for _, g in gauges_df.iterrows():
        i = np.abs(lat0 - g.lat).argmin()
        j = np.abs(lon0 - g.lon).argmin()
        try:
            da = ds_daily.RIVER_DISCHARGE_OVER_LAND_LIQ[:, i, j]
            sim_ser = da.to_series().dropna()
            sim_ser.index = pd.to_datetime(sim_ser.index.astype(str))
            sim = sim_ser.loc['1985-01-01':'1989-12-31']
        except Exception as e:
            print(f"Error extracting sim for gauge {g.gauge_id}: {e}")
            continue
        obs_file = f"./data/sample/obs/streamflow/{int(g.gauge_id)}.csv"
        try:
            df = pd.read_csv(obs_file, parse_dates=['date'])
            obs = df.set_index('date').discharge_m3s.loc['1985-01-01':'1989-12-31'].dropna()
        except Exception as e:
            print(f"Error reading obs {obs_file}: {e}")
            continue
        df_pair = pd.concat([sim, obs], axis=1, keys=['sim', 'obs']).dropna()
        vsim = df_pair['sim'].values
        vobs = df_pair['obs'].values
        if len(vobs) == 0:
            continue
        vol_bias = (vsim.sum() - vobs.sum()) / vobs.sum() * 100.0
        wdist = wasserstein_distance(vobs, vsim)
        results.append({
            'gauge_id': g.gauge_id,
            'basin_id': g.basin_id,
            'vol_bias': vol_bias,
            'wasserstein': wdist
        })
    return pd.DataFrame(results)

def main():
    case = "sample.v3.LR.historical"
    basin_ids = [3629000,4121801,4115200,6742900,2969100,1159100]
    output_dir = "/Users/zhou014/Local_Drive/Git_repo/esmflow/benchmark/results/o4-mini_baseline/task_07_integrated_diagnostic/run2_output"
    ensure_dir(output_dir)
    basins = gpd.read_file("./data/sample/obs/basin_polygons.geojson")
    basins = basins[basins.grdc_no.isin(basin_ids)]
    mdl = load_model_elm(case)
    ilamb_dir = "./data/ilamb"
    ensure_dir(ilamb_dir)
    pr_obs = load_obs_field('pr','GPCCv2018','pr.nc','pr',ilamb_dir)
    et_obs = load_obs_field('evspsbl','MODIS','et_0.5x0.5.nc','et',ilamb_dir)
    q_obs  = load_obs_field('mrro','LORA','LORA.nc','mrro',ilamb_dir)
    metrics = []
    for bid in basin_ids:
        pm = basin_average(mdl['P'], basins, bid)
        pe = basin_average(pr_obs, basins, bid)
        etm = basin_average(mdl['ET'], basins, bid)
        ete = basin_average(et_obs, basins, bid)
        qm = basin_average(mdl['Q'], basins, bid)
        qe = basin_average(q_obs, basins, bid)
        pbias = (pm - pe) / pe * 100.0 if pe != 0 else np.nan
        ebias = (etm - ete) / ete * 100.0 if ete != 0 else np.nan
        qbias = (qm - qe) / qe * 100.0 if qe != 0 else np.nan
        res_mod = pm - etm - qm
        res_obs = pe - ete - qe
        metrics.append({
            'basin_id': bid,
            'P_model': pm,
            'P_obs': pe,
            'ET_model': etm,
            'ET_obs': ete,
            'Q_model': qm,
            'Q_obs': qe,
            'P_bias(%)': pbias,
            'ET_bias(%)': ebias,
            'Q_bias(%)': qbias,
            'res_mod': res_mod,
            'res_obs': res_obs
        })
    metrics_df = pd.DataFrame(metrics).set_index('basin_id')
    gauge_meta = pd.read_csv("./data/sample/obs/gauge_metadata.csv")
    gauge_meta_matched = match_gauges_to_basins(gauge_meta, basins, basin_ids)
    mosart_h0 = f"./data/sample/e3sm/rof/{case}.mosart.h0.1985-01.nc"
    daily_pattern = f"./data/sample/e3sm/rof/{case}.mosart.h1.*-*-*-00000.nc"
    sf_df = compute_streamflow_metrics(case, gauge_meta_matched, mosart_h0, daily_pattern)
    sf_basin = sf_df.groupby('basin_id').agg({'vol_bias':'mean','wasserstein':'mean'})
    summary = metrics_df.join(sf_basin)
    summary.to_csv(os.path.join(output_dir, "basin_metrics_summary.csv"))
    try:
        fig, ax = plt.subplots(figsize=(10,6))
        x = np.arange(len(basin_ids))
        width = 0.15
        ax.bar(x - width*2, summary.P_obs, width, label='P obs')
        ax.bar(x - width, summary.P_model, width, label='P mod')
        ax.bar(x, summary.ET_obs, width, label='ET obs')
        ax.bar(x + width, summary.ET_model, width, label='ET mod')
        ax.bar(x + width*2, summary.Q_obs, width, label='Q obs')
        ax.bar(x + width*3, summary.Q_model, width, label='Q mod')
        ax.set_xticks(x + width)
        ax.set_xticklabels([str(b) for b in basin_ids], rotation=45)
        ax.legend(ncol=2)
        ax.set_ylabel('mm/day or m3/s')
        plt.tight_layout()
        plt.savefig(os.path.join(output_dir, "bar_p_et_q.png"))
        plt.close(fig)
    except Exception as e:
        print(f"Error plotting bar chart: {e}")
    try:
        labels = ['P_bias','ET_bias','Q_bias','Qf_bias','Wasserstein']
        angles = np.linspace(0, 2*np.pi, len(labels), endpoint=False).tolist()
        angles += angles[:1]
        fig = plt.figure(figsize=(8,8))
        ax = fig.add_subplot(111, polar=True)
        for bid in basin_ids:
            vals = [
                summary.at[bid,'P_bias(%)'],
                summary.at[bid,'ET_bias(%)'],
                summary.at[bid,'Q_bias(%)'],
                summary.at[bid,'vol_bias'],
                summary.at[bid,'wasserstein']
            ]
            vals += vals[:1]
            ax.plot(angles, vals, label=str(bid))
            ax.fill(angles, vals, alpha=0.1)
        ax.set_thetagrids(np.degrees(angles), labels)
        ax.legend(loc='upper right', bbox_to_anchor=(1.3, 1.1))
        plt.tight_layout()
        plt.savefig(os.path.join(output_dir, "radar_diagnostics.png"))
        plt.close(fig)
    except Exception as e:
        print(f"Error plotting radar chart: {e}")

if __name__ == "__main__":
    main()
