import os
import numpy as np
import pandas as pd
import xarray as xr
from scipy.stats import wasserstein_distance
import matplotlib.pyplot as plt
import cartopy.crs as ccrs
import cartopy.feature as cfeature
import matplotlib.gridspec as gridspec
from datetime import datetime

# Configuration
case = "sample.v3.LR.historical"

# Data directories
data_dir = "./data/sample"
rof_dir = os.path.join(data_dir, "e3sm", "rof")
obs_stream_dir = os.path.join(data_dir, "obs", "streamflow")
meta_file = os.path.join(data_dir, "obs", "gauge_metadata.csv")

out_dir = "/Users/zhou014/Local_Drive/Git_repo/esmflow/benchmark/results/o4-mini_baseline/task_04_streamflow_fdc/run2_output"
os.makedirs(out_dir, exist_ok=True)

# Read gauge metadata
meta = pd.read_csv(meta_file, dtype={'gauge_id': str})
meta = meta.set_index('gauge_id')

# Time range
start = datetime(1985,1,1)
end = datetime(1989,12,31)
time_index = pd.date_range(start, end, freq='D')

# Read monthly h0 file to get grid lat/lon
sample_file = os.path.join(rof_dir, f"{case}.mosart.h0.{start.strftime('%Y-%m')}.nc")
ds0 = xr.open_dataset(sample_file)
lats = ds0['lat'].values
lons = ds0['lon'].values
ds0.close()

# Determine if lon needs wrap
lon_wrap = (lons.min() >= 0)

# Find nearest grid cell for each gauge
gauge_idx = {}
for gid, row in meta.iterrows():
    lat0 = row['lat']
    lon0 = row['lon']
    if lon_wrap:
        lon0 = lon0 % 360
    dist = np.sqrt((lats - lat0)**2 + (lons - lon0)**2)
    idx = np.unravel_index(np.argmin(dist), dist.shape)
    gauge_idx[gid] = idx

# Load observed streamflow
obs_ts = {}
for gid in meta.index:
    path = os.path.join(obs_stream_dir, f"{gid}.csv")
    try:
        df = pd.read_csv(path, parse_dates=['date'])
        df = df.set_index('date').reindex(time_index)
        obs_ts[gid] = df['discharge_m3s'].values
    except Exception as e:
        print(f"Error reading obs for {gid}: {e}")
        obs_ts[gid] = np.full(len(time_index), np.nan)

# Load simulated discharge daily (h1)
sim_ts = {gid: [] for gid in meta.index}
for dt in time_index:
    fn = os.path.join(rof_dir, f"{case}.mosart.h1.{dt.strftime('%Y-%m-%d')}-00000.nc")
    try:
        ds = xr.open_dataset(fn)
        var = ds['RIVER_DISCHARGE_OVER_LAND_LIQ'].values
        ds.close()
        for gid, idx in gauge_idx.items():
            sim_ts[gid].append(var[idx])
    except Exception as e:
        # missing daily file, append nan
        for gid in sim_ts:
            sim_ts[gid].append(np.nan)
for gid in sim_ts:
    sim_ts[gid] = np.array(sim_ts[gid])

# Compute metrics and FDCs
metrics = []
fdc_percentiles = np.arange(1,100)
fdc_data = {}
for gid in meta.index:
    obs = obs_ts[gid]
    sim = sim_ts[gid]
    valid = ~np.isnan(obs) & ~np.isnan(sim)
    obs1 = obs[valid]
    sim1 = sim[valid]
    if len(obs1) == 0:
        continue
    vol_bias = (sim1.sum() - obs1.sum()) / obs1.sum() * 100.0
    emd = wasserstein_distance(obs1, sim1)
    qs = [10,50,90]
    qo = np.percentile(obs1, qs)
    qs_ = np.percentile(sim1, qs)
    qr = qs_ / qo
    metrics.append({
        'gauge_id': gid,
        'vol_bias_pct': vol_bias,
        'wasserstein_dist': emd,
        'Q10_ratio': qr[0],
        'Q50_ratio': qr[1],
        'Q90_ratio': qr[2]
    })
    exceed = fdc_percentiles
    obs_fdc = np.percentile(obs1, 100 - exceed)
    sim_fdc = np.percentile(sim1, 100 - exceed)
    df_fdc = pd.DataFrame({
        'exceedance_pct': exceed,
        'obs_m3s': obs_fdc,
        'sim_m3s': sim_fdc
    })
    fdc_data[gid] = df_fdc
    try:
        df_fdc.to_csv(os.path.join(out_dir, f"fdc_{gid}.csv"), index=False)
    except Exception as e:
        print(f"Error saving FDC CSV for {gid}: {e}")

# Save metrics
df_metrics = pd.DataFrame(metrics)
try:
    df_metrics.to_csv(os.path.join(out_dir, "streamflow_metrics.csv"), index=False)
except Exception as e:
    print(f"Error saving metrics CSV: {e}")

# Plot map and FDC panels
selected = ['3629000','4121801','4115200','6742900','2969100','1159100']
try:
    fig = plt.figure(figsize=(14,10))
    gs = gridspec.GridSpec(3,3, height_ratios=[1,1,1])
    ax_map = fig.add_subplot(gs[0,:], projection=ccrs.PlateCarree())
    ax_map.coastlines()
    ax_map.add_feature(cfeature.BORDERS, linewidth=0.5)
    merged = df_metrics.set_index('gauge_id').join(meta[['lon','lat']])
    sc = ax_map.scatter(merged['lon'], merged['lat'], c=merged['wasserstein_dist'],
                        cmap='viridis', s=50, edgecolor='k', transform=ccrs.PlateCarree())
    cb = plt.colorbar(sc, ax=ax_map, orientation='vertical', pad=0.02)
    cb.set_label('Wasserstein Distance (m3/s)')
    ax_map.set_title('Gauge Wasserstein Distance Map')
    for i, gid in enumerate(selected):
        row = 1 + i//3
        col = i%3
        ax = fig.add_subplot(gs[row, col])
        if gid not in fdc_data:
            ax.set_visible(False)
            continue
        df = fdc_data[gid]
        ax.plot(df['exceedance_pct'], df['obs_m3s'], label='Obs', color='blue')
        ax.plot(df['exceedance_pct'], df['sim_m3s'], label='Sim', color='red')
        name = meta.loc[gid,'river_name'] if gid in meta.index else gid
        ax.set_title(f"{name} ({gid})")
        ax.set_xlabel('Exceedance (%)')
        ax.set_ylabel('Discharge (m3/s)')
        ax.invert_xaxis()
        ax.grid(True, linestyle=':')
        if i==0:
            ax.legend()
    plt.tight_layout()
    figpath = os.path.join(out_dir, "streamflow_fdc_map_panels.png")
    plt.savefig(figpath, dpi=300)
    plt.close(fig)
except Exception as e:
    print(f"Error during plotting: {e}")
