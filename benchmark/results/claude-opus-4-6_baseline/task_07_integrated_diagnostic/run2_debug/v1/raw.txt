def fetch_ilamb(variable, dataset, filename, force=False):
    url = f"{ILAMB_BASE}/{variable}/{dataset}/{filename}"
    local_path = os.path.join(OBS_CACHE, f"{variable}_{dataset}_{filename}")
    if os.path.exists(local_path) and not force:
        fsize = os.path.getsize(local_path)
        if fsize < 1000:
            log(f"Cached file too small ({fsize} bytes), re-downloading: {local_path}")
            os.remove(local_path)
        else:
            log(f"Using cached: {local_path} ({fsize} bytes)")
            return local_path
    if os.path.exists(local_path) and force:
        os.remove(local_path)
    log(f"Downloading: {url}")
    try:
        import urllib.request
        urllib.request.urlretrieve(url, local_path)
        fsize = os.path.getsize(local_path)
        log(f"Downloaded to: {local_path} ({fsize} bytes)")
        if fsize < 1000:
            log(f"WARNING: Downloaded file suspiciously small ({fsize} bytes)")
            return None
        return local_path
    except Exception as e:
        log(f"Download failed: {e}")
        if os.path.exists(local_path):
            os.remove(local_path)
        return None

def get_basin_polygon(geojson, gauge_id):
    gauge_id_str = str(gauge_id)
    for feature in geojson['features']:
        props = feature['properties']
        fid = str(props.get('grdc_no', props.get('gauge_id', '')))
        if fid == gauge_id_str:
            return feature['geometry']
    return None

def point_in_polygon_ray(x, y, polygon_coords):
    n = len(polygon_coords)
    inside = False
    j = n - 1
    for i in range(n):
        xi, yi = polygon_coords[i]
        xj, yj = polygon_coords[j]
        if ((yi > y) != (yj > y)) and (x < (xj - xi) * (y - yi) / (yj - yi) + xi):
            inside = not inside
        j = i
    return inside

def get_all_rings(geometry):
    """Extract all exterior rings from a geometry."""
    rings = []
    if geometry['type'] == 'Polygon':
        rings.append(geometry['coordinates'][0])
    elif geometry['type'] == 'MultiPolygon':
        for poly in geometry['coordinates']:
            rings.append(poly[0])
    return rings

def basin_mean_gridded(data_field, lat_1d, lon_1d, geometry, lon_360=False):
    if data_field is None or geometry is None:
        return np.nan

    if isinstance(data_field, xr.DataArray):
        data = data_field.values.copy()
    else:
        data = np.array(data_field).copy()

    if isinstance(lat_1d, xr.DataArray):
        lats = lat_1d.values
    else:
        lats = np.array(lat_1d)
    if isinstance(lon_1d, xr.DataArray):
        lons = lon_1d.values
    else:
        lons = np.array(lon_1d)

    if lats.ndim != 1 or lons.ndim != 1:
        log(f"  WARNING: Expected 1D lat/lon but got shapes {lats.shape}, {lons.shape}")
        return np.nan

    # Convert lon to -180..180 for matching
    if lon_360:
        lons_adj = np.where(lons > 180, lons - 360, lons)
    else:
        lons_adj = lons.copy()

    rings = get_all_rings(geometry)
    if len(rings) == 0:
        return np.nan

    # Get bounding box of all rings
    all_coords = np.concatenate([np.array(r) for r in rings])
    min_lon_b = all_coords[:, 0].min() - 1
    max_lon_b = all_coords[:, 0].max() + 1
    min_lat_b = all_coords[:, 1].min() - 1
    max_lat_b = all_coords[:, 1].max() + 1

    # Find lat/lon indices within bounding box
    lat_mask = (lats >= min_lat_b) & (lats <= max_lat_b)
    lon_mask = (lons_adj >= min_lon_b) & (lons_adj <= max_lon_b)

    lat_indices = np.where(lat_mask)[0]
    lon_indices = np.where(lon_mask)[0]

    if len(lat_indices) == 0 or len(lon_indices) == 0:
        log(f"  No grid points in bounding box. lat range: {lats.min():.1f}-{lats.max():.1f}, lon range: {lons_adj.min():.1f}-{lons_adj.max():.1f}")
        log(f"  Basin bbox: lat {min_lat_b:.1f}-{max_lat_b:.1f}, lon {min_lon_b:.1f}-{max_lon_b:.1f}")
        return np.nan

    # Check each grid point in bounding box
    weighted_sum = 0.0
    weight_sum = 0.0
    count = 0

    for i in lat_indices:
        lt = lats[i]
        w = np.cos(np.deg2rad(lt))
        for j in lon_indices:
            ln = lons_adj[j]
            val = data[i, j]
            if np.isnan(val):
                continue
            for ring in rings:
                if point_in_polygon_ray(ln, lt, ring):
                    weighted_sum += val * w
                    weight_sum += w
                    count += 1
                    break

    if count == 0:
        log(f"  No valid grid points found inside polygon (checked {len(lat_indices)*len(lon_indices)} candidates)")
        return np.nan

    log(f"  Found {count} grid points inside basin")
    return float(weighted_sum / weight_sum)

def find_nearest_ij(target_lat, target_lon, lat_1d, lon_1d):
    """Find nearest (i_lat, i_lon) indices on a regular grid."""
    lon_adj = np.where(lon_1d > 180, lon_1d - 360, lon_1d)
    i_lat = int(np.argmin(np.abs(lat_1d - target_lat)))
    i_lon = int(np.argmin(np.abs(lon_adj - target_lon)))
    log(f"  Nearest cell: lat[{i_lat}]={lat_1d[i_lat]:.2f}, lon[{i_lon}]={lon_adj[i_lon]:.2f}")
    return i_lat, i_lon

    ET_bias = r['model_ET'] - r['obs_ET'] if not (np.isnan(r['model_ET']) or np.isnan(r['obs_ET'])) else np.nan
    Q_bias = r['model_Q'] - r['obs_Q'] if not (np.isnan(r['model_Q']) or np.isnan(r['obs_Q'])) else np.nan

    P_bias_pct = P_bias / r['obs_P'] * 100 if not np.isnan(P_bias) and abs(r['obs_P']) > 1e-10 else np.nan
    ET_bias_pct = ET_bias / r['obs_ET'] * 100 if not np.isnan(ET_bias) and abs(r['obs_ET']) > 1e-10 else np.nan
    Q_bias_pct = Q_bias / r['obs_Q'] * 100 if not np.isnan(Q_bias) and abs(r['obs_Q']) > 1e-10 else np.nan

    model_wb = r['model_P'] - r['model_ET'] - r['model_Q'] if not any(np.isnan(x) for x in [r['model_P'], r['model_ET'], r['model_Q']]) else np.nan
    obs_wb = r['obs_P'] - r['obs_ET'] - r['obs_Q'] if not any(np.isnan(x) for x in [r['obs_P'], r['obs_ET'], r['obs_Q']]) else np.nan

    summary_rows.append({
        'gauge_id': gid,
        'basin_name': bname,
        'model_P_mm_day': r['model_P'],
        'obs_P_mm_day': r['obs_P'],
        'P_bias_mm_day': P_bias,
        'P_bias_pct': P_bias_pct,
        'model_ET_mm_day': r['model_ET'],
        'obs_ET_mm_day': r['obs_ET'],
        'ET_bias_mm_day': ET_bias,
        'ET_bias_pct': ET_bias_pct,
        'model_Q_mm_day': r['model_Q'],
        'obs_Q_mm_day': r['obs_Q'],
        'Q_bias_mm_day': Q_bias,
        'Q_bias_pct': Q_bias_pct,
        'model_streamflow_m3s': r.get('model_streamflow_mean', np.nan),
        'obs_streamflow_m3s': r.get('obs_streamflow_mean', np.nan),
        'streamflow_volume_bias_pct': r.get('streamflow_volume_bias', np.nan),
        'wasserstein_distance': r.get('wasserstein_dist', np.nan),
        'model_water_balance_residual': model_wb,
        'obs_water_balance_residual': obs_wb,
    })

df_summary = pd.DataFrame(summary_rows)

# ---- Radar chart: Multi-variable diagnostic per basin ----
log("Creating radar chart...")
try:
    categories = ['|P Bias| (%)', '|ET Bias| (%)', '|Runoff Bias| (%)',
                  '|Streamflow\nBias| (%)', '|WB Residual|\n(mm/day)', 'Wasserstein\n(m³/s)']
    N = len(categories)
    angles = np.linspace(0, 2 * np.pi, N, endpoint=False).tolist()
    angles += angles[:1]

    raw_metrics = np.zeros((n_basins, N))
    for i, (gid, bname) in enumerate(BASINS.items()):
        row = df_summary[df_summary['gauge_id'] == gid].iloc[0]
        raw_metrics[i, 0] = abs(row['P_bias_pct']) if not np.isnan(row['P_bias_pct']) else 0
        raw_metrics[i, 1] = abs(row['ET_bias_pct']) if not np.isnan(row['ET_bias_pct']) else 0
        raw_metrics[i, 2] = abs(row['Q_bias_pct']) if not np.isnan(row['Q_bias_pct']) else 0
        raw_metrics[i, 3] = abs(row['streamflow_volume_bias_pct']) if not np.isnan(row['streamflow_volume_bias_pct']) else 0
        raw_metrics[i, 4] = abs(row['model_water_balance_residual']) if not np.isnan(row['model_water_balance_residual']) else 0
        raw_metrics[i, 5] = row['wasserstein_distance'] if not np.isnan(row['wasserstein_distance']) else 0

    # Normalize each metric to [0, 1]
    norm_metrics = np.zeros_like(raw_metrics)
    for j in range(N):
        col_max = raw_metrics[:, j].max()
        if col_max > 0:
            norm_metrics[:, j] = raw_metrics[:, j] / col_max

    fig, ax = plt.subplots(figsize=(10, 10), subplot_kw=dict(polar=True))
    colors = plt.cm.tab10(np.linspace(0, 0.6, n_basins))

    for i, (gid, bname) in enumerate(BASINS.items()):
        values = norm_metrics[i, :].tolist()
        values += values[:1]
        ax.plot(angles, values, 'o-', linewidth=2, label=bname, color=colors[i], markersize=6)
        ax.fill(angles, values, alpha=0.08, color=colors[i])

    ax.set_xticks(angles[:-1])
    ax.set_xticklabels(categories, fontsize=10)
    ax.set_ylim(0, 1.15)
    ax.set_yticks([0.25, 0.5, 0.75, 1.0])
    ax.set_yticklabels(['0.25', '0.50', '0.75', '1.00'], fontsize=8)
    ax.set_title('Normalized Multi-Variable Basin Diagnostic\n(0 = best, 1 = worst among basins)',
                 fontsize=14, fontweight='bold', pad=25)
    ax.legend(loc='upper right', bbox_to_anchor=(1.35, 1.1), fontsize=10)
    ax.grid(True)

    fig_path = os.path.join(OUTDIR, "basin_radar_diagnostic.png")
    plt.savefig(fig_path, dpi=200, bbox_inches='tight')
    plt.close()
    log(f"Radar chart saved to {fig_path}")
except Exception as e:
    log(f"Error creating radar chart: {e}")
    import traceback; traceback.print_exc()
