def download_ilamb(variable, dataset, filename):
    """Download ILAMB observation data if not cached."""
    cache_path = os.path.join(ILAMB_CACHE, f"{dataset}_{os.path.basename(filename)}")
    if not os.path.exists(cache_path):
        url = f"{ILAMB_BASE}/{variable}/{dataset}/{filename}"
        print(f"  Downloading {url}")
        try:
            urllib.request.urlretrieve(url, cache_path)
        except Exception as e:
            print(f"  ERROR downloading {url}: {e}")
            return None
    else:
        print(f"  Using cached {cache_path}")
    return cache_path

def get_basin_mask_gridded(lats, lons, polygon, lon_is_0_360=False):
    """Create a boolean mask for gridded data within a polygon."""
    if lats.ndim == 1 and lons.ndim == 1:
        lon2d, lat2d = np.meshgrid(lons, lats)
    else:
        lat2d, lon2d = lats, lons

    mask = np.zeros(lat2d.shape, dtype=bool)
    # Check if we need to convert longitudes
    minx, miny, maxx, maxy = polygon.bounds

    for i in range(lat2d.shape[0]):
        for j in range(lat2d.shape[1]):
            lat_val = lat2d[i, j]
            lon_val = lon2d[i, j]
            # Convert 0-360 to -180-180 if needed
            if lon_is_0_360 and lon_val > 180:
                lon_check = lon_val - 360
            elif not lon_is_0_360 and minx > 100 and lon_val < 0:
                lon_check = lon_val + 360
            else:
                lon_check = lon_val
            # Quick bounding box check
            if miny - 1 <= lat_val <= maxy + 1:
                if minx - 1 <= lon_check <= maxx + 1:
                    if polygon.contains(Point(lon_check, lat_val)):
                        mask[i, j] = True
    return mask


def compute_basin_mean_gridded(field, lats, lons, polygon):
    """Compute basin-averaged value from a gridded field."""
    if field is None:
        return np.nan

    vals = field.values if hasattr(field, 'values') else field
    if lats.ndim == 1 and lons.ndim == 1:
        lon2d, lat2d = np.meshgrid(lons, lats)
    else:
        lat2d, lon2d = lats, lons

    # Determine if lon is 0-360
    lon_is_0_360 = lons.max() > 180 if lons.ndim == 1 else lon2d.max() > 180

    minx, miny, maxx, maxy = polygon.bounds
    # Convert polygon bounds if 0-360
    # We need to handle the case where polygon is in -180..180 but data is in 0..360
    values_in_basin = []
    weights_in_basin = []

    for i in range(lat2d.shape[0]):
        for j in range(lat2d.shape[1]):
            lat_val = float(lat2d[i, j])
            lon_val = float(lon2d[i, j])

            # Convert lon for polygon check
            if lon_is_0_360 and lon_val > 180:
                lon_check = lon_val - 360.0
            else:
                lon_check = lon_val

            pt = Point(lon_check, lat_val)
            if polygon.contains(pt):
                v = float(vals[i, j])
                if np.isfinite(v):
                    w = np.cos(np.deg2rad(lat_val))
                    values_in_basin.append(v)
                    weights_in_basin.append(w)

    if len(values_in_basin) == 0:
        return np.nan

    values_in_basin = np.array(values_in_basin)
    weights_in_basin = np.array(weights_in_basin)
    return np.average(values_in_basin, weights=weights_in_basin)


def compute_basin_mean_unstructured(field, lats, lons, polygon):
    """Compute basin-averaged value from an unstructured grid field."""
    if field is None:
        return np.nan

    vals = field.values if hasattr(field, 'values') else field
    vals = vals.ravel()
    lats_flat = lats.ravel()
    lons_flat = lons.ravel()

    minx, miny, maxx, maxy = polygon.bounds

    values_in_basin = []
    weights_in_basin = []

    for k in range(len(lats_flat)):
        lat_val = float(lats_flat[k])
        lon_val = float(lons_flat[k])

        # Convert lon if > 180
        lon_check = lon_val - 360.0 if lon_val > 180 else lon_val

        # Bounding box check
        if miny - 1 <= lat_val <= maxy + 1 and minx - 1 <= lon_check <= maxx + 1:
            pt = Point(lon_check, lat_val)
            if polygon.contains(pt):
                v = float(vals[k])
                if np.isfinite(v):
                    w = np.cos(np.deg2rad(lat_val))
                    values_in_basin.append(v)
                    weights_in_basin.append(w)

    if len(values_in_basin) == 0:
        return np.nan

    return np.average(np.array(values_in_basin), weights=np.array(weights_in_basin))

def find_nearest_mosart_cell(target_lat, target_lon, grid_lat, grid_lon):
    """Find index of nearest MOSART grid cell to target location."""
    if grid_lat is None:
        return None

    target_lon_adj = target_lon
    if target_lon < 0 and grid_lon.max() > 180:
        target_lon_adj = target_lon + 360
    elif target_lon > 180 and grid_lon.min() < 0:
        target_lon_adj = target_lon - 360

    if grid_lat.ndim == 1:
        # 1D coords: structured grid
        lat_idx = np.argmin(np.abs(grid_lat - target_lat))
        lon_idx = np.argmin(np.abs(grid_lon - target_lon_adj))
        return (lat_idx, lon_idx)
    elif grid_lat.ndim == 2:
        # 2D structured
        dist = (grid_lat - target_lat)**2 + (grid_lon - target_lon_adj)**2
        idx = np.unravel_index(np.argmin(dist), dist.shape)
        return idx
    else:
        # Unstructured
        dist = (grid_lat.ravel() - target_lat)**2 + (grid_lon.ravel() - target_lon_adj)**2
        return np.argmin(dist)


for gid, bname in BASINS.items():
    print(f"\n  Processing streamflow for {bname} ({gid})...")

    basin_results[gid]['volume_bias'] = np.nan
    basin_results[gid]['wasserstein'] = np.nan
    basin_results[gid]['obs_mean_discharge'] = np.nan
    basin_results[gid]['model_mean_discharge'] = np.nan

    # Load observed streamflow
    obs_csv = os.path.join(STREAMFLOW_OBS_DIR, f"{gid}.csv")
    if not os.path.exists(obs_csv):
        print(f"    WARNING: No obs CSV found at {obs_csv}")
        continue

    try:
        obs_df = pd.read_csv(obs_csv, parse_dates=['date'])
        obs_df = obs_df[(obs_df['date'].dt.year >= 1985) & (obs_df['date'].dt.year <= 1989)]
        obs_df = obs_df.dropna(subset=['discharge_m3s'])
        print(f"    Obs: {len(obs_df)} daily records")

        if len(obs_df) == 0:
            print("    WARNING: No obs data in 1985-1989 period")
            continue

    except Exception as e:
        print(f"    ERROR loading obs: {e}")
        continue

    # Find gauge location
    gauge_row = gauge_meta[gauge_meta['gauge_id'].astype(str) == str(gid)]
    if len(gauge_row) == 0:
        print(f"    WARNING: Gauge {gid} not in metadata")
        continue

    gauge_lat = float(gauge_row['lat'].values[0])
    gauge_lon = float(gauge_row['lon'].values[0])
    print(f"    Gauge location: ({gauge_lat:.2f}, {gauge_lon:.2f})")

    # Extract model discharge at nearest grid cell
    if mosart_discharge is None or mosart_lat is None:
        print("    WARNING: No MOSART discharge data available")
        continue

    try:
        idx = find_nearest_mosart_cell(gauge_lat, gauge_lon, mosart_lat, mosart_lon)
        print(f"    Nearest MOSART cell index: {idx}")

        if mosart_lat.ndim == 1 and isinstance(idx, tuple):
            actual_lat = float(mosart_lat[idx[0]])
            actual_lon = float(mosart_lon[idx[1]])
            sim_ts = mosart_discharge[:, idx[0], idx[1]]
        elif mosart_lat.ndim == 2 and isinstance(idx, tuple):
            actual_lat = float(mosart_lat[idx[0], idx[1]])
            actual_lon = float(mosart_lon[idx[0], idx[1]])
            sim_ts = mosart_discharge[:, idx[0], idx[1]]
        else:
            # Unstructured
            actual_lat = float(mosart_lat.ravel()[idx])
            actual_lon = float(mosart_lon.ravel()[idx])
            # Get the dim name for the spatial dimension
            spatial_dims = [d for d in mosart_discharge.dims if d != 'time']
            if len(spatial_dims) == 1:
                sim_ts = mosart_discharge.isel({spatial_dims[0]: int(idx)})
            else:
                sim_ts = mosart_discharge.isel({spatial_dims[0]: int(idx)})

        print(f"    Matched to MOSART cell at ({actual_lat:.2f}, {actual_lon:.2f})")

        sim_vals = sim_ts.values.ravel()
        sim_vals = sim_vals[np.isfinite(sim_vals)]
        obs_vals = obs_df['discharge_m3s'].values

        if len(sim_vals) > 0 and len(obs_vals) > 0:
            # Volume bias
            obs_mean = np.mean(obs_vals)
            sim_mean = np.mean(sim_vals)
            if obs_mean > 0:
                vol_bias = (sim_mean - obs_mean) / obs_mean * 100
            else:
                vol_bias = np.nan

            basin_results[gid]['volume_bias'] = vol_bias
            basin_results[gid]['obs_mean_discharge'] = obs_mean
            basin_results[gid]['model_mean_discharge'] = sim_mean

            # Wasserstein distance on FDC
            obs_sorted = np.sort(obs_vals)[::-1]
            sim_sorted = np.sort(sim_vals)[::-1]

            # Normalize by obs mean for comparable metric
            obs_norm = obs_sorted / max(obs_mean, 1)
            sim_norm = sim_sorted / max(obs_mean, 1)

            wd = wasserstein_distance(obs_norm, sim_norm)
            basin_results[gid]['wasserstein'] = wd

            print(f"    Obs mean Q: {obs_mean:.1f} m3/s, Model mean Q: {sim_mean:.1f} m3/s")
            print(f"    Volume bias: {vol_bias:.1f}%, Wasserstein distance: {wd:.3f}")
        else:
            print(f"    WARNING: Empty sim ({len(sim_vals)}) or obs ({len(obs_vals)}) arrays")

    except Exception as e:
        print(f"    ERROR extracting MOSART discharge: {e}")
        import traceback; traceback.print_exc()

    mod_Q = br.get('model_Q', np.nan)
    obs_Q = br.get('obs_Q', np.nan)

    ET_bias = ((mod_ET - obs_ET) / obs_ET * 100) if (np.isfinite(obs_ET) and obs_ET != 0) else np.nan
    Q_bias = ((mod_Q - obs_Q) / obs_Q * 100) if (np.isfinite(obs_Q) and obs_Q != 0) else np.nan

    # Water balance residual: P - ET - Q (should be ~0 for long term)
    model_wb = mod_P - mod_ET - mod_Q if all(np.isfinite([mod_P, mod_ET, mod_Q])) else np.nan
    obs_wb = obs_P - obs_ET - obs_Q if all(np.isfinite([obs_P, obs_ET, obs_Q])) else np.nan

    rows.append({
        'basin_id': gid,
        'basin_name': bname,
        'model_P_mm_day': mod_P,
        'obs_P_mm_day': obs_P,
        'P_bias_pct': P_bias,
        'model_ET_mm_day': mod_ET,
        'obs_ET_mm_day': obs_ET,
        'ET_bias_pct': ET_bias,
        'model_Q_mm_day': mod_Q,
        'obs_Q_mm_day': obs_Q,
        'Q_bias_pct': Q_bias,
        'streamflow_vol_bias_pct': br.get('volume_bias', np.nan),
        'wasserstein_distance': br.get('wasserstein', np.nan),
        'model_WB_residual_mm_day': model_wb,
        'obs_WB_residual_mm_day': obs_wb,
        'obs_mean_discharge_m3s': br.get('obs_mean_discharge', np.nan),
        'model_mean_discharge_m3s': br.get('model_mean_discharge', np.nan),
