import os
import json
import warnings
import numpy as np
import pandas as pd
import xarray as xr
import matplotlib.pyplot as plt
import matplotlib.gridspec as gridspec
from matplotlib.path import Path as MplPath
import cartopy.crs as ccrs
import cartopy.feature as cfeature

warnings.filterwarnings("ignore")

outdir = "/Users/zhou014/Local_Drive/Git_repo/esmflow/benchmark/results/claude-opus-4-6_baseline/task_06_water_balance/run4_output"
os.makedirs(outdir, exist_ok=True)

case_name = "sample.v3.LR.historical"
elm_dir = "./data/sample/e3sm/lnd"
years = range(1985, 1990)

basin_info = {
    "3629000": "Amazon",
    "4121801": "Missouri",
    "4115200": "Columbia",
    "6742900": "Danube",
    "2969100": "Mekong",
    "1159100": "Orange",
}

# ============================================================
# 1. Load monthly ELM data for 1985-1989
# ============================================================
print("Loading ELM monthly data for 1985-1989...")

file_list = []
for year in years:
    for month in range(1, 13):
        fname = os.path.join(elm_dir, f"{case_name}.elm.h0.{year:04d}-{month:02d}.nc")
        if os.path.exists(fname):
            file_list.append(fname)

print(f"  Found {len(file_list)} files")

ds = xr.open_mfdataset(
    file_list, combine="by_coords", data_vars="minimal",
    coords="minimal", compat="override",
)
print(f"  Dataset dimensions: {dict(ds.dims)}")

# ============================================================
# 2. Compute time-mean fields
# ============================================================
print("Computing climatological time-mean fields...")
sec_per_day = 86400.0

P = (ds["RAIN"] + ds["SNOW"]).mean(dim="time") * sec_per_day
ET = (ds["QVEGE"] + ds["QVEGT"] + ds["QSOIL"]).mean(dim="time") * sec_per_day
Q = ds["QRUNOFF"].mean(dim="time") * sec_per_day
residual = P - ET - Q

for v, n, ln in [(P, "P", "Precipitation"), (ET, "ET", "Evapotranspiration"),
                  (Q, "Q", "Total Runoff"), (residual, "residual", "Residual")]:
    v.name = n
    v.attrs["units"] = "mm/day"
    v.attrs["long_name"] = ln

# Force compute to avoid repeated lazy evaluation
P = P.compute()
ET = ET.compute()
Q = Q.compute()
residual = residual.compute()

print(f"  P shape: {P.shape}")

# ============================================================
# 3. Compute area-weighted global means
# ============================================================
print("Computing area-weighted global means...")

area = ds["area"].isel(time=0) if "time" in ds["area"].dims else ds["area"]
landfrac = ds["landfrac"].isel(time=0) if "time" in ds["landfrac"].dims else ds["landfrac"]
area = area.compute()
landfrac = landfrac.compute()

weights = area * landfrac
weights = weights.where(P.notnull())
total_weight = float(weights.sum())

P_global = float((P * weights).sum() / total_weight)
ET_global = float((ET * weights).sum() / total_weight)
Q_global = float((Q * weights).sum() / total_weight)
res_global = P_global - ET_global - Q_global

print(f"  Global mean P:        {P_global:.4f} mm/day")
print(f"  Global mean ET:       {ET_global:.4f} mm/day")
print(f"  Global mean Q:        {Q_global:.4f} mm/day")
print(f"  Global mean Residual: {res_global:.4f} mm/day")

global_df = pd.DataFrame({
    "Component": ["P (RAIN+SNOW)", "ET (QVEGE+QVEGT+QSOIL)", "Q (QRUNOFF)", "Residual (P-ET-Q)"],
    "Global_Mean_mm_per_day": [P_global, ET_global, Q_global, res_global],
    "Global_Mean_mm_per_year": [P_global*365.25, ET_global*365.25, Q_global*365.25, res_global*365.25],
})
global_df.to_csv(os.path.join(outdir, "global_water_balance.csv"), index=False)
print("  Saved global_water_balance.csv")

# ============================================================
# 4. Load basin polygons
# ============================================================
print("Loading basin polygons...")

with open("./data/sample/obs/basin_polygons.geojson", "r") as f:
    basins_geojson = json.load(f)

gauge_meta = pd.read_csv("./data/sample/obs/gauge_metadata.csv")

lat_vals = P.coords["lat"].values
lon_vals = P.coords["lon"].values
lon2d, lat2d = np.meshgrid(lon_vals, lat_vals)
lon2d_180 = np.where(lon2d > 180, lon2d - 360, lon2d)


def polygon_to_mpl_path(geom_dict):
    """Convert GeoJSON geometry to list of matplotlib Paths (handles Polygon and MultiPolygon)."""
    gtype = geom_dict["type"]
    paths = []
    if gtype == "Polygon":
        coords = np.array(geom_dict["coordinates"][0])
        paths.append(MplPath(coords[:, :2]))
    elif gtype == "MultiPolygon":
        for poly_coords in geom_dict["coordinates"]:
            coords = np.array(poly_coords[0])
            paths.append(MplPath(coords[:, :2]))
    return paths


def get_exterior_coords(geom_dict):
    """Get list of exterior coordinate arrays for plotting."""
    gtype = geom_dict["type"]
    result = []
    if gtype == "Polygon":
        result.append(np.array(geom_dict["coordinates"][0]))
    elif gtype == "MultiPolygon":
        for poly_coords in geom_dict["coordinates"]:
            result.append(np.array(poly_coords[0]))
    return result


# Build basin data
basin_features = {}
for feature in basins_geojson["features"]:
    gid = str(feature["properties"].get("grdc_no", feature["properties"].get("gauge_id", "")))
    if gid in basin_info:
        basin_features[gid] = feature
        print(f"  Found {basin_info[gid]} ({gid})")

# ============================================================
# 5. Vectorized basin masking using matplotlib Path
# ============================================================
print("Computing basin masks (vectorized)...")

# Flatten grid points for vectorized contains_points
flat_lon = lon2d_180.ravel()
flat_lat = lat2d.ravel()
flat_points = np.column_stack([flat_lon, flat_lat])

P_vals = P.values
ET_vals = ET.values
Q_vals = Q.values
weights_vals = weights.values

basin_results = {}

for gid, bname in basin_info.items():
    if gid not in basin_features:
        print(f"  Skipping {bname}: no polygon")
        continue

    feature = basin_features[gid]
    geom = feature["geometry"]
    mpl_paths = polygon_to_mpl_path(geom)

    mask_flat = np.zeros(len(flat_points), dtype=bool)
    for path in mpl_paths:
        # Use bounding box prefilter
        bb = path.get_extents()
        bbox_mask = ((flat_lon >= bb.x0 - 1) & (flat_lon <= bb.x1 + 1) &
                     (flat_lat >= bb.y0 - 1) & (flat_lat <= bb.y1 + 1))
        candidates = np.where(bbox_mask)[0]
        if len(candidates) > 0:
            inside = path.contains_points(flat_points[candidates])
            mask_flat[candidates[inside]] = True

    mask_2d = mask_flat.reshape(P.shape)
    ncells = int(mask_2d.sum())
    print(f"  {bname}: {ncells} grid cells")

    if ncells == 0:
        continue

    w = np.where(mask_2d, weights_vals, 0.0)
    w = np.where(np.isfinite(P_vals), w, 0.0)
    wsum = w.sum()

    if wsum > 0:
        P_basin = float((np.nan_to_num(P_vals) * w).sum() / wsum)
        ET_basin = float((np.nan_to_num(ET_vals) * w).sum() / wsum)
        Q_basin = float((np.nan_to_num(Q_vals) * w).sum() / wsum)
    else:
        P_basin = float(np.nanmean(P_vals[mask_2d]))
        ET_basin = float(np.nanmean(ET_vals[mask_2d]))
        Q_basin = float(np.nanmean(Q_vals[mask_2d]))

    res_basin = P_basin - ET_basin - Q_basin

    basin_results[bname] = {
        "gauge_id": gid,
        "P_mm_day": P_basin,
        "ET_mm_day": ET_basin,
        "Q_mm_day": Q_basin,
        "Residual_mm_day": res_basin,
        "ncells": ncells,
        "geom": geom,
    }
    print(f"    P={P_basin:.3f}, ET={ET_basin:.3f}, Q={Q_basin:.3f}, Res={res_basin:.4f} mm/day")

# Save basin CSV
basin_rows = []
for bname, bd in basin_results.items():
    basin_rows.append({
        "Basin": bname, "gauge_id": bd["gauge_id"],
        "P_mm_day": bd["P_mm_day"], "ET_mm_day": bd["ET_mm_day"],
        "Q_mm_day": bd["Q_mm_day"], "Residual_mm_day": bd["Residual_mm_day"],
        "P_mm_year": bd["P_mm_day"]*365.25, "ET_mm_year": bd["ET_mm_day"]*365.25,
        "Q_mm_year": bd["Q_mm_day"]*365.25, "Residual_mm_year": bd["Residual_mm_day"]*365.25,
        "ncells": bd["ncells"],
    })
pd.DataFrame(basin_rows).to_csv(os.path.join(outdir, "basin_water_balance.csv"), index=False)
print("  Saved basin_water_balance.csv")

# ============================================================
# 6. Save fields to NetCDF
# ============================================================
print("Saving spatial fields to NetCDF...")
out_ds = xr.Dataset({"P": P, "ET": ET, "Q": Q, "residual": residual})
out_ds.attrs["description"] = "ELM water balance components (1985-1989 climatological mean)"
out_ds.to_netcdf(os.path.join(outdir, "water_balance_fields.nc"))
print("  Saved water_balance_fields.nc")

# ============================================================
# 7. Create composite figure
# ============================================================
print("Creating composite figure...")

basin_colors = {
    "Amazon": "#e41a1c", "Missouri": "#377eb8", "Columbia": "#4daf4a",
    "Danube": "#984ea3", "Mekong": "#ff7f00", "Orange": "#a65628",
}

try:
    fig = plt.figure(figsize=(18, 14))
    gs = gridspec.GridSpec(2, 1, height_ratios=[1.2, 1], hspace=0.3)

    ax_map = fig.add_subplot(gs[0], projection=ccrs.Robinson())
    ax_map.set_global()
    ax_map.coastlines(linewidth=0.5)
    ax_map.add_feature(cfeature.BORDERS, linewidth=0.3, linestyle=":")

    res_data = residual.values.copy()
    plot_lon = lon_vals.copy()
    if np.any(plot_lon > 180):
        lon_shifted = np.where(plot_lon > 180, plot_lon - 360, plot_lon)
        sort_idx = np.argsort(lon_shifted)
        lon_shifted = lon_shifted[sort_idx]
        res_data = res_data[:, sort_idx]
    else:
        lon_shifted = plot_lon

    finite = res_data[np.isfinite(res_data)]
    vmax = max(0.1, float(np.percentile(np.abs(finite), 95))) if len(finite) > 0 else 1.0

    im = ax_map.pcolormesh(
        lon_shifted, lat_vals, res_data,
        transform=ccrs.PlateCarree(), cmap="RdBu",
        vmin=-vmax, vmax=vmax, shading="auto",
    )
    cb = plt.colorbar(im, ax=ax_map, orientation="horizontal", pad=0.05, shrink=0.7, aspect=40)
    cb.set_label("Water Balance Residual (P - ET - Q) [mm/day]", fontsize=11)

    # Basin outlines
    for bname, bd in basin_results.items():
        color = basin_colors.get(bname, "black")
        ext_coords = get_exterior_coords(bd["geom"])
        for k, coords in enumerate(ext_coords):
            ax_map.plot(coords[:, 0], coords[:, 1],
                        transform=ccrs.PlateCarree(), color=color, linewidth=1.5,
                        label=bname if k == 0 else None)

    legend_lines = [plt.Line2D([0], [0], color=basin_colors.get(b, "black"), lw=2, label=b)
                    for b in basin_results]
    ax_map.legend(handles=legend_lines, loc="lower left", fontsize=8, ncol=2, framealpha=0.9)
    ax_map.set_title(
        f"ELM Water Balance Residual (P - ET - Q)\n"
        f"1985-1989 Climatological Mean | Global mean residual = {res_global:.4f} mm/day",
        fontsize=13, fontweight="bold")

    # Bottom: bar charts
    n_basins = len(basin_results)
    if n_basins > 0:
        gs_bot = gridspec.GridSpecFromSubplotSpec(1, n_basins, subplot_spec=gs[1], wspace=0.35)
        for idx, (bname, bd) in enumerate(basin_results.items()):
            ax_bar = fig.add_subplot(gs_bot[idx])
            comps = ["P", "ET", "Q", "Residual"]
            vals = [bd["P_mm_day"], bd["ET_mm_day"], bd["Q_mm_day"], bd["Residual_mm_day"]]
            cols = ["#2166ac", "#b2182b", "#1b7837", "#636363"]
            bars = ax_bar.bar(comps, vals, color=cols, edgecolor="black", linewidth=0.5)
            for bar, val in zip(bars, vals):
                yp = bar.get_height()
                ax_bar.text(bar.get_x() + bar.get_width()/2, yp,
                           f"{val:.3f}", ha="center",
                           va="bottom" if yp >= 0 else "top", fontsize=7, fontweight="bold")
            ax_bar.axhline(0, color="black", linewidth=0.5)
            ax_bar.set_title(f"{bname}\n(ID: {bd['gauge_id']})", fontsize=9, fontweight="bold",
                           color=basin_colors.get(bname, "black"))
            if idx == 0:
                ax_bar.set_ylabel("mm/day", fontsize=9)
            ax_bar.tick_params(axis="x", labelsize=8, rotation=30)
            ax_bar.tick_params(axis="y", labelsize=8)
            ymax_v = max(vals) * 1.3 if max(vals) > 0 else 1.0
            ymin_v = min(0, min(vals) * 1.5)
            ax_bar.set_ylim(ymin_v, ymax_v)

    plt.savefig(os.path.join(outdir, "water_balance_composite.png"), dpi=200, bbox_inches="tight")
    print("  Saved water_balance_composite.png")
    plt.close()
except Exception as e:
    print(f"Error creating figure: {e}")
    import traceback
    traceback.print_exc()

print("\n" + "="*60)

    pct = bd["Residual_mm_day"]/bd["P_mm_day"]*100 if bd["P_mm_day"] != 0 else 0
    print(f"  {bname:<12s} {bd['P_mm_day']:8.3f} {bd['ET_mm_day']:8.3f} {bd['Q_mm_day']:8.3f} {bd['Residual_mm_day']:10.4f} {pct:8.2f}")
print(f"\nOutputs saved to: {outdir}")
print("Done.")
