is_structured = (P_vals.ndim == 2 and lat_vals.ndim == 1 and lon_vals.ndim == 1)

# Compute area-weighted global means
print("Computing area-weighted global means...")
try:
    if "area" in ds:
        area_da = ds["area"]
        if "time" in area_da.dims:
            area_da = area_da.isel(time=0)
        area_v = area_da.values
    else:
        if is_structured:
            cos_lat = np.cos(np.deg2rad(lat_vals))
            area_v = np.broadcast_to(cos_lat[:, np.newaxis], P_vals.shape).copy()
        else:
            area_v = np.cos(np.deg2rad(lat_vals))

    if "landfrac" in ds:
        lf = ds["landfrac"]
        if "time" in lf.dims:
            lf = lf.isel(time=0)
        lf_v = lf.values
    else:
        lf_v = np.ones_like(P_vals)

    w = area_v * lf_v
    valid = np.isfinite(P_vals) & (w > 0)
    wv = w[valid]
    P_global = np.average(P_vals[valid], weights=wv)
    ET_global = np.average(ET_vals[valid], weights=wv)
    Q_global = np.average(Q_vals[valid], weights=wv)
    R_global = P_global - ET_global - Q_global
except Exception as e:
    print(f"  Fallback to simple means: {e}")
    P_global = float(np.nanmean(P_vals))
    ET_global = float(np.nanmean(ET_vals))
    Q_global = float(np.nanmean(Q_vals))
    R_global = P_global - ET_global - Q_global

print(f"  P={P_global:.4f}, ET={ET_global:.4f}, Q={Q_global:.4f}, R={R_global:.4f} mm/day")

def fast_basin_mask(geometry, flat_lon, flat_lat, flat_lon_180):
    """Vectorized basin masking using matplotlib Path."""
    if geometry["type"] == "Polygon":
        polys = [geometry["coordinates"]]
    elif geometry["type"] == "MultiPolygon":
        polys = geometry["coordinates"]
    else:
        return np.zeros(len(flat_lon), dtype=bool)

    # Check lon convention from polygon
    sample_lons = []
    for poly in polys:
        for coord in poly[0][:10]:
            sample_lons.append(coord[0])
    use_180 = any(l < 0 for l in sample_lons)
    use_lon = flat_lon_180 if use_180 else flat_lon

    mask = np.zeros(len(flat_lon), dtype=bool)

    for poly_coords in polys:
        exterior = np.array(poly_coords[0])
        # Bounding box pre-filter
        lon_min, lon_max = exterior[:, 0].min() - 0.5, exterior[:, 0].max() + 0.5
        lat_min, lat_max = exterior[:, 1].min() - 0.5, exterior[:, 1].max() + 0.5

        bbox_mask = (use_lon >= lon_min) & (use_lon <= lon_max) & (flat_lat >= lat_min) & (flat_lat <= lat_max)
        bbox_idx = np.where(bbox_mask)[0]

        if len(bbox_idx) == 0:
            continue

        path = Path(exterior)
        points = np.column_stack([use_lon[bbox_idx], flat_lat[bbox_idx]])
        inside = path.contains_points(points)
        mask[bbox_idx[inside]] = True

    return mask


basin_results = {}

print("Computing basin averages...")
for basin_name, gauge_id in basins.items():
    print(f"  {basin_name} ({gauge_id})...")
    feature = features_by_id.get(gauge_id)
    if feature is None:
        print(f"    No polygon found")
        basin_results[basin_name] = {"P": np.nan, "ET": np.nan, "Q": np.nan, "R": np.nan}
        continue

    try:
        mask_flat = fast_basin_mask(feature["geometry"], flat_lon, flat_lat, flat_lon_180)
        n_cells = np.sum(mask_flat)
        print(f"    {n_cells} grid cells")

        if n_cells == 0:
            basin_results[basin_name] = {"P": np.nan, "ET": np.nan, "Q": np.nan, "R": np.nan}
            continue

        basin_masks[basin_name] = mask_flat

        P_flat = P_vals.ravel()[mask_flat]
        ET_flat = ET_vals.ravel()[mask_flat]
        Q_flat = Q_vals.ravel()[mask_flat]

        w_basin = np.cos(np.deg2rad(flat_lat[mask_flat]))
        valid_b = np.isfinite(P_flat) & (w_basin > 0)

        if np.sum(valid_b) > 0:
            P_avg = np.average(P_flat[valid_b], weights=w_basin[valid_b])
            ET_avg = np.average(ET_flat[valid_b], weights=w_basin[valid_b])
            Q_avg = np.average(Q_flat[valid_b], weights=w_basin[valid_b])
        else:
            P_avg = np.nanmean(P_flat)
            ET_avg = np.nanmean(ET_flat)
            Q_avg = np.nanmean(Q_flat)

        R_avg = P_avg - ET_avg - Q_avg
        basin_results[basin_name] = {"P": P_avg, "ET": ET_avg, "Q": Q_avg, "R": R_avg}
        print(f"    P={P_avg:.3f}, ET={ET_avg:.3f}, Q={Q_avg:.3f}, R={R_avg:.4f}")

    except Exception as e:
        print(f"    Error: {e}")
        basin_results[basin_name] = {"P": np.nan, "ET": np.nan, "Q": np.nan, "R": np.nan}

    basin_df.to_csv(os.path.join(output_dir, "basin_water_balance.csv"))
