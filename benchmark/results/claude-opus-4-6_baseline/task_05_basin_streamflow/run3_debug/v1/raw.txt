import pandas as pd
import xarray as xr

output_dir = "/Users/zhou014/Local_Drive/Git_repo/esmflow/benchmark/results/claude-opus-4-6_baseline/task_05_basin_streamflow/run3_output"

case_name = "sample.v3.LR.historical"

basins = {

    6742900: "Danube",
    2969100: "Mekong",
    1159100: "Orange",

year_start, year_end = 1985, 1989

# ── 1. Load gauge metadata ──────────────────────────────────────────────────
try:
    gauge_meta = pd.read_csv(gauge_meta_path)
    print(f"Loaded gauge metadata: {len(gauge_meta)} gauges")
    print(gauge_meta.head())
except Exception as e:
    print(f"Error loading gauge metadata: {e}")
    gauge_meta = pd.DataFrame()

# ── 2. Load basin polygons ──────────────────────────────────────────────────
basin_polygons = {}
try:
    with open(basin_poly_path, "r") as f:
        geojson = json.load(f)
    for feature in geojson["features"]:
        gid = feature["properties"].get("grdc_no") or feature["properties"].get("gauge_id")
        if gid is not None:
            gid = int(gid)
            basin_polygons[gid] = feature["geometry"]
    print(f"Loaded {len(basin_polygons)} basin polygons")
except Exception as e:
    print(f"Error loading basin polygons: {e}")

# ── 3. Load MOSART monthly output ──────────────────────────────────────────
mosart_files = []
for year in range(year_start, year_end + 1):
    for month in range(1, 13):
        fname = os.path.join(rof_dir, f"{case_name}.mosart.h0.{year:04d}-{month:02d}.nc")
        if os.path.exists(fname):
            mosart_files.append(fname)

print(f"Found {len(mosart_files)} MOSART monthly files")

mosart_ds = None
if mosart_files:
    try:
        mosart_ds = xr.open_mfdataset(mosart_files, combine="by_coords", decode_times=True)
        print(f"MOSART dataset loaded: {mosart_ds.dims}")
        print(f"Variables: {list(mosart_ds.data_vars)}")
    except Exception as e:
        print(f"Error loading MOSART files: {e}")
        try:
            mosart_ds = xr.open_mfdataset(mosart_files, combine="nested", concat_dim="time", decode_times=True)

    except Exception as e:

    if candidates:
        discharge_var = candidates[0]
        print(f"Using alternative variable: {discharge_var}")
    else:
        print(f"Available variables: {list(sim_ds.data_vars)}")
        print("WARNING: Discharge variable not found")

if "lat" in sim_ds.coords:
    model_lat = sim_ds["lat"].values
    model_lon = sim_ds["lon"].values
elif "lat" in sim_ds.data_vars:
    model_lat = sim_ds["lat"].values
    model_lon = sim_ds["lon"].values
else:
    coord_names_lat = [c for c in sim_ds.coords if "lat" in c.lower()]
    coord_names_lon = [c for c in sim_ds.coords if "lon" in c.lower()]
    model_lat = sim_ds[coord_names_lat[0]].values if coord_names_lat else None
    model_lon = sim_ds[coord_names_lon[0]].values if coord_names_lon else None

print(f"Model lon shape: {model_lon.shape if model_lon is not None else 'None'}")

# Build KDTree for nearest-neighbor matching
if model_lat is not None and model_lon is not None:
    if model_lat.ndim == 1 and model_lon.ndim == 1:
        lon2d, lat2d = np.meshgrid(model_lon, model_lat)
        grid_points = np.column_stack([lat2d.ravel(), lon2d.ravel()])
        grid_shape = lat2d.shape
        grid_type = "regular"
    elif model_lat.ndim == 2:
        grid_points = np.column_stack([model_lat.ravel(), model_lon.ravel()])
        grid_shape = model_lat.shape
        grid_type = "2d"
    else:
        grid_points = np.column_stack([model_lat, model_lon])
        grid_shape = None
        grid_type = "1d"
    
    tree = cKDTree(grid_points)
    print(f"KDTree built with {len(grid_points)} grid points, type={grid_type}")


def find_nearest_grid_cell(lat, lon):
    """Find nearest grid cell index for a given lat/lon."""
    query_lon = lon
    if model_lon.min() >= 0 and lon < 0:
        query_lon = lon + 360
    elif model_lon.max() <= 180 and lon > 180:
        query_lon = lon - 360
    
    dist, idx = tree.query([lat, query_lon])
    
    if grid_type == "regular":
        ilat, ilon = np.unravel_index(idx, grid_shape)
        return ilat, ilon, dist
    elif grid_type == "2d":
        i, j = np.unravel_index(idx, grid_shape)
        return i, j, dist
    else:
        return idx, None, dist


def cftime_to_datetime(cftime_arr):
    """Convert cftime array to pandas DatetimeIndex."""
    dates = []
    for t in cftime_arr:
        dates.append(pd.Timestamp(year=t.year, month=t.month, day=t.day,
                                  hour=t.hour, minute=t.minute, second=t.second))
    return pd.DatetimeIndex(dates)


# ── 5. Load observations and match to model ────────────────────────────────
def load_obs(gauge_id, start_year, end_year):
    """Load observation streamflow CSV for a gauge."""
    csv_path = os.path.join(obs_dir, f"{gauge_id}.csv")
    if not os.path.exists(csv_path):
        print(f"  Observation file not found: {csv_path}")
        return None
    df = pd.read_csv(csv_path, parse_dates=["date"])
    df = df.set_index("date")
    df = df.loc[f"{start_year}":f"{end_year}"]
    return df


def extract_sim_ts(ilat, ilon, discharge_var_name):
    """Extract simulated time series at a grid point."""
    try:
        da = sim_ds[discharge_var_name]
        dims = da.dims
        
        if grid_type == "regular":
            lat_dim = [d for d in dims if "lat" in d.lower() or d == "y"]
            lon_dim = [d for d in dims if "lon" in d.lower() or d == "x"]
            
            if lat_dim and lon_dim:
                ts = da.isel({lat_dim[0]: ilat, lon_dim[0]: ilon})
            else:
                non_time = [d for d in dims if d != "time"]
                if len(non_time) == 2:
                    ts = da.isel({non_time[0]: ilat, non_time[1]: ilon})
                else:
                    ts = da.isel({non_time[0]: ilat})
        elif grid_type == "1d":
            spatial_dims = [d for d in dims if d != "time"]
            ts = da.isel({spatial_dims[0]: ilat})
        else:
            non_time = [d for d in dims if d != "time"]
            if len(non_time) == 2:
                ts = da.isel({non_time[0]: ilat, non_time[1]: ilon})
            else:
                ts = da.isel({non_time[0]: ilat})
        
        return ts
    except Exception as e:
        print(f"  Error extracting sim TS: {e}")
        return None


# ── 6. Compute validation metrics ──────────────────────────────────────────
def compute_metrics(sim, obs):
    """Compute RMSE, NSE, KGE, PBIAS between sim and obs (aligned arrays)."""
    mask = np.isfinite(sim) & np.isfinite(obs)
    s = sim[mask]
    o = obs[mask]
    
    if len(o) < 2:
        return {"RMSE": np.nan, "NSE": np.nan, "KGE": np.nan, "PBIAS": np.nan, "n": 0}
    
    rmse = np.sqrt(np.mean((s - o) ** 2))
    nse = 1 - np.sum((s - o) ** 2) / np.sum((o - np.mean(o)) ** 2)
    
    r = np.corrcoef(s, o)[0, 1]
    alpha = np.std(s) / np.std(o)
    beta = np.mean(s) / np.mean(o)
    kge = 1 - np.sqrt((r - 1) ** 2 + (alpha - 1) ** 2 + (beta - 1) ** 2)
    
    pbias = 100 * np.sum(s - o) / np.sum(o)
    
    return {"RMSE": rmse, "NSE": nse, "KGE": kge, "PBIAS": pbias, "n": len(o)}

    if gauge_meta is not None and len(gauge_meta) > 0:

        glat = ginfo["lat"].values[0]

    else:

    print(f"  Location: ({glat:.2f}, {glon:.2f}), Area: {gauge_area:.0f} km2, River: {river_name}")
    
    # Match to model grid
    ilat, ilon, dist = find_nearest_grid_cell(glat, glon)
    print(f"  Nearest grid cell: ilat={ilat}, ilon={ilon}, distance={dist:.4f} degrees")
    
    # Extract simulated discharge
    sim_ts = extract_sim_ts(ilat, ilon, discharge_var)
    if sim_ts is None:
        print("  Could not extract simulated time series, skipping")

    # Convert to dataframe, handling cftime

    sim_time = sim_ts["time"].values
    
    # Convert cftime to pandas datetime
