

```python
import os
import glob
import numpy as np
import xarray as xr
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
import cartopy.crs as ccrs
import cartopy.feature as cfeature

# Output directory
output_dir = "/Users/zhou014/Local_Drive/Git_repo/esmflow/benchmark/results/claude-opus-4-6_baseline/task_02_seasonal_runoff/run4_output"
os.makedirs(output_dir, exist_ok=True)

# Data paths
data_dir = "./data/sample/e3sm/lnd"
case_name = "sample.v3.LR.historical"

# Collect all monthly files for 1985-1989
years = range(1985, 1990)
file_list = []
for year in years:
    for month in range(1, 13):
        fname = os.path.join(data_dir, f"{case_name}.elm.h0.{year:04d}-{month:02d}.nc")
        if os.path.exists(fname):
            file_list.append(fname)
        else:
            print(f"Warning: file not found: {fname}")

print(f"Found {len(file_list)} files for 1985-1989")

if len(file_list) == 0:
    # Try glob as fallback
    pattern = os.path.join(data_dir, f"{case_name}.elm.h0.198[5-9]-*.nc")
    file_list = sorted(glob.glob(pattern))
    print(f"Glob fallback found {len(file_list)} files")

if len(file_list) == 0:
    raise FileNotFoundError("No ELM h0 files found for 1985-1989")

file_list.sort()
print(f"Processing {len(file_list)} files: {os.path.basename(file_list[0])} to {os.path.basename(file_list[-1])}")

# Open dataset
try:
    ds = xr.open_mfdataset(file_list, combine='nested', concat_dim='time', data_vars='minimal',
                           coords='minimal', compat='override')
except Exception as e:
    print(f"Error opening with nested combine: {e}")
    ds = xr.open_mfdataset(file_list, combine='by_coords', data_vars='minimal',
                           coords='minimal', compat='override')

print("Dataset dimensions:", dict(ds.dims))
print("Available variables:", list(ds.data_vars)[:20])

# Extract QRUNOFF
if 'QRUNOFF' not in ds:
    raise KeyError("QRUNOFF not found in dataset. Available variables: " + str(list(ds.data_vars)))

qrunoff = ds['QRUNOFF']
print(f"QRUNOFF shape: {qrunoff.shape}")
print(f"QRUNOFF dims: {qrunoff.dims}")
print(f"QRUNOFF units: {qrunoff.attrs.get('units', 'unknown')}")

# Compute climatological mean (time mean)
qrunoff_mean = qrunoff.mean(dim='time')
print(f"Climatological mean shape: {qrunoff_mean.shape}")

# Determine grid type and get coordinates
is_unstructured = False
if 'lat' in ds.dims and 'lon' in ds.dims:
    # Structured grid
    lats = ds['lat'].values
    lons = ds['lon'].values
    print(f"Structured grid: lat({len(lats)}) x lon({len(lons)})")
elif 'lndgrid' in ds.dims or 'gridcell' in ds.dims:
    is_unstructured = True
    grid_dim = 'lndgrid' if 'lndgrid' in ds.dims else 'gridcell'
    print(f"Unstructured grid with dim: {grid_dim}")
    # Try to find lat/lon coordinates
    lat_var = None
    lon_var = None
    for vname in ['grid1d_lat', 'lat', 'LATIXY']:
        if vname in ds:
            lat_var = vname
            break
    for vname in ['grid1d_lon', 'lon', 'LONGXY']:
        if vname in ds:
            lon_var = vname
            break
    if lat_var and lon_var:
        lats_1d = ds[lat_var].values
        lons_1d = ds[lon_var].values
        print(f"Found coordinates: {lat_var}, {lon_var}")
        print(f"Lat range: [{lats_1d.min():.2f}, {lats_1d.max():.2f}]")
        print(f"Lon range: [{lons_1d.min():.2f}, {lons_1d.max():.2f}]")
    else:
        raise ValueError(f"Cannot find lat/lon variables. Available: {list(ds.coords)} {list(ds.data_vars)[:30]}")
else:
    # Check all dimensions
    print(f"All dims: {dict(ds.dims)}")
    # Might have 2D lat/lon
    if 'lat' in ds and 'lon' in ds:
        lats = ds['lat'].values
        lons = ds['lon'].values
        if lats.ndim == 1 and lons.ndim == 1:
            print(f"1D lat/lon found")
        else:
            print(f"2D lat/lon: lat shape {lats.shape}, lon shape {lons.shape}")
    else:
        is_unstructured = True
        grid_dim = list(ds.dims.keys())[0] if 'time' not in list(ds.dims.keys())[0] else list(ds.dims.keys())[1]
        for vname in ['grid1d_lat', 'lat', 'LATIXY']:
            if vname in ds:
                lat_var = vname
                break
        for vname in ['grid1d_lon', 'lon', 'LONGXY']:
            if vname in ds:
                lon_var = vname
                break
        lats_1d = ds[lat_var].values
        lons_1d = ds[lon_var].values

# Get area weights
area_var = None
for vname in ['area', 'AREA', 'grid1d_area', 'Area']:
    if vname in ds:
        area_var = vname
        break

landfrac_var = None
for vname in ['landfrac', 'LANDFRAC', 'landmask']:
    if vname in ds:
        landfrac_var = vname
        break

print(f"Area variable: {area_var}")
print(f"Landfrac variable: {landfrac_var}")

# Convert units: QRUNOFF is in mm/s, convert to mm/day for display
conversion_factor = 86400.0  # s/day
unit_label = "mm/day"

qrunoff_mean_mmday = qrunoff_mean * conversion_factor

# Compute area-weighted global mean
try:
    if is_unstructured:
        data_vals = qrunoff_mean_mmday.values
        valid = np.isfinite(data_vals)

        if area_var:
            area_weights = ds[area_var].values
            if landfrac_var:
                landfrac = ds[landfrac_var].values
                if landfrac.ndim > 1:
                    landfrac = landfrac[0]  # take first time step if time-varying
                weights = area_weights * landfrac
            else:
                weights = area_weights
        else:
            # Use cos(lat) as approximate weight
            weights = np.cos(np.deg2rad(lats_1d))

        weights_valid = weights[valid]
        data_valid = data_vals[valid]
        global_mean = np.sum(data_valid * weights_valid) / np.sum(weights_valid)
    else:
        # Structured grid
        data_vals = qrunoff_mean_mmday.values

        if area_var:
            area_weights = ds[area_var].values
            if landfrac_var:
                landfrac = ds[landfrac_var].values
                if landfrac.ndim > 2:
                    landfrac = landfrac[0]
                weights = area_weights * landfrac
            else:
                weights = area_weights
        else:
            # cos(lat) weighting
            if isinstance(lats, np.ndarray) and lats.ndim == 1:
                weights = np.cos(np.deg2rad(lats))[:, np.newaxis] * np.ones((1, len(lons)))
            else:
                weights = np.cos(np.deg2rad(lats))

        if landfrac_var and area_var is None:
            landfrac = ds[landfrac_var].values
            if landfrac.ndim > 2:
                landfrac = landfrac[0]
            weights = weights * landfrac

        valid = np.isfinite(data_vals)
        if weights.shape != data_vals.shape:
            print(f"Shape mismatch: weights {weights.shape}, data {data_vals.shape}")
            # Try to broadcast
            if weights.ndim < data_vals.ndim:
                weights = np.broadcast_to(weights, data_vals.shape)

        global_mean = np.nansum(data_vals * weights) / np.nansum(weights * valid)

    print(f"Area-weighted global mean QRUNOFF: {global_mean:.4f} {unit_label}")
except Exception as e:
    print(f"Error computing area-weighted mean: {e}")
    global_mean = float(np.nanmean(qrunoff_mean_mmday.values))
    print(f"Simple global mean QRUNOFF (fallback): {global_mean:.4f} {unit_label}")

# Also compute simple stats
data_for_stats = qrunoff_mean_mmday.values
global_max = float(np.nanmax(data_for_stats))
global_min = float(np.nanmin(data_for_stats))
global_median = float(np.nanmedian(data_for_stats))
simple_mean = float(np.nanmean(data_for_stats))

print(f"\nGlobal Statistics (1985-1989 climatological mean QRUNOFF):")
print(f"  Area-weighted mean: {global_mean:.4f} {unit_label}")
print(f"  Simple mean:        {simple_mean:.4f} {unit_label}")
print(f"  Median:             {global_median:.4f} {unit_label}")
print(f"  Min:                {global_min:.4f} {unit_label}")
print(f"  Max:                {global_max:.4f} {unit_label}")

# Save statistics to CSV
try:
    import pandas as pd
    stats_df = pd.DataFrame({
        'Statistic': ['Area-weighted mean', 'Simple mean', 'Median', 'Min', 'Max'],
        'Value (mm/day)': [global_mean, simple_mean, global_median, global_min, global_max]
    })
    stats_csv = os.path.join(output_dir, "qrunoff_clim_mean_1985_1989_stats.csv")
    stats_df.to_csv(stats_csv, index=False, float_format='%.6f')
    print(f"\nSaved statistics to: {stats_csv}")
except Exception as e:
    print(f"Error saving stats CSV: {e}")

# Save mean field to NetCDF
try:
    ds_out = qrunoff_mean_mmday.to_dataset(name='QRUNOFF_clim_mean')
    ds_out.attrs['description'] = 'Climatological mean total runoff (QRUNOFF) from E3SM ELM, 1985-1989'
    ds_out.attrs['units'] = unit_label
    ds_out.attrs['area_weighted_global_mean'] = f'{global_mean:.6f} {unit_label}'
    nc_out = os.path.join(output_dir, "qrunoff_clim_mean_1985_1989.nc")
    ds_out.to_netcdf(nc_out)
    print(f"Saved NetCDF to: {nc_out}")
except Exception as e:
    print(f"Error saving NetCDF: {e}")

# Create map
try:
    fig = plt.figure(figsize=(14, 8))
    ax = fig.add_subplot(1, 1, 1, projection=ccrs.Robinson())
    ax.set_global()
    ax.add_feature(cfeature.COASTLINE, linewidth=0.5, color='gray')
    ax.add_feature(cfeature.BORDERS, linewidth=0.3, color='gray')

    # Set up colormap
    vmin = 0
    vmax_val = min(np.nanpercentile(data_for_stats[data_for_stats > 0], 98) if np.any(data_for_stats > 0) else 5.0, 10.0)
    cmap = plt.cm.YlGnBu

    if is_unstructured:
        # For unstructured grid, use scatter plot
        vals = qrunoff_mean_mmday.values
        valid_mask = np.isfinite(vals)

        # Adjust lons to [-180, 180] if needed
        plot_lons = lons_1d.copy()
        if np.any(plot_lons > 180):
            plot_lons[plot_lons > 180] -= 360

        sc = ax.scatter(
            plot_lons[valid_mask], lats_1d[valid_mask],
            c=vals[valid_mask],
            s=0.5,
            cmap=cmap,
            vmin=vmin, vmax=vmax_val,
            transform=ccrs.PlateCarree(),
            rasterized=True
        )
        cbar = plt.colorbar(sc, ax=ax, orientation='horizontal', pad=0.05, shrink=0.7, aspect=40)
    else:
        # Structured grid - use pcolormesh
        plot_lons = lons.copy()
        if plot_lons.ndim == 1:
            lon2d, lat2d = np.meshgrid(lons, lats)
        else:
            lon2d, lat2d = lons, lats

        vals = qrunoff_mean_mmday.values

        # Shift to -180 to 180 if needed
        if lon2d.ndim == 2:
            pc = ax.pcolormesh(
                lon2d, lat2d, vals,
                cmap=cmap, vmin=vmin, vmax=vmax_val,
                transform=ccrs.PlateCarree(),
                rasterized=True
            )
        else:
            pc = ax.pcolormesh(
                lons, lats, vals,
                cmap=cmap, vmin=vmin, vmax=vmax_val,
                transform=ccrs.PlateCarree(),
                rasterized=True,
                shading='auto'
            )
        cbar = plt.colorbar(pc, ax=ax, orientation='horizontal', pad=0.05, shrink=0.7, aspect=40)

    cbar.set_label(f'QRUNOFF ({unit_label})', fontsize=12)

    # Title and stats text
    ax.set_title('E3SM ELM Climatological Mean Total Runoff (QRUNOFF)\n1985-1989', fontsize=14, fontweight='bold')

    stats_text = (
        f"Area-weighted global mean: {global_mean:.4f} {unit_label}\n"
        f"Simple mean: {simple_mean:.4f} {unit_label}\n"
        f"Median: {global_median:.4f} {unit_label}\n"
        f"Min: {global_min:.4f} / Max: {global_max:.2f} {unit_label}"
    )
    ax.text(
        0.02, 0.02, stats_text,
        transform=ax.transAxes,
        fontsize=9,
        verticalalignment='bottom',
        bbox=dict(boxstyle='round,pad=0.5', facecolor='white', alpha=0.85, edgecolor='gray')
    )

    plt.tight_layout()
    map_path = os.path.join(output_dir, "qrunoff_clim_mean_1985_1989_map.png")
    fig.savefig(map_path, dpi=200, bbox_inches='tight')
    plt.close(fig)
    print(f"Saved map to: {map_path}")
except Exception as e:
    print(f"Error creating map: {e}")
    import traceback
    traceback.print_exc()

# Close dataset
ds.close()

print("\nDone! All outputs saved to:", output_dir)
```
