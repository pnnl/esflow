import os
import glob
import numpy as np
import xarray as xr
import matplotlib.pyplot as plt
import cartopy.crs as ccrs
import cartopy.feature as cfeature

# Output directory
output_dir = "/Users/zhou014/Local_Drive/Git_repo/esmflow/benchmark/results/claude-opus-4-6_baseline/task_02_seasonal_runoff/run1_output"
os.makedirs(output_dir, exist_ok=True)

# Data paths
data_dir = "./data/sample/e3sm/lnd"
case_name = "sample.v3.LR.historical"

# Gather all ELM monthly files for 1985-1989
file_pattern = os.path.join(data_dir, f"{case_name}.elm.h0.*.nc")
all_files = sorted(glob.glob(file_pattern))

# Filter to 1985-1989
years = range(1985, 1990)
selected_files = []
for f in all_files:
    basename = os.path.basename(f)
    parts = basename.split(".")
    date_part = parts[-2]  # e.g., "1985-01"
    year = int(date_part.split("-")[0])
    if year in years:
        selected_files.append(f)

print(f"Found {len(selected_files)} ELM files for 1985-1989")
if len(selected_files) == 0:
    print("No files found! Checking available files...")
    print(f"Total files matching pattern: {len(all_files)}")
    if all_files:
        print(f"First file: {all_files[0]}")
        print(f"Last file: {all_files[-1]}")

# Load the dataset
try:
    ds = xr.open_mfdataset(selected_files, combine='by_coords', decode_times=True)
    print(f"Dataset loaded successfully. Variables: {list(ds.data_vars)[:20]}")
    print(f"Dimensions: {dict(ds.sizes)}")
except Exception as e:
    print(f"Error loading dataset: {e}")
    raise

# Extract QRUNOFF (total runoff, mm/s)
try:
    qrunoff = ds['QRUNOFF']
    print(f"QRUNOFF shape: {qrunoff.shape}")
    print(f"QRUNOFF units: {qrunoff.attrs.get('units', 'unknown')}")
    print(f"QRUNOFF long_name: {qrunoff.attrs.get('long_name', 'unknown')}")
except KeyError:
    print("QRUNOFF not found. Available variables:")
    for v in sorted(ds.data_vars):
        print(f"  {v}")
    raise

# Compute the climatological (time) mean of QRUNOFF
qrunoff_mean = qrunoff.mean(dim='time')
print(f"Mean QRUNOFF shape: {qrunoff_mean.shape}")

# Convert from mm/s to mm/day for more intuitive units
mm_s_to_mm_day = 86400.0
qrunoff_mean_mmday = qrunoff_mean * mm_s_to_mm_day

# Get area weights for global mean
# ELM has 'area' and 'landfrac' - take first time step or mean over time
if 'area' in ds:
    area = ds['area']
    print(f"Area variable found, shape: {area.shape}, units: {area.attrs.get('units', 'unknown')}")
    # Remove time dimension if present
    if 'time' in area.dims:
        area = area.isel(time=0, drop=True)
        print(f"  -> Selected first time step, new shape: {area.shape}")
elif 'AREA' in ds:
    area = ds['AREA']
    if 'time' in area.dims:
        area = area.isel(time=0, drop=True)
else:
    area = None
    print("No area variable found in dataset")

if 'landfrac' in ds:
    landfrac = ds['landfrac']
    print(f"landfrac found, shape: {landfrac.shape}")
    if 'time' in landfrac.dims:
        landfrac = landfrac.isel(time=0, drop=True)
        print(f"  -> Selected first time step, new shape: {landfrac.shape}")
elif 'LANDFRAC' in ds:
    landfrac = ds['LANDFRAC']
    if 'time' in landfrac.dims:
        landfrac = landfrac.isel(time=0, drop=True)
else:
    landfrac = None
    print("No landfrac variable found")

# Compute area-weighted global mean
if area is not None and landfrac is not None:
    weights = area * landfrac
elif area is not None:
    weights = area
else:
    weights = None

if weights is not None:
    q_flat = qrunoff_mean_mmday.values.flatten()
    w_flat = weights.values.flatten()
    
    print(f"q_flat shape: {q_flat.shape}, w_flat shape: {w_flat.shape}")
    
    valid = np.isfinite(q_flat) & np.isfinite(w_flat) & (w_flat > 0)
    
    global_mean_mmday = np.average(q_flat[valid], weights=w_flat[valid])
    global_mean_mms = global_mean_mmday / mm_s_to_mm_day
    global_std_mmday = np.sqrt(np.average((q_flat[valid] - global_mean_mmday)**2, weights=w_flat[valid]))
else:
    global_mean_mmday = float(np.nanmean(qrunoff_mean_mmday.values))
    global_mean_mms = global_mean_mmday / mm_s_to_mm_day
    global_std_mmday = float(np.nanstd(qrunoff_mean_mmday.values))

print(f"\nGlobal area-weighted mean QRUNOFF: {global_mean_mmday:.4f} mm/day ({global_mean_mms:.6e} mm/s)")
print(f"Global area-weighted std QRUNOFF: {global_std_mmday:.4f} mm/day")
print(f"Min QRUNOFF: {float(np.nanmin(qrunoff_mean_mmday.values)):.4f} mm/day")
print(f"Max QRUNOFF: {float(np.nanmax(qrunoff_mean_mmday.values)):.4f} mm/day")

# Save statistics to CSV
try:
    import pandas as pd
    stats = pd.DataFrame({
        'Statistic': ['Global Area-Weighted Mean (mm/day)',
                       'Global Area-Weighted Mean (mm/s)',
                       'Global Std (mm/day)',
                       'Min (mm/day)',
                       'Max (mm/day)',
                       'Period',
                       'N_months'],
        'Value': [f"{global_mean_mmday:.4f}",
                  f"{global_mean_mms:.6e}",
                  f"{global_std_mmday:.4f}",
                  f"{float(np.nanmin(qrunoff_mean_mmday.values)):.4f}",
                  f"{float(np.nanmax(qrunoff_mean_mmday.values)):.4f}",
                  "1985-1989",
                  f"{len(selected_files)}"]
    })
    stats_file = os.path.join(output_dir, "global_runoff_statistics.csv")
    stats.to_csv(stats_file, index=False)
    print(f"Statistics saved to {stats_file}")
except Exception as e:
    print(f"Error saving statistics: {e}")

# Now create the map
try:
    plot_lat = ds['lat'].values
    plot_lon = ds['lon'].values
    plot_data = qrunoff_mean_mmday.values.copy()

    # Handle lon convention: shift from 0-360 to -180-180 if needed
    if np.any(plot_lon > 180):
        shift_idx = np.argsort(np.where(plot_lon > 180, plot_lon - 360, plot_lon))
        plot_lon = np.where(plot_lon > 180, plot_lon - 360, plot_lon)
        plot_lon = plot_lon[shift_idx]
        plot_data = plot_data[:, shift_idx]

    # Create the map figure
    fig = plt.figure(figsize=(14, 8))
    ax = fig.add_subplot(1, 1, 1, projection=ccrs.Robinson())

    ax.set_global()
    ax.add_feature(cfeature.COASTLINE, linewidth=0.5)
    ax.add_feature(cfeature.BORDERS, linewidth=0.3, linestyle='--')

    lon2d, lat2d = np.meshgrid(plot_lon, plot_lat)

    # Use a nice colormap and sensible levels
    levels = np.array([0, 0.1, 0.2, 0.5, 1.0, 2.0, 3.0, 5.0, 8.0, 12.0])

    im = ax.contourf(
        lon2d, lat2d, plot_data,
        levels=levels,
        cmap='YlGnBu',
        extend='max',
        transform=ccrs.PlateCarree()
    )

    # Add colorbar
    cbar = plt.colorbar(im, ax=ax, orientation='horizontal', pad=0.05, shrink=0.7, aspect=40)
    cbar.set_label('Total Runoff (mm/day)', fontsize=12)

    # Title
    ax.set_title('E3SM (ELM) Climatological Mean Total Runoff (QRUNOFF)\n1985–1989',
                 fontsize=14, fontweight='bold')

    # Overlay global statistics as text
    stats_text = (
        f"Global Area-Weighted Mean: {global_mean_mmday:.3f} mm/day\n"
        f"Std Dev: {global_std_mmday:.3f} mm/day\n"
        f"Min: {float(np.nanmin(qrunoff_mean_mmday.values)):.3f}, "
        f"Max: {float(np.nanmax(qrunoff_mean_mmday.values)):.3f} mm/day"
    )

    ax.text(
        0.02, 0.02, stats_text,
        transform=ax.transAxes,
        fontsize=10,
        verticalalignment='bottom',
        bbox=dict(boxstyle='round,pad=0.5', facecolor='white', alpha=0.85, edgecolor='gray')
    )

    plt.tight_layout()

    map_file = os.path.join(output_dir, "qrunoff_climatology_map_1985_1989.png")
    fig.savefig(map_file, dpi=200, bbox_inches='tight')
    print(f"Map saved to {map_file}")
    plt.close(fig)

except Exception as e:
    print(f"Error creating map: {e}")
    import traceback
    traceback.print_exc()

# Save the climatological mean field as NetCDF
try:
    out_ds = xr.Dataset()

    out_ds['QRUNOFF_mean_mmday'] = xr.DataArray(
        qrunoff_mean_mmday.values,
        dims=['lat', 'lon'],
        coords={'lat': ds['lat'].values, 'lon': ds['lon'].values},
        attrs={'units': 'mm/day', 'long_name': 'Climatological mean total runoff 1985-1989'}
    )

    out_ds.attrs['description'] = 'E3SM ELM climatological mean QRUNOFF for 1985-1989'
    out_ds.attrs['global_mean_mmday'] = float(global_mean_mmday)
    out_ds.attrs['global_std_mmday'] = float(global_std_mmday)
    out_ds.attrs['source'] = f'E3SM case: {case_name}'

    nc_file = os.path.join(output_dir, "qrunoff_climatology_1985_1989.nc")
    out_ds.to_netcdf(nc_file)
    print(f"NetCDF saved to {nc_file}")
except Exception as e:
    print(f"Error saving NetCDF: {e}")
    import traceback
    traceback.print_exc()

print("\nDone!")
