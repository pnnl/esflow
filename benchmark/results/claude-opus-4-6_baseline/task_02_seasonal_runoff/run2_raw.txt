

```python
import os
import glob
import numpy as np
import xarray as xr
import matplotlib.pyplot as plt
import cartopy.crs as ccrs
import cartopy.feature as cfeature

# Output directory
output_dir = "/Users/zhou014/Local_Drive/Git_repo/esmflow/benchmark/results/claude-opus-4-6_baseline/task_02_seasonal_runoff/run2_output"
os.makedirs(output_dir, exist_ok=True)

# Data paths
data_dir = "./data/sample/e3sm/lnd"
case_name = "sample.v3.LR.historical"

# Gather all ELM monthly files for 1985-1989
file_list = []
for year in range(1985, 1990):
    for month in range(1, 13):
        fname = os.path.join(data_dir, f"{case_name}.elm.h0.{year:04d}-{month:02d}.nc")
        if os.path.exists(fname):
            file_list.append(fname)

file_list.sort()
print(f"Found {len(file_list)} ELM monthly files for 1985-1989")

if len(file_list) == 0:
    # Try glob as fallback
    pattern = os.path.join(data_dir, f"{case_name}.elm.h0.198[5-9]-*.nc")
    file_list = sorted(glob.glob(pattern))
    print(f"Glob fallback found {len(file_list)} files")

if len(file_list) == 0:
    raise FileNotFoundError(f"No ELM files found in {data_dir}")

# Load data
try:
    ds = xr.open_mfdataset(file_list, combine='by_coords', data_vars='minimal',
                           coords='minimal', compat='override')
    print(f"Dataset loaded successfully")
    print(f"Dimensions: {dict(ds.dims)}")
    print(f"Available variables (subset): {[v for v in ds.data_vars if 'RUNOFF' in v.upper() or 'QRUNOFF' in v.upper() or v in ['area', 'landfrac', 'landmask']]}")
except Exception as e:
    print(f"Error loading dataset: {e}")
    raise

# Extract QRUNOFF
if 'QRUNOFF' not in ds:
    print("Available variables:", list(ds.data_vars))
    raise KeyError("QRUNOFF not found in dataset")

qrunoff = ds['QRUNOFF']
print(f"QRUNOFF shape: {qrunoff.shape}")
print(f"QRUNOFF units: {qrunoff.attrs.get('units', 'unknown')}")
print(f"QRUNOFF long_name: {qrunoff.attrs.get('long_name', 'unknown')}")

# Get coordinates
if 'lat' in ds:
    lat = ds['lat']
    lon = ds['lon']
elif 'lsmlat' in ds.dims:
    lat = ds['lsmlat']
    lon = ds['lsmlon']
else:
    # Check coordinate names
    print(f"Coordinates: {list(ds.coords)}")
    for coord_name in ['lat', 'latitude', 'LATIXY', 'latixy']:
        if coord_name in ds:
            lat = ds[coord_name]
            break
    for coord_name in ['lon', 'longitude', 'LONGXY', 'longxy']:
        if coord_name in ds:
            lon = ds[coord_name]
            break

print(f"Lat shape: {lat.shape}, Lon shape: {lon.shape}")

# Compute climatological mean (time average over 1985-1989)
qrunoff_mean = qrunoff.mean(dim='time')
print(f"Mean QRUNOFF shape: {qrunoff_mean.shape}")

# Convert units: mm/s to mm/day for more intuitive display
mm_per_s_to_mm_per_day = 86400.0
qrunoff_mean_mmday = qrunoff_mean * mm_per_s_to_mm_per_day

# Get area weights
# ELM typically has 'area' (in km^2 or radians^2) and 'landfrac'
area = None
landfrac = None

for area_name in ['area', 'AREA']:
    if area_name in ds:
        area = ds[area_name]
        print(f"Found area variable '{area_name}', shape: {area.shape}, units: {area.attrs.get('units', 'unknown')}")
        break

for lf_name in ['landfrac', 'LANDFRAC', 'landmask']:
    if lf_name in ds:
        landfrac = ds[lf_name]
        print(f"Found landfrac variable '{lf_name}', shape: {landfrac.shape}")
        break

# Compute area-weighted global mean
if area is not None:
    # Remove time dimension from area/landfrac if present
    if 'time' in area.dims:
        area = area.isel(time=0)
    if landfrac is not None and 'time' in landfrac.dims:
        landfrac = landfrac.isel(time=0)

    if landfrac is not None:
        weights = area * landfrac
    else:
        weights = area

    # Mask where qrunoff is valid
    valid = np.isfinite(qrunoff_mean.values)
    if hasattr(qrunoff_mean, 'values'):
        qr_vals = qrunoff_mean.values.copy()
    else:
        qr_vals = np.array(qrunoff_mean)

    w_vals = weights.values.copy() if hasattr(weights, 'values') else np.array(weights)

    # Handle grid dimensions
    if qr_vals.ndim != w_vals.ndim:
        print(f"Dimension mismatch: qrunoff {qr_vals.ndim}D, weights {w_vals.ndim}D")
        # If weights are 2D and qrunoff is 1D (unstructured), flatten weights
        if w_vals.ndim > qr_vals.ndim:
            w_vals = w_vals.ravel()

    valid = np.isfinite(qr_vals) & np.isfinite(w_vals) & (w_vals > 0)
    global_mean_mms = np.average(qr_vals[valid], weights=w_vals[valid])
    global_mean_mmday = global_mean_mms * mm_per_s_to_mm_per_day
    total_weight = np.sum(w_vals[valid])
else:
    # Fallback: use cos(lat) weighting
    print("No area variable found, using cos(lat) weighting")
    lat_vals = lat.values
    if lat_vals.ndim == 1 and qrunoff_mean.ndim == 1:
        # Unstructured grid
        cos_weights = np.cos(np.deg2rad(lat_vals))
    elif lat_vals.ndim == 2:
        cos_weights = np.cos(np.deg2rad(lat_vals))
    elif lat_vals.ndim == 1 and qrunoff_mean.ndim == 2:
        cos_weights = np.cos(np.deg2rad(lat_vals))[:, np.newaxis] * np.ones((1, qrunoff_mean.shape[1]))
    else:
        cos_weights = np.cos(np.deg2rad(lat_vals))

    qr_vals = qrunoff_mean.values
    if cos_weights.shape != qr_vals.shape:
        cos_weights = cos_weights.ravel()
        qr_vals_flat = qr_vals.ravel()
    else:
        qr_vals_flat = qr_vals.ravel()
        cos_weights = cos_weights.ravel()

    valid = np.isfinite(qr_vals_flat) & (cos_weights > 0)
    global_mean_mms = np.average(qr_vals_flat[valid], weights=cos_weights[valid])
    global_mean_mmday = global_mean_mms * mm_per_s_to_mm_per_day
    total_weight = np.sum(cos_weights[valid])

print(f"\nGlobal area-weighted mean QRUNOFF: {global_mean_mms:.6e} mm/s = {global_mean_mmday:.4f} mm/day")

# Also compute basic stats
qr_all = qrunoff_mean.values.ravel()
qr_valid = qr_all[np.isfinite(qr_all)]
global_max_mmday = np.max(qr_valid) * mm_per_s_to_mm_per_day
global_min_mmday = np.min(qr_valid) * mm_per_s_to_mm_per_day
global_std_mmday = np.std(qr_valid) * mm_per_s_to_mm_per_day

print(f"Max QRUNOFF: {global_max_mmday:.4f} mm/day")
print(f"Min QRUNOFF: {global_min_mmday:.6f} mm/day")
print(f"Std QRUNOFF: {global_std_mmday:.4f} mm/day")

# Save statistics to CSV
try:
    import pandas as pd
    stats_df = pd.DataFrame({
        'Statistic': ['Global Area-Weighted Mean (mm/s)',
                       'Global Area-Weighted Mean (mm/day)',
                       'Global Max (mm/day)',
                       'Global Min (mm/day)',
                       'Global Std (mm/day)',
                       'Number of grid cells',
                       'Period'],
        'Value': [f"{global_mean_mms:.6e}",
                  f"{global_mean_mmday:.4f}",
                  f"{global_max_mmday:.4f}",
                  f"{global_min_mmday:.6f}",
                  f"{global_std_mmday:.4f}",
                  f"{len(qr_valid)}",
                  "1985-1989"]
    })
    stats_csv = os.path.join(output_dir, "global_runoff_statistics.csv")
    stats_df.to_csv(stats_csv, index=False)
    print(f"\nSaved statistics to {stats_csv}")
except Exception as e:
    print(f"Error saving statistics CSV: {e}")

# Save mean runoff field as NetCDF
try:
    out_ds = xr.Dataset()
    out_ds['QRUNOFF_mean'] = qrunoff_mean
    out_ds['QRUNOFF_mean'].attrs['units'] = 'mm/s'
    out_ds['QRUNOFF_mean'].attrs['long_name'] = 'Climatological mean total runoff (1985-1989)'
    out_ds['QRUNOFF_mean_mmday'] = qrunoff_mean_mmday
    out_ds['QRUNOFF_mean_mmday'].attrs['units'] = 'mm/day'
    out_ds['QRUNOFF_mean_mmday'].attrs['long_name'] = 'Climatological mean total runoff (1985-1989)'
    out_nc = os.path.join(output_dir, "mean_runoff_1985_1989.nc")
    out_ds.to_netcdf(out_nc)
    print(f"Saved NetCDF to {out_nc}")
except Exception as e:
    print(f"Error saving NetCDF: {e}")

# Create map
try:
    fig = plt.figure(figsize=(14, 8))
    ax = fig.add_subplot(1, 1, 1, projection=ccrs.Robinson())
    ax.set_global()
    ax.add_feature(cfeature.COASTLINE, linewidth=0.5)
    ax.add_feature(cfeature.BORDERS, linewidth=0.3, linestyle='--')

    # Get lat/lon values
    lat_vals = lat.values
    lon_vals = lon.values

    # Convert runoff to mm/day for plotting
    plot_data = qrunoff_mean_mmday.values

    # Determine if data is on structured or unstructured grid
    is_structured = (lat_vals.ndim == 2) or (plot_data.ndim == 2 and lat_vals.ndim == 1)
    is_unstructured = (lat_vals.ndim == 1 and plot_data.ndim == 1)

    if is_unstructured:
        # Scatter plot for unstructured grid
        valid_mask = np.isfinite(plot_data)
        lons_plot = lon_vals[valid_mask]
        lats_plot = lat_vals[valid_mask]
        data_plot = plot_data[valid_mask]

        # Clip extreme values for better visualization
        vmax = np.percentile(data_plot[data_plot > 0], 95) if np.any(data_plot > 0) else 5.0
        vmin = 0

        sc = ax.scatter(lons_plot, lats_plot,
                        c=data_plot, cmap='YlGnBu',
                        s=0.5, vmin=vmin, vmax=vmax,
                        transform=ccrs.PlateCarree(),
                        rasterized=True)
    elif is_structured and plot_data.ndim == 2:
        # Pcolormesh for structured 2D grid
        if lat_vals.ndim == 1:
            lon_2d, lat_2d = np.meshgrid(lon_vals, lat_vals)
        else:
            lon_2d, lat_2d = lon_vals, lat_vals

        vmax = np.nanpercentile(plot_data[plot_data > 0], 95) if np.any(plot_data > 0) else 5.0
        vmin = 0

        sc = ax.pcolormesh(lon_2d, lat_2d, plot_data,
                           cmap='YlGnBu', vmin=vmin, vmax=vmax,
                           transform=ccrs.PlateCarree(),
                           rasterized=True)
    else:
        # Try regridding unstructured to regular grid for visualization
        from scipy.interpolate import griddata
        valid_mask = np.isfinite(plot_data.ravel())
        lons_flat = lon_vals.ravel()[valid_mask]
        lats_flat = lat_vals.ravel()[valid_mask]
        data_flat = plot_data.ravel()[valid_mask]

        # Create regular grid
        grid_lon = np.linspace(-180, 180, 720)
        grid_lat = np.linspace(-90, 90, 360)
        grid_lon_2d, grid_lat_2d = np.meshgrid(grid_lon, grid_lat)

        # Adjust longitudes if necessary
        if np.min(lons_flat) >= 0:
            lons_flat = np.where(lons_flat > 180, lons_flat - 360, lons_flat)

        grid_data = griddata((lons_flat, lats_flat), data_flat,
                             (grid_lon_2d, grid_lat_2d), method='nearest')

        vmax = np.nanpercentile(data_flat[data_flat > 0], 95) if np.any(data_flat > 0) else 5.0
        vmin = 0

        sc = ax.pcolormesh(grid_lon_2d, grid_lat_2d, grid_data,
                           cmap='YlGnBu', vmin=vmin, vmax=vmax,
                           transform=ccrs.PlateCarree(),
                           rasterized=True)

    # Colorbar
    cbar = plt.colorbar(sc, ax=ax, orientation='horizontal', pad=0.05, shrink=0.7, aspect=40)
    cbar.set_label('Total Runoff (mm/day)', fontsize=12)

    # Title
    ax.set_title('E3SM ELM Climatological Mean Total Runoff (QRUNOFF)\n1985-1989', fontsize=14, fontweight='bold')

    # Overlay global statistics as text
    stats_text = (
        f"Global Area-Weighted Mean: {global_mean_mmday:.3f} mm/day\n"
        f"Max: {global_max_mmday:.2f} mm/day | Min: {global_min_mmday:.4f} mm/day\n"
        f"Std Dev: {global_std_mmday:.3f} mm/day | Grid cells: {len(qr_valid)}"
    )
    ax.text(0.02, 0.02, stats_text,
            transform=ax.transAxes,
            fontsize=9, verticalalignment='bottom',
            bbox=dict(boxstyle='round,pad=0.5', facecolor='white', alpha=0.85, edgecolor='gray'))

    plt.tight_layout()
    map_file = os.path.join(output_dir, "mean_runoff_map_1985_1989.png")
    fig.savefig(map_file, dpi=200, bbox_inches='tight')
    plt.close(fig)
    print(f"Saved map to {map_file}")

except Exception as e:
    print(f"Error creating map: {e}")
    import traceback
    traceback.print_exc()

print("\nDone! All outputs saved to:", output_dir)
```
