name: "Task02 Seasonal Runoff"
description: |
  Extract and map the climatological mean runoff (QRUNOFF) from ELM output for 1985-1989.

settings:
  case_name: "sample.v3.LR.historical"
  data_dir: "./data/sample/e3sm"
  output_dir: "./output/task02_seasonal_runoff"
  style: cream_ink

steps:
  - id: extract_runoff_field
    tool: extract_gridded_field
    params:
      data_dir: "${settings.data_dir}"
      case_name: "${settings.case_name}"
      component: "elm"
      variable: "QRUNOFF"
      years: [1985,1986,1987,1988,1989]
    outputs:
      field_file: runoff_mean_1985_1989.nc

  - id: compute_global_runoff_mean
    tool: compute_zonal_stats
    params:
      field_file: "${extract_runoff_field.outputs.field_file}"
    outputs:
      stats_file: runoff_global_stats.csv

  - id: plot_runoff_map
    tool: plot_gridded_map
    params:
      field_file: "${extract_runoff_field.outputs.field_file}"
      title: "Mean Runoff (1985-1989)"
      stats_file: "${compute_global_runoff_mean.outputs.stats_file}"
    outputs:
      plot_file: runoff_mean_map.png
