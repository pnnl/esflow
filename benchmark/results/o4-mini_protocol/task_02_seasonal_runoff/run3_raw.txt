name: "Seasonal Runoff Climatology"
description: |
  Extracts the climatological mean runoff (QRUNOFF) from ELM monthly output for 1985-1989,
  computes the global area-weighted mean, and produces a map with global statistics.

settings:
  case_name: "sample.v3.LR.historical"
  data_dir: "./data/sample/e3sm"
  output_dir: "./output/task02_seasonal_runoff"
  style: cream_ink

steps:
  - id: extract_runoff
    tool: extract_gridded_field
    params:
      data_dir: "${settings.data_dir}"
      case_name: "${settings.case_name}"
      component: "elm"
      variable: "QRUNOFF"
      years: [1985, 1986, 1987, 1988, 1989]
    outputs:
      field_file: "${settings.output_dir}/runoff_1985_1989.nc"

  - id: compute_global_mean
    tool: compute_zonal_stats
    params:
      field_file: "${extract_runoff.outputs.field_file}"
      lat_bands: ""
    outputs:
      stats_file: "${settings.output_dir}/global_runoff_stats.csv"

  - id: plot_runoff_map
    tool: plot_gridded_map
    params:
      field_file: "${extract_runoff.outputs.field_file}"
      title: "Mean Runoff 1985-1989"
      stats_file: "${compute_global_mean.outputs.stats_file}"
    outputs:
      plot_file: "${settings.output_dir}/runoff_map.png"
