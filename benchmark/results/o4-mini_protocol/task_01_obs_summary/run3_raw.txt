name: "Extract and summarize observed streamflow"
description: |
  This workflow loads GRDC gauge metadata, extracts monthly streamflow observations
  for selected gauges from 1985 to 1989, and computes summary statistics for each gauge.
settings:
  case_name: "task01"
  data_dir: "./data"
  output_dir: "./output/task01_obs_summary"
  style: cream_ink
steps:
  - id: load_obs_metadata
    tool: load_obs_metadata
    params:
      metadata_file: "${settings.data_dir}/sample/obs/gauge_metadata.csv"
    outputs:
      metadata_file: "${settings.output_dir}/gauge_metadata_validated.csv"
      n_gauges: "${settings.output_dir}/n_gauges.txt"
  - id: extract_obs_timeseries
    tool: extract_obs_timeseries
    params:
      obs_dir: "${settings.data_dir}/sample/obs/streamflow"
      gauge_metadata: "${load_obs_metadata.outputs.metadata_file}"
      years: [1985, 1986, 1987, 1988, 1989]
      gauge_ids: "3629000,4121801,4115200,6742900,2969100,1159100"
    outputs:
      timeseries_file: "${settings.output_dir}/obs_timeseries.csv"
      n_gauges: "${settings.output_dir}/n_extracted_gauges.txt"
  - id: compute_summary_stats
    tool: compute_summary_stats
    params:
      timeseries_file: "${extract_obs_timeseries.outputs.timeseries_file}"
      gauge_metadata: "${load_obs_metadata.outputs.metadata_file}"
    outputs:
      stats_file: "${settings.output_dir}/summary_stats.csv"
      n_columns: "${settings.output_dir}/n_columns.txt"
