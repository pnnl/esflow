name: "Streamflow Observation Summary Statistics"
description: |
  Load GRDC gauge metadata, extract monthly streamflow observations for six selected gauges from 1985 to 1989, and compute summary statistics (mean, std, min, max).

settings:
  metadata_file: "./data/sample/obs/gauge_metadata.csv"
  obs_dir: "./data/sample/obs/streamflow"
  output_dir: "./output/task01_obs_summary"
  style: cream_ink

steps:
  - id: load_obs_metadata
    tool: load_obs_metadata
    params:
      metadata_file: "${settings.metadata_file}"
    outputs:
      metadata_file: "gauge_metadata_valid.csv"
      n_gauges: "n_gauges.txt"

  - id: extract_obs_timeseries
    tool: extract_obs_timeseries
    params:
      obs_dir: "${settings.obs_dir}"
      gauge_metadata: "${load_obs_metadata.outputs.metadata_file}"
      years: [1985, 1986, 1987, 1988, 1989]
      gauge_ids: "3629000,4121801,4115200,6742900,2969100,1159100"
    outputs:
      timeseries_file: "obs_timeseries_1985_1989.csv"
      n_gauges: "extracted_n_gauges.txt"

  - id: compute_summary_stats
    tool: compute_summary_stats
    params:
      timeseries_file: "${extract_obs_timeseries.outputs.timeseries_file}"
      gauge_metadata: "${load_obs_metadata.outputs.metadata_file}"
      top_n: 0
      rank_by: "mean"
    outputs:
      stats_file: "obs_summary_stats.csv"
      n_columns: "n_columns.txt"
