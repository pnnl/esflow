name: "Task06 Water Balance Closure"
description: |
  Compute monthly climatological mean fields for precipitation, evapotranspiration, and runoff from ELM over 1985-1989.
  Compute global area-weighted mean for each component.
  Compute the spatial residual (P − ET − Q).
  Extract basin-mean values for six major basins and plot a composite water balance figure.
settings:
  case_name: "sample.v3.LR.historical"
  data_dir: "./data/sample/e3sm"
  output_dir: "./output/task06_water_balance"
  style: cream_ink

steps:
  - id: extract_precip
    tool: extract_gridded_field
    params:
      data_dir: "${settings.data_dir}"
      case_name: "${settings.case_name}"
      component: "elm"
      variable: "RAIN+SNOW"
      years: [1985,1989]
    outputs:
      field_file: "${settings.output_dir}/precip_mean.nc"

  - id: extract_et
    tool: extract_gridded_field
    params:
      data_dir: "${settings.data_dir}"
      case_name: "${settings.case_name}"
      component: "elm"
      variable: "QVEGE+QVEGT+QSOIL"
      years: [1985,1989]
    outputs:
      field_file: "${settings.output_dir}/et_mean.nc"

  - id: extract_runoff
    tool: extract_gridded_field
    params:
      data_dir: "${settings.data_dir}"
      case_name: "${settings.case_name}"
      component: "elm"
      variable: "QRUNOFF"
      years: [1985,1989]
    outputs:
      field_file: "${settings.output_dir}/runoff_mean.nc"

  - id: extract_etq
    tool: extract_gridded_field
    params:
      data_dir: "${settings.data_dir}"
      case_name: "${settings.case_name}"
      component: "elm"
      variable: "QVEGE+QVEGT+QSOIL+QRUNOFF"
      years: [1985,1989]
    outputs:
      field_file: "${settings.output_dir}/etq_mean.nc"

  - id: global_precip_stats
    tool: compute_zonal_stats
    params:
      field_file: "${extract_precip.outputs.field_file}"
    outputs:
      stats_file: "${settings.output_dir}/precip_global_stats.csv"

  - id: global_et_stats
    tool: compute_zonal_stats
    params:
      field_file: "${extract_et.outputs.field_file}"
    outputs:
      stats_file: "${settings.output_dir}/et_global_stats.csv"

  - id: global_runoff_stats
    tool: compute_zonal_stats
    params:
      field_file: "${extract_runoff.outputs.field_file}"
    outputs:
      stats_file: "${settings.output_dir}/runoff_global_stats.csv"

  - id: compute_residual
    tool: compute_spatial_bias
    params:
      field_a: "${extract_precip.outputs.field_file}"
      field_b: "${extract_etq.outputs.field_file}"
    outputs:
      bias_file: "${settings.output_dir}/residual_field.nc"
      stats_file: "${settings.output_dir}/residual_global_stats.csv"

  - id: basin_precip
    tool: extract_basin_mean
    params:
      field_file: "${extract_precip.outputs.field_file}"
      basins_file: "./data/sample/obs/basin_polygons.geojson"
      gauge_ids: "3629000,4121801,4115200,6742900,2969100,1159100"
    outputs:
      basin_means_file: "${settings.output_dir}/basin_precip.csv"

  - id: basin_et
    tool: extract_basin_mean
    params:
      field_file: "${extract_et.outputs.field_file}"
      basins_file: "./data/sample/obs/basin_polygons.geojson"
      gauge_ids: "3629000,4121801,4115200,6742900,2969100,1159100"
    outputs:
      basin_means_file: "${settings.output_dir}/basin_et.csv"

  - id: basin_runoff
    tool: extract_basin_mean
    params:
      field_file: "${extract_runoff.outputs.field_file}"
      basins_file: "./data/sample/obs/basin_polygons.geojson"
      gauge_ids: "3629000,4121801,4115200,6742900,2969100,1159100"
    outputs:
      basin_means_file: "${settings.output_dir}/basin_runoff.csv"

  - id: basin_residual
    tool: extract_basin_mean
    params:
      field_file: "${compute_residual.outputs.bias_file}"
      basins_file: "./data/sample/obs/basin_polygons.geojson"
      gauge_ids: "3629000,4121801,4115200,6742900,2969100,1159100"
    outputs:
      basin_means_file: "${settings.output_dir}/basin_residual.csv"

  - id: plot_water_balance
    tool: plot_water_balance_basins
    params:
      residual_file: "${compute_residual.outputs.bias_file}"
      basin_precip_file: "${basin_precip.outputs.basin_means_file}"
      basin_et_file: "${basin_et.outputs.basin_means_file}"
      basin_runoff_file: "${basin_runoff.outputs.basin_means_file}"
      basin_residual_file: "${basin_residual.outputs.basin_means_file}"
      basins_file: "./data/sample/obs/basin_polygons.geojson"
      gauge_ids: "3629000,4121801,4115200,6742900,2969100,1159100"
      locations_file: "./data/sample/obs/gauge_metadata.csv"
    outputs:
      plot_file: "${settings.output_dir}/water_balance_basins.png"
