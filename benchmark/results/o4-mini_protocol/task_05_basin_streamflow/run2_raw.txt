name: "Basin-Scale Streamflow Analysis"
description: |
  Perform basin-scale streamflow analysis for six major river basins (Amazon, Missouri,
  Columbia, Danube, Mekong, Orange) over the period 1985-1989. Match observation
  gauges to the E3SM MOSART grid, extract simulated and observed river discharge,
  compute validation metrics, and generate per-basin time series plots.
settings:
  case_name: "sample.v3.LR.historical"
  data_dir: "./data/sample/e3sm/"
  output_dir: "./output/task05_basin_streamflow"
  style: cream_ink
  metadata_file: "./data/sample/obs/gauge_metadata.csv"
  obs_dir: "./data/sample/obs/streamflow/"
  basins_file: "./data/sample/obs/basin_polygons.geojson"
  years: [1985,1986,1987,1988,1989]
  variable: "RIVER_DISCHARGE_OVER_LAND_LIQ"
  component: "mosart"
  gauge_ids: "3629000,4121801,4115200,6742900,2969100,1159100"
steps:
  - id: load_metadata
    tool: load_obs_metadata
    params:
      metadata_file: "${settings.metadata_file}"
    outputs:
      metadata_file: "${settings.output_dir}/metadata_validated.csv"
      n_gauges: n_gauges
  - id: match_gauges
    tool: match_to_grid
    params:
      gauge_metadata: "${load_metadata.outputs.metadata_file}"
      data_dir: "${settings.data_dir}"
      case_name: "${settings.case_name}"
      component: "${settings.component}"
    outputs:
      matched_file: "${settings.output_dir}/matched_gauges.csv"
      n_matched: n_matched
  - id: extract_sim
    tool: extract_e3sm_timeseries
    params:
      data_dir: "${settings.data_dir}"
      case_name: "${settings.case_name}"
      component: "${settings.component}"
      variable: "${settings.variable}"
      locations_file: "${match_gauges.outputs.matched_file}"
      years: "${settings.years}"
    outputs:
      timeseries_file: "${settings.output_dir}/sim_timeseries.csv"
      n_locations: n_locations
  - id: extract_obs
    tool: extract_obs_timeseries
    params:
      obs_dir: "${settings.obs_dir}"
      gauge_metadata: "${load_metadata.outputs.metadata_file}"
      years: "${settings.years}"
      gauge_ids: "${settings.gauge_ids}"
    outputs:
      timeseries_file: "${settings.output_dir}/obs_timeseries.csv"
      n_gauges: n_gauges
  - id: compute_stats
    tool: compute_metrics
    params:
      sim_file: "${extract_sim.outputs.timeseries_file}"
      obs_file: "${extract_obs.outputs.timeseries_file}"
    outputs:
      metrics_file: "${settings.output_dir}/metrics.csv"
      n_gauges: n_gauges
  - id: plot_basins
    tool: plot_basin_timeseries
    params:
      basins_file: "${settings.basins_file}"
      sim_file: "${extract_sim.outputs.timeseries_file}"
      obs_file: "${extract_obs.outputs.timeseries_file}"
      locations_file: "${load_metadata.outputs.metadata_file}"
      gauge_ids: "${settings.gauge_ids}"
      metrics_file: "${compute_stats.outputs.metrics_file}"
    outputs:
      plot_files:
        - "${settings.output_dir}/basin_timeseries"
