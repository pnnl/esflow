name: "Per-Basin Integrated Water Cycle Evaluation"
description: |
  This workflow performs a per-basin integrated water cycle evaluation for E3SM over the period 1985-1989.
  It extracts climatological time-mean model fields (precipitation, evapotranspiration, runoff),
  fetches and processes corresponding observation datasets, computes basin-averaged values for six major river basins,
  evaluates streamflow distributional metrics via FDC comparisons, combines all biases into a water budget summary,
  and visualizes the results with bar and radar charts.
settings:
  case_name: "sample.v3.LR.historical"
  data_dir: "./data/sample/e3sm"
  output_dir: "./output/task07_integrated_diagnostic"
  style: cream_ink
steps:
  - id: extract_model_precip
    tool: extract_gridded_field
    params:
      data_dir: "${settings.data_dir}"
      case_name: "${settings.case_name}"
      component: "elm"
      variable: "RAIN+SNOW"
      years: [1985, 1986, 1987, 1988, 1989]
    outputs:
      field_file: "model_precip_1985_1989.nc"

  - id: extract_model_et
    tool: extract_gridded_field
    params:
      data_dir: "${settings.data_dir}"
      case_name: "${settings.case_name}"
      component: "elm"
      variable: "QVEGE+QVEGT+QSOIL"
      years: [1985, 1986, 1987, 1988, 1989]
    outputs:
      field_file: "model_et_1985_1989.nc"

  - id: extract_model_runoff
    tool: extract_gridded_field
    params:
      data_dir: "${settings.data_dir}"
      case_name: "${settings.case_name}"
      component: "elm"
      variable: "QRUNOFF"
      years: [1985, 1986, 1987, 1988, 1989]
    outputs:
      field_file: "model_runoff_1985_1989.nc"

  - id: fetch_obs_precip
    tool: fetch_ilamb_data
    params:
      variable: "pr"
      dataset: "GPCCv2018"
    outputs:
      data_file: "obs_pr_GPCCv2018.nc"

  - id: extract_obs_precip
    tool: extract_gridded_field
    params:
      obs_file: "${fetch_obs_precip.outputs.data_file}"
      variable: "pr"
      years: [1985, 1986, 1987, 1988, 1989]
    outputs:
      field_file: "obs_precip_1985_1989.nc"

  - id: fetch_obs_et
    tool: fetch_ilamb_data
    params:
      variable: "evspsbl"
      dataset: "MODIS"
    outputs:
      data_file: "obs_evspsbl_MODIS.nc"

  - id: extract_obs_et
    tool: extract_gridded_field
    params:
      obs_file: "${fetch_obs_et.outputs.data_file}"
      variable: "et"
      years: [1985, 1986, 1987, 1988, 1989]
    outputs:
      field_file: "obs_et_1985_1989.nc"

  - id: fetch_obs_runoff
    tool: fetch_ilamb_data
    params:
      variable: "mrro"
      dataset: "LORA"
    outputs:
      data_file: "obs_mrro_LORA.nc"

  - id: extract_obs_runoff
    tool: extract_gridded_field
    params:
      obs_file: "${fetch_obs_runoff.outputs.data_file}"
      variable: "mrro"
      years: [1985, 1986, 1987, 1988, 1989]
    outputs:
      field_file: "obs_runoff_1985_1989.nc"

  - id: basin_mean_model_precip
    tool: extract_basin_mean
    params:
      field_file: "${extract_model_precip.outputs.field_file}"
      basins_file: "./data/sample/obs/basin_polygons.geojson"
      gauge_ids: "3629000,4121801,4115200,6742900,2969100,1159100"
    outputs:
      basin_means_file: "model_precip_basin_means.csv"

  - id: basin_mean_model_et
    tool: extract_basin_mean
    params:
      field_file: "${extract_model_et.outputs.field_file}"
      basins_file: "./data/sample/obs/basin_polygons.geojson"
      gauge_ids: "3629000,4121801,4115200,6742900,2969100,1159100"
    outputs:
      basin_means_file: "model_et_basin_means.csv"

  - id: basin_mean_model_runoff
    tool: extract_basin_mean
    params:
      field_file: "${extract_model_runoff.outputs.field_file}"
      basins_file: "./data/sample/obs/basin_polygons.geojson"
      gauge_ids: "3629000,4121801,4115200,6742900,2969100,1159100"
    outputs:
      basin_means_file: "model_runoff_basin_means.csv"

  - id: basin_mean_obs_precip
    tool: extract_basin_mean
    params:
      field_file: "${extract_obs_precip.outputs.field_file}"
      basins_file: "./data/sample/obs/basin_polygons.geojson"
      gauge_ids: "3629000,4121801,4115200,6742900,2969100,1159100"
    outputs:
      basin_means_file: "obs_precip_basin_means.csv"

  - id: basin_mean_obs_et
    tool: extract_basin_mean
    params:
      field_file: "${extract_obs_et.outputs.field_file}"
      basins_file: "./data/sample/obs/basin_polygons.geojson"
      gauge_ids: "3629000,4121801,4115200,6742900,2969100,1159100"
    outputs:
      basin_means_file: "obs_et_basin_means.csv"

  - id: basin_mean_obs_runoff
    tool: extract_basin_mean
    params:
      field_file: "${extract_obs_runoff.outputs.field_file}"
      basins_file: "./data/sample/obs/basin_polygons.geojson"
      gauge_ids: "3629000,4121801,4115200,6742900,2969100,1159100"
    outputs:
      basin_means_file: "obs_runoff_basin_means.csv"

  - id: load_metadata
    tool: load_obs_metadata
    params:
      metadata_file: "./data/sample/obs/gauge_metadata.csv"
    outputs:
      metadata_file: "validated_gauge_metadata.csv"
      n_gauges: n_gauges

  - id: match_gauges
    tool: match_to_grid
    params:
      gauge_metadata: "${load_metadata.outputs.metadata_file}"
      data_dir: "${settings.data_dir}"
      case_name: "${settings.case_name}"
      component: "mosart"
    outputs:
      matched_file: "matched_gauges.csv"
      n_matched: n_matched

  - id: extract_sim_streamflow
    tool: extract_e3sm_timeseries
    params:
      data_dir: "${settings.data_dir}"
      case_name: "${settings.case_name}"
      component: "mosart"
      variable: "RIVER_DISCHARGE_OVER_LAND_LIQ"
      locations_file: "${match_gauges.outputs.matched_file}"
      years: [1985, 1986, 1987, 1988, 1989]
    outputs:
      timeseries_file: "sim_streamflow_1985_1989.csv"
      n_locations: n_locations

  - id: extract_obs_streamflow
    tool: extract_obs_timeseries
    params:
      obs_dir: "./data/sample/obs/streamflow/"
      gauge_metadata: "${load_metadata.outputs.metadata_file}"
      years: [1985, 1986, 1987, 1988, 1989]
    outputs:
      timeseries_file: "obs_streamflow_1985_1989.csv"
      n_gauges: n_gauges_obs

  - id: compute_streamflow_fdc
    tool: compute_fdc_metrics
    params:
      sim_file: "${extract_sim_streamflow.outputs.timeseries_file}"
      obs_file: "${extract_obs_streamflow.outputs.timeseries_file}"
    outputs:
      metrics_file: "streamflow_fdc_metrics.csv"
      fdc_file: "streamflow_fdc_data.csv"

  - id: compute_budget
    tool: compute_basin_budget
    params:
      model_precip_file: "${basin_mean_model_precip.outputs.basin_means_file}"
      model_et_file: "${basin_mean_model_et.outputs.basin_means_file}"
      model_runoff_file: "${basin_mean_model_runoff.outputs.basin_means_file}"
      obs_precip_file: "${basin_mean_obs_precip.outputs.basin_means_file}"
      obs_et_file: "${basin_mean_obs_et.outputs.basin_means_file}"
      obs_runoff_file: "${basin_mean_obs_runoff.outputs.basin_means_file}"
      fdc_metrics_file: "${compute_streamflow_fdc.outputs.metrics_file}"
      locations_file: "${load_metadata.outputs.metadata_file}"
    outputs:
      budget_file: "basin_water_budget.csv"

  - id: plot_budget_comparison
    tool: plot_basin_budget_comparison
    params:
      budget_file: "${compute_budget.outputs.budget_file}"
    outputs:
      plot_file: "basin_budget_comparison.png"

  - id: plot_radar
    tool: plot_basin_radar
    params:
      budget_file: "${compute_budget.outputs.budget_file}"
    outputs:
      plot_file: "basin_radar_chart.png"
