name: "ET Benchmark ELM vs MODIS (1985-1989)"
description: |
  This workflow benchmarks E3SM ELM evapotranspiration against MODIS observations
  by computing the 5-year climatological mean ET from 1985-1989, calculating spatial bias,
  and plotting the bias field.

settings:
  case_name: "sample.v3.LR.historical"
  data_dir: "./data/sample/e3sm"
  output_dir: "./output/task03_et_benchmark"
  style: cream_ink

steps:
  - id: fetch_obs_et
    tool: fetch_ilamb_data
    params:
      variable: "evspsbl"
      dataset: "MODIS"
    outputs:
      data_file: "${settings.output_dir}/modis_evspsbl_MODIS.nc"

  - id: extract_model_et
    tool: extract_gridded_field
    params:
      data_dir: "${settings.data_dir}"
      case_name: "${settings.case_name}"
      component: "elm"
      variable: "QVEGE+QVEGT+QSOIL"
      years: [1985, 1986, 1987, 1988, 1989]
    outputs:
      field_file: "${settings.output_dir}/model_et_1985_1989.nc"

  - id: extract_obs_et_clim
    tool: extract_gridded_field
    params:
      obs_file: "${fetch_obs_et.outputs.data_file}"
      variable: "et"
      years: [1985, 1986, 1987, 1988, 1989]
    outputs:
      field_file: "${settings.output_dir}/obs_et_1985_1989.nc"

  - id: compute_et_bias
    tool: compute_spatial_bias
    params:
      field_a: "${extract_model_et.outputs.field_file}"
      field_b: "${extract_obs_et_clim.outputs.field_file}"
    outputs:
      bias_file: "${settings.output_dir}/et_bias_elm_modis.nc"
      stats_file: "${settings.output_dir}/et_bias_stats.csv"

  - id: plot_et_bias_map
    tool: plot_gridded_map
    params:
      field_file: "${compute_et_bias.outputs.bias_file}"
      title: "ET Bias (ELM − MODIS), 1985-1989"
      stats_file: "${compute_et_bias.outputs.stats_file}"
    outputs:
      plot_file: "${settings.output_dir}/et_bias_map.png"
